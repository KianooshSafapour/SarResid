    49→  if (h && r) return `${faNum(h)} ساعت و ${faNum(r)} دقیقه`
    50→  if (h) return `${faNum(h)} ساعت`
    51→  if (r) return `${faNum(r)} دقیقه`
    52→  return '۰ دقیقه'

   157→/**
   158→ * ورودی دوگانهٔ ساعت + دقیقه (17-E) — مثل کارت زمان‌سنجی کاغذی.
   159→ * هر ستون کارکرد (اضافه‌کاری/جمعه/شب) دو ورودی دارد و مجموع به «دقیقه» ذخیره می‌شود؛
   160→ * چیپ‌ها به مجموع دقیقه اضافه می‌کنند و انتقال دهانه (carry) خودکار است (۱:۴۵ + ۰:۳۰ = ۲:۱۵).
   161→ * سقف здрав‌سنجی: ۱۶ ساعت (۹۶۰ دقیقه) — هرگز ورود داده را مسدود نمی‌کند.
   162→ */
   163→function HMField({ minutes, onMinutes, disabled }: { minutes: number; onMinutes: (v: number) => void; disabled?: boolean }) {
   164→  const h = Math.floor(minutes / 60)
   165→  const m = minutes % 60
   166→  const setH = (v: number) => onMinutes(Math.min(960, v * 60 + m))
   167→  const setM = (v: number) => onMinutes(Math.min(960, h * 60 + v))
   168→  const parse = (s: string) => {
   169→    const c = enDigits(s).replace(/\D/g, '')
   170→    return c === '' ? 0 : Number(c)
   171→  }
   172→  return (
   173→    <div className="flex flex-col items-center gap-0.5">
   174→      <div dir="ltr" className="flex items-center gap-0.5">
   175→        <input type="text" inputMode="numeric" aria-label="ساعت" disabled={disabled}
   176→          value={h === 0 ? '' : faNum(h)} placeholder="۰"
   177→          onChange={(e) => setH(Math.min(16, parse(e.target.value)))}
   178→          className="w-8 rounded-lg border border-input bg-white/90 px-0.5 py-1 text-center text-xs font-bold tabular-nums outline-none focus:border-primary disabled:opacity-50" />
   179→        <span className="text-[10px] font-black text-muted-foreground">:</span>
   180→        <input type="text" inputMode="numeric" aria-label="دقیقه" disabled={disabled}
   181→          value={m === 0 ? '' : faNum(m)} placeholder="۰"
   182→          onChange={(e) => setM(Math.min(59, parse(e.target.value)))}
   183→          className="w-9 rounded-lg border border-input bg-white/90 px-0.5 py-1 text-center text-xs font-bold tabular-nums outline-none focus:border-primary disabled:opacity-50" />
   184→      </div>
   185→      <span className="h-3 text-[9px] font-bold leading-3 tabular-nums text-muted-foreground" title={hmLong(minutes)}>
   186→        {minutes > 0 ? hm(minutes) : ''}
   187→      </span>
   188→    </div>
   189→  )
   190→}
