   165→/** برچسب ناوبری چندزبانه — در نبود ترجمه، برچسب فارسی اصلی */
   166→function navLabel(key: string, faLabel: string, lang: string): string {
   167→  if (lang === 'fa') return faLabel
   168→  const v = tI18n(`nav.${key}`)
   169→  return v.startsWith('nav.') ? faLabel : v
   170→}
   171→function groupTitle(faTitle: string, lang: string): string {
   172→  if (lang === 'fa') return faTitle
   173→  const v = tI18n(GROUP_TITLE_KEYS[faTitle] || '')
   174→  return v.startsWith('nav.') ? faTitle : v
   175→}

   182→function parseHash(): { view: string; param: string } {
   183→  const h = window.location.hash.replace(/^#\/?/, '')

   185→  return { view: view || 'dashboard', param: param || '' }

   196→  const bellRef = useRef<HTMLButtonElement>(null)
   197→  const [notifs, setNotifs] = useState<{ id: string; severity: string; icon: string; title: string; detail: string; go: string }[]>([])
   198→  const [snoozedUntil, setSnoozedUntil] = useState('')
   199→  const [socket, setSocket] = useState<Socket | null>(null)
   200→  const [extraViews, setExtraViews] = useState<string[]>([]) // بخش‌هایی که نقش‌های سفارشی باز می‌کنند
   201→  const [msgTick, setMsgTick] = useState(0)
   202→  const socketRef = useRef<Socket | null>(null)
   203→  // round-16: نصب/سندباکس/لایسنس
   204→  const [setupDone, setSetupDone] = useState<boolean | null>(null) // null = در حال خواندن
   205→  const [sandbox, setSandbox] = useState(false)
   206→  const [businessName, setBusinessName] = useState('هایپر زیتون')
   207→  const [license, setLicense] = useState<LicenseState | null>(null)
   208→  const [deskNotif, setDeskNotif] = useState<boolean>(
   209→    () => typeof window !== 'undefined' && 'Notification' in window && localStorage.getItem('hz-desktop-notifs') === '1'
   210→  )
   211→  const [deskPerm, setDeskPerm] = useState<string>(() =>
   212→    typeof window !== 'undefined' && 'Notification' in window ? Notification.permission : 'unsupported'
   213→  ) // 'granted' | 'denied' | 'default' | 'unsupported'
   214→
   215→  /* اعلان دسکتاپ — وقتی تب در پس‌زمینه است، اعلان‌های فوری را با Notification API نشان بده.
   216→   * dedupe با نقشه id→count در localStorage؛ وقتی تب دیده می‌شود فقط وضعیت را به‌روز می‌کنیم (تُست خودش خبر می‌دهد). */
   217→  const fireDesktopNotifs = useCallback((list: { id: string; severity: string; icon: string; title: string; detail: string; count?: number }[]) => {
   218→    try {
   219→      if (!('Notification' in window) || Notification.permission !== 'granted') return
   220→      const enabled = localStorage.getItem('hz-desktop-notifs') === '1'
   221→      const seen: Record<string, number> = JSON.parse(localStorage.getItem('hz-desktop-seen') || '{}')
   222→      const hidden = document.visibilityState === 'hidden'
   223→      for (const n of list) {
   224→        const cur = n.count ?? 1
   225→        const wasSeen = seen[n.id]
   226→        const isNew = wasSeen === undefined || cur > wasSeen
   227→        if (enabled && hidden && isNew && n.severity === 'critical') {
   228→          try {
   229→            const notif = new Notification(`${businessName} • سررسید`, {
   230→              body: `${n.icon} ${n.title}\n${n.detail.slice(0, 110)}`,
   231→              tag: n.id,
   232→            })
   233→            notif.onclick = () => {
   234→              window.focus()
   235→              notif.close()
   236→            }
   237→          } catch {
   238→            /* some engines throw on constructor — ignore */
   239→          }
   240→        }
   241→        seen[n.id] = cur
   242→      }
   243→      // prune keys that no longer exist
   244→      const live = new Set(list.map((n) => n.id))
   245→      for (const k of Object.keys(seen)) if (!live.has(k)) delete seen[k]
   246→      localStorage.setItem('hz-desktop-seen', JSON.stringify(seen))
   247→    } catch {
   248→      /* storage unavailable — silent */
   249→    }
   250→  }, [businessName])
   251→
   252→  const toggleDeskNotif = useCallback(async () => {
   253→    if (!('Notification' in window)) return toast.info('مرورگر شما از اعلان دسکتاپ پشتیبانی نمی‌کند')
   254→    if (Notification.permission === 'denied') return toast.error('اجازه اعلان داده نشده — از نوار آدرس مرورگر (🔒) اجازه دهید')
   255→    if (deskNotif) {
   256→      localStorage.setItem('hz-desktop-notifs', '0')
   257→      setDeskNotif(false)
   258→      toast.info('اعلان دسکتاپ خاموش شد — اعلان‌ها فقط داخل سامانه می‌آیند')
   259→      return
   260→    }
   261→    try {
   262→      const perm = await Notification.requestPermission()
   263→      setDeskPerm(perm)
   264→      if (perm === 'granted') {
   265→        localStorage.setItem('hz-desktop-notifs', '1')
   266→        setDeskNotif(true)
   267→        toast.success('اعلان دسکتاپ روشن شد — وقتی سامانه در پس‌زمینه است، فوری‌ها 🚨 روی دسکتاپ می‌آیند')
   268→        try {
   269→          const t = new Notification(`${businessName} • سررسید`, { body: 'اعلان دسکتاپ فعال شد ✅ — این یک پیام آزمایشی است' })
   270→          t.onclick = () => {
   271→            window.focus()
   272→            t.close()
   273→          }
   274→        } catch {
   275→          /* ignore */
   276→        }
   277→      } else {
   278→        toast.info('برای فعال‌سازی، اجازه اعلان را در مرورگر تأیید کنید')
   279→      }
   280→    } catch {
   281→      toast.error('درخواست اجازه اعلان ناموفق بود')
   282→    }
   283→  }, [deskNotif, businessName])
   284→
   285→  useEffect(() => {
   286→    api<{ user: AppCtx['user']; extraViews?: string[]; token?: string }>('/api/auth/me')
   287→      .then((d) => {
   288→        if (d.token) setSessionToken(d.token) // self-heal session for cookie-blocked mobile webviews
   289→        setUser(d.user)
   290→        setExtraViews(d.extraViews || [])
   291→      })
   292→      .catch(() => {})
   293→      .finally(() => setLoading(false))
   294→    const applyHash = () => setNav(parseHash())
   295→    const raf = requestAnimationFrame(applyHash)
   296→    window.addEventListener('hashchange', applyHash)
   297→    return () => {
   298→      cancelAnimationFrame(raf)
   299→      window.removeEventListener('hashchange', applyHash)
   300→    }
   301→  }, [])
   302→
   303→  /* round-16: وضعیت نصب/سندباکس/لایسنس — پیش از لاگین هم لازم است (جادوگر) */
   304→  const refreshSetup = useCallback(() => {

   312→      .catch(() => setSetupDone(true)) // خطای شبکه → فرض نصب‌شده تا جادوگر بی‌جهت نمایش داده نشود
   313→  }, [])
   314→  useEffect(() => {
   315→    refreshSetup()
   316→  }, [refreshSetup])
   317→
   318→  // websocket for real-time messaging (URL configurable for local server deployments)
   319→  useEffect(() => {
   320→    if (!user || socketRef.current) return
   321→    const sockUrl = process.env.NEXT_PUBLIC_SOCKET_URL || '/?XTransformPort=3003'
   322→    const s = io(sockUrl, {
   323→      transports: ['websocket', 'polling'],
   324→      reconnection: true,
   325→      reconnectionAttempts: 8,
   326→      reconnectionDelay: 1500,
   327→    })
   328→    socketRef.current = s
   329→    s.on('connect', () => {
   330→      s.emit('hz-join', { userId: user.id, name: user.name })

   340→      s.disconnect()

   363→    return () => window.removeEventListener('keydown', onKey)
   364→  }, [])
   365→
   366→  const refreshNotifications = useCallback(async () => {

   385→      toast.error(e.message)
   386→    }
   387→  }
   388→

   390→    try {
   391→      await api('/api/notif-prefs', { method: 'PUT', body: { snoozeUntil: '' } })

   393→      refreshNotifications()

   395→      toast.error(e.message)
   396→    }
   397→  }
   398→
   399→  useEffect(() => {
   400→    if (!user) return

   403→    const t = setInterval(tick, 45000)
   404→    return () => {
   405→      clearTimeout(t0)
   406→      clearInterval(t)
   407→    }
   408→  }, [user, refreshNotifications, msgTick])
   409→
   410→  const logout = async () => {
   411→    try { await api('/api/auth/logout', { method: 'POST' }) } catch { /* still clear locally */ }
   412→    clearSessionToken()
   413→    socketRef.current?.disconnect()
   414→    socketRef.current = null
   415→    setSocket(null)
   416→    setUser(null)
   417→    navigate('dashboard')
   418→  }
   419→
   420→  const ctx: AppCtx = useMemo(
   421→    () => ({ user, navigate, view: nav.view, param: nav.param, refreshNotifications, socket, socketReady: !!socket }),
   422→    [user, navigate, nav, refreshNotifications, socket]
   423→  )
   424→
   425→  if (loading) {
   426→    return (
   427→      <div className="pattern-girih flex min-h-screen items-center justify-center">
   428→        <div className="flex flex-col items-center gap-4">
   429→          <div className="h-14 w-14 animate-spin rounded-full border-4 border-[#c9a227]/30 border-t-[#c9a227]" />
   430→          <p className="text-sm font-bold text-[#e9f0e4]">در حال بارگذاری سامانه…</p>
   431→        </div>
   432→      </div>
   433→    )
