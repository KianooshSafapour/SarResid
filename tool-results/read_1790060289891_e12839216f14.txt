    61→  // ── مودال‌ها و حالت‌های قبلی (همه حفظ شده) ──

   175→            if (!alive) return
   176→            setModeRows(all)

   178→            const all = await collectPages('&lowstock=1', 3)
   179→            if (!alive) return
   180→            setModeRows(all.filter((p) => p.stock <= p.reorderLevel))
   181→          }
   182→        } catch (e: any) {
   183→          if (alive) setModeErr(e.message || 'خطا در دریافت فهرست کالاها')

   185→          if (alive) setModeBusy(false)
   186→        }
   187→      })()
   188→    }, 0)

   190→      alive = false
   191→      clearTimeout(t)

   239→

   244→        subtitle="دسته‌بندی هوشمند + صفحه‌بندی سرور (آمادهٔ ده‌ها هزار کالا) — رنگ هر کارت وضعیت موجودی است: قرمز کمبود جدی، زرد رو به اتمام، سبز مناسب"
   245→        icon={<ShoppingBasket size={18} />}

   247→          canEdit && (

'border-[#c96f4a] bg-[#c96f4a] text-white shadow-md' : 'border-border bg-card hover:border-[#c96f4a]/50')} title="کالاهایی که در ۳۰ روز اخیر ۵ عدد یا بیشتر فروخته‌اند (روی همین صفحه)">
   304→                پرفروش‌ها 🔥

   306→              {data && (

   326→            {loadErr && (

   330→            {/* ── شبکهٔ کالاها ── */}

   704→/* ───────────────── Smart reorder: low-stock → DRAFT orders per provider ───────────────── */

   706→type ProviderRow = { id: string; name: string }
   707→
   708→function SmartReorderModal({ onClose, onDone }: { onClose: () => void; onDone: (created: number) => void }) {
   709→  const [products, setProducts] = useState<Product[]>([])
   710→  const [loaded, setLoaded] = useState(false)
   711→  const [providers, setProviders] = useState<ProviderRow[]>([])
   712→  const [qtys, setQtys] = useState<Record<string, number>>({})
   713→  const [excluded, setExcluded] = useState<Record<string, boolean>>({}) // by providerId
   714→  const [skipped, setSkipped] = useState<Record<string, boolean>>({}) // per product
   715→  const [busy, setBusy] = useState(false)
   716→
   717→  // فهرست کم‌موجودی‌ها از سرور (پارامتر lowstock=1 — مقیاس‌پذیر برای ۱۰هزار کالا)
   718→  useEffect(() => {
   719→    api<{ products: Product[] }>('/api/products?lowstock=1&page=1&pageSize=500')
   720→      .then((d) => { setProducts(d.products); setLoaded(true) })
   721→      .catch(() => setLoaded(true))
   722→    api<{ providers: ProviderRow[] }>('/api/providers').then((d) => setProviders(d.providers)).catch(() => {})
   723→  }, [])
   724→
   725→  const low = useMemo(() => products.filter((p) => p.stock <= p.reorderLevel), [products])
   726→  const suggested = (p: Product) => Math.max(p.reorderLevel * 2 - p.stock, 1)
   727→
   728→  // initialize suggestion quantities once
   729→  useEffect(() => {
   730→    if (Object.keys(qtys).length === 0 && low.length > 0) {
   731→      const init: Record<string, number> = {}
   732→      for (const p of low) init[p.id] = suggested(p)
   733→      setQtys(init)
   734→    }
   735→  }, [low.length])
   736→
   737→  const groups = useMemo(() => {
   738→    const m = new Map<string, Product[]>()
   739→    for (const p of low) {
   740→      const key = p.providerId || '_none'
   741→      if (!m.has(key)) m.set(key, [])
   742→      m.get(key)!.push(p)
   743→    }
   744→    return [...m.entries()]
   745→  }, [low])
   746→
   747→  const activeGroups = groups.filter(([pid]) => !excluded[pid])
   748→  const activeItems = activeGroups.flatMap(([, ps]) => ps).filter((p) => !skipped[p.id])
   749→  const estTotal = activeItems.reduce((s, p) => s + (qtys[p.id] || 0) * p.buyPrice, 0)
   750→
   751→  const create = async () => {
   752→    if (activeItems.length === 0) return
   753→    setBusy(true)
   754→    const d = new Date()
   755→    d.setDate(d.getDate() + 2)
   756→    const deliveryDate = d.toISOString().slice(0, 10)
   757→    let created = 0
   758→    try {
   759→      for (const [pid, ps] of activeGroups) {

  1693→function BarcodeLabel({ name, price, barcode }: { name: string; price: number; barcode: string }) {
  1694→  const svgRef = useRef<SVGSVGElement>(null)
  1695→  useEffect(() => {
  1696→    if (!svgRef.current || !barcode) return
  1697→    const opts = {
  1698→      width: 1.15,
  1699→      height: 30,
  1700→      fontSize: 10,
  1701→      fontOptions: 'bold' as const,
  1702→      margin: 0,
  1703→      displayValue: true,
  1704→    }
  1705→    try {
  1706→      JsBarcode(svgRef.current!, barcode, {
  1707→        ...opts,
  1708→        format: barcode.length === 13 ? 'EAN13' : barcode.length === 8 ? 'EAN8' : 'CODE128',
  1709→      })
  1710→    } catch {
  1711→      // بارکد با چک‌سام نامعتبر — به CODE128 می‌افتیم که هر رشته‌ای را می‌پذیرد
  1712→      try {
  1713→        JsBarcode(svgRef.current!, barcode, { ...opts, format: 'CODE128' })
  1714→      } catch (e) {
  1715→        console.error('barcode render fail', barcode, e)
  1716→      }
  1717→    }
  1718→  }, [barcode])
  1719→  return (
  1720→    <div className="shelf-label overflow-hidden">
  1721→      <p className="truncate text-[11px] font-black leading-4">{name}</p>
  1722→      <div className="flex items-end justify-between gap-1">
  1723→        <p className="shrink-0 text-sm font-black text-[#0e7a4a]">{faMoney(price)} <span className="text-[8px] font-bold">تومان</span></p>
  1724→        {barcode ? <svg ref={svgRef} className="shelf-label-barcode max-w-[60%]" /> : <span className="text-[8px] text-muted-foreground">هایپر زیتون</span>}
  1725→      </div>
  1726→    </div>
  1727→  )
  1728→}

  1738→  const [busy, setBusy] = useState(false)
  1739→  const [result, setResult] = useState<any>(null)
  1740→
  1741→  useEffect(() => {
  1742→    api<{ products: Product[] }>('/api/products').then((d) => setExisting(d.products))
  1743→  }, [])
  1744→
  1745→  const onFile = async (e: React.ChangeEvent<HTMLInputElement>) => {
  1746→    const file = e.target.files?.[0]
  1747→    if (!file) return
  1748→    try {
  1749→      const buf = await file.arrayBuffer()
  1750→      const wb = XLSX.read(buf, { type: 'array' })
  1751→      const ws = wb.Sheets[wb.SheetNames[0]]
  1752→      const json = XLSX.utils.sheet_to_json<any>(ws, { defval: '' })
  1753→      const mapped: ImportRow[] = json.map((r: any) => {
  1754→        // try to map common Holoo columns (Persian/English), with fallbacks
  1755→        const get = (...keys: string[]) => {
  1756→          for (const k of keys) {
  1757→            const found = Object.keys(r).find((rk) => rk.trim() === k || rk.includes(k))
  1758→            if (found && r[found] !== '') return r[found]
  1759→          }
  1760→          return ''
  1761→        }
  1762→        const name = String(get('کالا', 'شرح کالا', 'نام کالا', 'شرح', 'name', 'product', 'کالای موجود') || '').trim()
  1763→        const barcode = String(get('بارکد', 'کد بارکد', 'barcode', 'بارکد کالا') || '').trim()
  1764→        const row: ImportRow = {
  1765→          name,
  1766→          barcode,
  1767→          holooCode: String(get('کد هلو', 'کد کالا', 'کد', 'holooCode', 'code') || ''),

  1770→          stock: Number(String(get('موجودی', 'تعداد', 'stock', 'شمار')).replace(/[^\d.]/g, '')) || 0,
  1771→          unit: String(get('واحد', 'unit') || 'عدد'),
  1772→          brand: String(get('برند', 'برند کالا', 'brand') || ''),
  1773→          category: String(get('دسته', 'گروه', 'category') || 'عمومی'),
  1774→          duplicate: false,
  1775→        }
  1776→        return row
  1777→      }).filter((r: ImportRow) => r.name)
  1778→
  1779→      // duplicate detection
  1780→      const byBarcode = new Map<string, Product>()
  1781→      const byName = new Map<string, Product>()
  1782→      for (const p of existing) {
  1783→        for (const b of p.barcodes) byBarcode.set(b, p)
  1784→        byName.set(p.name.trim().toLowerCase(), p)
  1785→      }
  1786→      for (const r of mapped) {
  1787→        const hit = (r.barcode && byBarcode.get(r.barcode)) || byName.get(r.name.toLowerCase())
  1788→        if (hit) { r.duplicate = true; r.dupWith = hit.name }
  1789→      }
  1790→      setRows(mapped)
  1791→      if (!mapped.length) toast.error('سطر قابل‌نگاشت پیدا نشد — ستون‌های «نام کالا» و «بارکد» را بررسی کنید')
  1792→      else toast.success(`${faNum(mapped.length)} سطر خوانده شد، ${faNum(mapped.filter((x) => x.duplicate).length)} مورد تکراری تشخیص داده شد`)
  1793→    } catch (err: any) {
  1794→      toast.error('خواندن فایل ناموفق بود: ' + err.message)
  1795→    }
  1796→  }
  1797→
  1798→  const doImport = async () => {
  1799→    setBusy(true)
  1800→    try {
  1801→      const res = await api<any>('/api/products/import', { method: 'POST', body: { items: rows, mode } })
  1802→      setResult(res)
  1803→      toast.success(`درون‌ریزی انجام شد: ${faNum(res.created)} جدید، ${faNum(res.updated)} به‌روزرسانی، ${faNum(res.skipped)} ردشد`)
  1804→    } catch (e: any) { toast.error(e.message) } finally { setBusy(false) }
  1805→  }
  1806→
  1807→  return (
  1808→    <Modal title="ورود کالاها از فایل اکسل هلو" onClose={onClose} wide>
  1809→      {!rows.length ? (
  1810→        <div className="rounded-2xl border-2 border-dashed border-[#c9a227]/50 bg-[#fdf6dd]/40 p-8 text-center">
  1811→          <FileSpreadsheet size={40} className="mx-auto mb-3 text-[#8a6d10]" />
  1812→          <p className="text-sm font-black">فایل اکسل خروجی هلو را انتخاب کنید (.xls / .xlsx / .csv)</p>
  1813→          <p className="mx-auto mt-1 max-w-md text-[11px] leading-5 text-muted-foreground">
  1814→            ستون‌های رایج هلو (نام کالا، بارکد، کد، قیمت خرید، قیمت فروش، موجودی) به‌طور خودکار شناسایی می‌شوند و تکراری‌ها مشخص می‌گردند.
  1815→          </p>
  1816→          <label className="mt-4 inline-flex cursor-pointer items-center gap-2 rounded-xl bg-primary px-6 py-3 text-sm font-extrabold text-white">
  1817→            📁 انتخاب فایل اکسل
  1818→            <input type="file" accept=".xls,.xlsx,.csv" hidden onChange={onFile} />
  1819→          </label>
  1820→        </div>
  1821→      ) : result ? (
  1822→        <div className="space-y-3 text-center">
  1823→          <p className="text-3xl">🎉</p>
  1824→          <p className="text-sm font-black">درون‌ریزی کامل شد</p>
  1825→          <div className="grid grid-cols-3 gap-2">
  1826→            <KeyValue k="جدید" v={faNum(result.created)} />
  1827→            <KeyValue k="به‌روزرسانی" v={faNum(result.updated)} />
  1828→            <KeyValue k="ردشده" v={faNum(result.skipped)} />
  1829→          </div>
  1830→          {result.log?.length > 0 && (
  1831→            <div className="scroll-gold max-h-40 overflow-y-auto rounded-xl bg-muted/40 p-3 text-right text-[11px]">
  1832→              {result.log.map((l: string, i: number) => <p key={i}>• {l}</p>)}
  1833→            </div>
  1834→          )}
  1835→          <button onClick={onDone} className="w-full rounded-xl bg-primary py-2.5 text-xs font-extrabold text-white">بستن</button>
  1836→        </div>
  1837→      ) : (
  1838→        <>
  1839→          <div className="mb-3 flex flex-wrap items-center justify-between gap-2">
  1840→            <p className="text-xs font-bold">{faNum(rows.length)} سطر خوانده شد — {faNum(rows.filter((r) => r.duplicate).length)} تکراری</p>
  1841→            <div className="flex gap-1.5">
  1842→              <button onClick={() => setMode('update')} className={cn('rounded-lg px-3 py-1.5 text-[11px] font-bold', mode === 'update' ? 'bg-primary text-white' : 'border')}>به‌روزرسانی تکراری‌ها</button>
  1843→              <button onClick={() => setMode('skip')} className={cn('rounded-lg px-3 py-1.5 text-[11px] font-bold', mode === 'skip' ? 'bg-primary text-white' : 'border')}>ردکردن تکراری‌ها</button>
  1844→            </div>
  1845→          </div>
  1846→          <div className="scroll-gold max-h-72 overflow-y-auto rounded-xl border border-border">
  1847→            <table className="w-full text-right text-[11px]">
  1848→              <thead className="sticky top-0 bg-secondary">
  1849→                <tr>
  1850→                  <th className="p-2">کالا</th><th className="p-2">بارکد</th><th className="p-2">خرید</th><th className="p-2">فروش</th><th className="p-2">موجودی</th><th className="p-2">وضعیت</th>

  1852→              </thead>
  1853→              <tbody>
  1854→                {rows.map((r, i) => (

  1878→function MergeWizard({ onClose, onDone }: { onClose: () => void; onDone: () => void }) {
  1879→  const [products, setProducts] = useState<Product[]>([])
  1880→  const [loaded, setLoaded] = useState(false)
  1881→
  1882→  // فهرست کامل از سرور (تشخیص تکراری سراسری، نه فقط صفحهٔ جاری)
  1883→  useEffect(() => {
  1884→    api<{ products: Product[] }>('/api/products')
  1885→      .then((d) => { setProducts(d.products); setLoaded(true) })
  1886→      .catch(() => setLoaded(true))
  1887→  }, [])
  1888→
  1889→  const groups = useMemo(() => {
  1890→    const byName = new Map<string, Product[]>()
  1891→    for (const p of products) {
  1892→      const key = p.name.replace(/\s+/g, ' ').trim().toLowerCase()
  1893→      const norm = key.replace(/[يیكک]/g, (m) => (m === 'ي' || m === 'ی' ? 'ی' : 'ک'))
  1894→      if (!byName.has(norm)) byName.set(norm, [])
  1895→      byName.get(norm)!.push(p)
  1896→    }
  1897→    return [...byName.values()].filter((g) => g.length > 1)
  1898→  }, [products])
  1899→  const [primary, setPrimary] = useState<Record<string, string>>({})
  1900→
  1901→  const merge = async (groupName: string, ids: string[]) => {
  1902→    const primaryId = primary[groupName] || ids[0]
  1903→    try {
  1904→      await api('/api/products/merge', { method: 'POST', body: { primaryId, duplicateIds: ids.filter((i) => i !== primaryId) } })
  1905→      toast.success('ادغام انجام شد ✅ — موجودی و بارکد‌ها ترکیب شدند')
  1906→      onDone()
  1907→    } catch (e: any) { toast.error(e.message) }
  1908→  }
  1909→
  1910→  return (
  1911→    <Modal title="پاک‌سازی کالاهای تکراری" onClose={onClose} wide>
  1912→      {!loaded ? (
  1913→        <div className="space-y-2">
  1914→          {Array.from({ length: 3 }).map((_, i) => <div key={i} className="h-16 animate-pulse rounded-xl bg-muted/60" />)}
  1915→        </div>
  1916→      ) : groups.length === 0 ? (
  1917→        <div className="py-6 text-center text-sm text-muted-foreground">
  1918→          <p className="text-3xl">✨</p>
  1919→          <p className="mt-2 font-bold">تکراری‌ای پیدا نشد — انبار تمیز است!</p>
  1920→        </div>
  1921→      ) : (
  1922→        <div className="space-y-4">
  1923→          <p className="rounded-xl bg-accent/60 p-3 text-[11px] leading-5">
  1924→            برای هر گروه تکراری، کالای «اصلی» را انتخاب کنید. موجودی‌ها جمع و همه بارکدها زیر کالای اصلی نگه‌داری می‌شوند و تکراری‌ها غیرفعال می‌گردند.
  1925→          </p>
  1926→          <div className="scroll-gold max-h-80 space-y-3 overflow-y-auto">
  1927→            {groups.map((g, gi) => {
  1928→              const key = `g${gi}`
  1929→              return (
  1930→                <div key={key} className="rounded-2xl border border-border p-3">
  1931→                  <p className="mb-2 text-xs font-black">{g[0].name.replace(/\s+/g, ' ').trim()} <span className="text-muted-foreground">({faNum(g.length)} مورد)</span></p>
  1932→                  <div className="grid gap-1.5">
  1933→                    {g.map((p) => (
  1934→                      <label key={p.id} className="flex cursor-pointer items-center gap-2 rounded-xl bg-muted/40 p-2 text-xs">
  1935→                        <input type="radio" name={key} defaultChecked={!primary[key] || primary[key] === p.id} onChange={() => setPrimary((s) => ({ ...s, [key]: p.id }))} />
  1936→                        <span className="flex-1 font-bold">{p.name}</span>
  1937→                        <span className="font-mono text-[10px] text-muted-foreground" dir="ltr">{p.barcodes.join(', ') || '—'}</span>
  1938→                        <span>موجودی: {faNum(p.stock)}</span>
  1939→                      </label>
  1940→                    ))}
  1941→                  </div>
  1942→                  <button onClick={() => merge(key, g.map((p) => p.id))} className="mt-2 rounded-lg bg-[#8a5a2b] px-4 py-1.5 text-[11px] font-extrabold text-white">
  1943→                    ادغام این گروه
  1944→                  </button>
  1945→                </div>
  1946→              )
  1947→            })}
  1948→          </div>
  1949→        </div>
  1950→      )}
  1951→    </Modal>
  1952→  )
