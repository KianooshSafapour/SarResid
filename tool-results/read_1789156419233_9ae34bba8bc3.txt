   111→/** تاریخ امروز به‌همراه تعطیلی (اگر تعطیل است) — چیپ زیبای نوار بالا */
   112→function TopbarDateChip() {
   113→  const holidays = useHolidays()
   114→  const today = todayIso()
   115→  const holidayName = holidays.get(today)
   116→  const isWeekend = [4, 5].includes(new Date().getDay()) // پنجشنبه/جمعه
   117→  return (
   118→    <div
   119→      className={cn(
   120→        'hidden items-center gap-2 rounded-xl border px-3 py-1.5 md:flex',
   121→        holidayName
   122→          ? 'border-[#b3372f]/30 bg-[#fee2e2]/50'
   123→          : isWeekend
   124→            ? 'border-[#c9a227]/40 bg-[#fdf6dd]/60'
   125→            : 'border-border bg-card'
   126→      )}
   127→      title={holidayName || undefined}
   128→    >
   129→      <CalendarDays size={15} className={holidayName || isWeekend ? 'text-[#b3372f]' : 'text-[#0e7a4a]'} />
   130→      <span className="text-[11px] font-extrabold text-foreground">{formatJalaliFull(todayIso())}</span>
   131→      {holidayName && (
   132→        <span className="rounded-lg bg-[#b3372f] px-2 py-0.5 text-[10px] font-black text-white">تعطیل — {holidayName}</span>
   133→      )}
   134→    </div>
   135→  )
   136→}

   138→function parseHash(): { view: string; param: string } {
   139→  const h = window.location.hash.replace(/^#\/?/, '')
   140→  const [view, param] = h.split('/')

   172→        const cur = n.count ?? 1
   173→        const wasSeen = seen[n.id]
   174→        const isNew = wasSeen === undefined || cur > wasSeen

   176→          try {
   177→            const notif = new Notification('هایپر زیتون 🫒', {
   178→              body: `${n.icon} ${n.title}\n${n.detail.slice(0, 110)}`,

   182→              window.focus()
   183→              notif.close()
   184→            }
   185→          } catch {
   186→            /* some engines throw on constructor — ignore */
   187→          }
   188→        }
   189→        seen[n.id] = cur
   190→      }
   191→      // prune keys that no longer exist
   192→      const live = new Set(list.map((n) => n.id))
   193→      for (const k of Object.keys(seen)) if (!live.has(k)) delete seen[k]
   194→      localStorage.setItem('hz-desktop-seen', JSON.stringify(seen))
