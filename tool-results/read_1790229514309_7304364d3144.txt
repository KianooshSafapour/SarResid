   258→  const fireDesktopNotifs = useCallback((list: { id: string; severity: string; icon: string; title: string; detail: string; count?: number }[]) => {
   259→    try {
   260→      if (!('Notification' in window) || Notification.permission !== 'granted') return

   265→        const cur = n.count ?? 1
   266→        const wasSeen = seen[n.id]
