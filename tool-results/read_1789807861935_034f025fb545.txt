   228→/** create / edit user modal */
   229→function UserFormModal({ user, onClose, onSaved }: { user: User | null; onClose: () => void; onSaved: () => void }) {
   230→  const [form, setForm] = useState({
   231→    name: user?.name || '',
   232→    role: user?.role || 'SALES',
   233→    pin: user?.pin || '1234',
   234→    color: user?.color || COLOR_SWATCHES[0],
   235→  })
   236→  const [secondary, setSecondary] = useState<string[]>(user?.secondaryRoles || [])
   237→  const [active, setActive] = useState(user?.active ?? true)
   238→  const [busy, setBusy] = useState(false)
   239→
   240→  const toggleSecondary = (k: string) =>
   241→    setSecondary((s) => (s.includes(k) ? s.filter((x) => x !== k) : [...s, k]))
   242→
   243→  const save = async () => {
   244→    if (!form.name.trim()) return toast.error('نام کاربر الزامی است')
   245→    setBusy(true)
   246→    try {
   247→      if (user) {
   248→        await api(`/api/users/${user.id}`, {
   249→          method: 'PATCH',
   250→          body: { name: form.name.trim(), role: form.role, pin: form.pin, color: form.color, secondaryRoles: secondary, active },
   251→        })
   252→        toast.success('اطلاعات کاربر ذخیره شد ✅')
   253→      } else {
   254→        await api('/api/users', {
   255→          method: 'POST',
   256→          body: { name: form.name.trim(), role: form.role, pin: form.pin || '1234', color: form.color, secondaryRoles: secondary },
   257→        })
   258→        toast.success('کاربر جدید ساخته شد ✅ — پین ورودش را به او بگویید')
   259→      }
   260→      onSaved()
   261→    } catch (e: any) {
   262→      toast.error(e.message)
   263→      setBusy(false)
   264→    }
   265→  }
   266→
   267→  return (
   268→    <Modal title={user ? `ویرایش «${user.name}»` : 'کاربر جدید'} onClose={onClose}>
   269→      <Labeled label="نام و نام خانوادگی *">
   270→        <input
   271→          value={form.name}
   272→          onChange={(e) => setForm({ ...form, name: e.target.value })}
   273→          className="w-full rounded-xl border border-input bg-white/90 p-3 text-sm"
   274→          placeholder="مثلاً: زهرا محمدی"
   275→        />
   276→      </Labeled>
   277→
   278→      <Labeled label="نقش اصلی *">
   279→        <select
   280→          value={form.role}
   281→          onChange={(e) => setForm({ ...form, role: e.target.value })}
   282→          className="w-full rounded-xl border border-input bg-white p-3 text-sm"
   283→        >
   284→          {Object.entries(ROLE_LABELS).map(([k, v]) => (
   285→            <option key={k} value={k}>{v}</option>
   286→          ))}
   287→        </select>
   288→      </Labeled>
   289→
   290→      <Labeled label="نقش‌های دوم (اختیاری)" hint="همکار می‌تواند چند نقش را هم‌زمان داشته باشد">
   291→        <div className="flex flex-wrap gap-1.5">
   292→          {SECONDARY_KEYS.map((k) => (
   293→            <button
   294→              key={k}
   295→              type="button"
   296→              onClick={() => toggleSecondary(k)}
   297→              className={cn(
   298→                'rounded-full border px-3.5 py-2 text-[11px] font-extrabold transition',
   299→                secondary.includes(k)
   300→                  ? 'border-[#77934a] bg-[#77934a] text-white'
   301→                  : 'border-border bg-card text-foreground/70 hover:border-primary/50'
   302→              )}
   303→            >
   304→              {SECONDARY_LABELS[k]}
   305→            </button>
   306→          ))}
   307→        </div>
   308→      </Labeled>
   309→
   310→      <div className="grid grid-cols-2 gap-3">
   311→        <Labeled label="پین ورود" hint="پیش‌فرض ۱۲۳۴">
   312→          <input
   313→            value={form.pin}
   314→            onChange={(e) => setForm({ ...form, pin: e.target.value.replace(/\D/g, '').slice(0, 6) })}
   315→            dir="ltr"
   316→            className="w-full rounded-xl border border-input bg-white/90 p-3 text-center text-sm font-black"
   317→          />
   318→        </Labeled>
   319→        <Labeled label="رنگ آواتار">
   320→          <div className="flex flex-wrap gap-1.5 pt-1">
   321→            {COLOR_SWATCHES.map((c) => (
   322→              <button
   323→                key={c}
   324→                type="button"
   325→                onClick={() => setForm({ ...form, color: c })}
   326→                className={cn('h-8 w-8 rounded-full border-2 transition', form.color === c ? 'scale-110 border-foreground/70' : 'border-transparent')}
   327→                style={{ background: c }}
   328→                title={c}
   329→              />
   330→            ))}
   331→          </div>
   332→        </Labeled>
   333→      </div>
   334→
   335→      {user && (
   336→        <label className="flex cursor-pointer items-center gap-2 rounded-xl bg-muted/40 px-3 py-2.5">
   337→          <input type="checkbox" checked={active} onChange={(e) => setActive(e.target.checked)} className="h-4 w-4 accent-[#0e7a4a]" />
   338→          <span className="text-xs font-bold">کاربر فعال است (اجازهٔ ورود دارد)</span>
   339→        </label>
   340→      )}
   341→
   342→      <button onClick={save} disabled={busy} className="w-full rounded-xl bg-primary py-3 text-sm font-extrabold text-white disabled:opacity-50">
   343→        {busy ? '…' : user ? 'ذخیره تغییرات' : 'ساخت کاربر'}
   344→      </button>
   345→    </Modal>
   346→  )
   347→}

   349→/* ═══════════════════════════ Tab 2: تعطیلات رسمی ═══════════════════════════ */
   350→function HolidaysTab({ isGMOM }: { isGMOM: boolean }) {
   351→  const [holidays, setHolidays] = useState<Holiday[]>([])
   352→  const [loading, setLoading] = useState(true)
   353→  const [syncing, setSyncing] = useState(false)
   354→  const [addOpen, setAddOpen] = useState(false)
   355→  const [delTarget, setDelTarget] = useState<Holiday | null>(null)
   356→
   357→  const load = async () => {
   358→    try {
   359→      const d = await api<{ holidays: Holiday[] }>('/api/holidays')
   360→      setHolidays(d.holidays)
   361→    } catch (e: any) {
   362→      toast.error(e.message)
   363→    } finally {
   364→      setLoading(false)
   365→    }
   366→  }
   367→  useEffect(() => { load() }, [])
   368→
   369→  const groups = useMemo(() => {
   370→    const m = new Map<string, { jy: number; jm: number; items: Holiday[] }>()
   371→    for (const h of holidays) {
   372→      const { jy, jm } = toJalaliParts(h.date)
   373→      const key = `${jy}-${String(jm).padStart(2, '0')}`
   374→      if (!m.has(key)) m.set(key, { jy, jm, items: [] })
   375→      m.get(key)!.items.push(h)
   376→    }
   377→    return [...m.values()].sort((a, b) => (a.jy - b.jy) || (a.jm - b.jm))
   378→  }, [holidays])
   379→
   380→  const sync = async () => {
   381→    setSyncing(true)
   382→    try {
   383→      const res = await api<{ ok: boolean; added: number; errors: string[] }>('/api/holidays/sync', { method: 'POST' })
   384→      if (res.added > 0) toast.success(`${faNum(res.added)} تعطیلی جدید از اینترنت افزوده شد ✅`)
   385→      else if (res.ok) toast.info('تعطیلات به‌روز است — مورد جدیدی پیدا نشد')
   386→      else toast.error(res.errors?.[0] || 'همگام‌سازی ناموفق بود')
   387→      await load()
   388→    } catch (e: any) {
   389→      toast.error(e.message)
   390→    } finally {
   391→      setSyncing(false)
   392→    }
   393→  }
   394→
   395→  return (
   396→    <>
   397→      <SectionCard
   398→        title="تعطیلات رسمی 📅"
   399→        subtitle="برنامه‌ریزی سررسید چک و سفارش بر اساس تقویم رسمی ایران"
   400→        icon={<CalendarDays size={18} />}
   401→        actions={
   402→          <>
   403→            <button
   404→              onClick={sync}
   405→              disabled={syncing}
   406→              className="flex items-center gap-1.5 rounded-xl border border-[#207a63]/40 bg-[#207a63]/10 px-4 py-2.5 text-xs font-extrabold text-[#207a63] disabled:opacity-50"
   407→            >
   408→              <RefreshCw size={13} className={syncing ? 'animate-spin' : ''} />
   409→              {syncing ? 'در حال همگام‌سازی…' : 'همگام‌سازی از اینترنت'}
   410→            </button>
   411→            <button
   412→              onClick={() => setAddOpen(true)}
   413→              className="flex items-center gap-1.5 rounded-xl bg-primary px-4 py-2.5 text-xs font-extrabold text-white"
   414→            >
   415→              <Plus size={14} /> افزودن تعطیلی
   416→            </button>
   417→          </>
   418→        }
   419→      >
   420→        {loading ? (
   421→          <p className="py-8 text-center text-sm text-muted-foreground">در حال بارگذاری تعطیلات…</p>
   422→        ) : groups.length === 0 ? (
   423→          <EmptyState emoji="📅" title="تعطیلی ثبت نشده" hint="با دکمهٔ همگام‌سازی یا افزودن دستی شروع کنید" />
   424→        ) : (
   425→          <div className="scroll-gold max-h-[62vh] space-y-4 overflow-y-auto pl-1">
   426→            {groups.map((g) => (
   427→              <div key={`${g.jy}-${g.jm}`}>
   428→                <div className="mb-1.5 flex items-center gap-2">
   429→                  <h3 className="text-sm font-black text-foreground">
   430→                    {J_MONTHS[g.jm - 1]} {faNum(g.jy)}
   431→                  </h3>
   432→                  <span className="rounded-full bg-muted px-2 py-0.5 text-[10px] font-bold text-muted-foreground">
   433→                    {faNum(g.items.length)} مورد
   434→                  </span>
   435→                </div>
   436→                <div className="space-y-1.5">
   437→                  {g.items.map((h) => {
   438→                    const src = SOURCE_LABELS[h.source] || SOURCE_LABELS.manual
   439→                    return (
   440→                      <div key={h.id} className="flex items-center justify-between gap-2 rounded-xl border border-border bg-white/80 px-3 py-2.5">
   441→                        <div className="flex min-w-0 items-center gap-2.5">
   442→                          <span className="shrink-0 rounded-lg bg-[#b3372f]/10 px-2.5 py-1 text-[11px] font-black text-[#b3372f]">
