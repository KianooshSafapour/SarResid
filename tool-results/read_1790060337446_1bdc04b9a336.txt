   104→export default function LeavesView({ ctx }: { ctx: AppCtx }) {
   105→  const [data, setData] = useState<LeavesData | null>(null)
   106→  const [loading, setLoading] = useState(true)
   107→  const holidays = useHolidays()
   108→
   109→  const init = toJalaliParts(todayIso())
   110→  const [jy, setJy] = useState(init.jy)
   111→  const [jm, setJm] = useState(init.jm)
   112→  const [selectedDay, setSelectedDay] = useState('')
   113→
   114→  const [addOpen, setAddOpen] = useState(false)
   115→  const [type, setType] = useState('DAILY')
   116→  const [fromDate, setFromDate] = useState(todayIso())
   117→  const [toDate, setToDate] = useState(todayIso())
   118→  const [fromHour, setFromHour] = useState('09:00')
   119→  const [toHour, setToHour] = useState('13:00')
   120→  const [reason, setReason] = useState('')
   121→  const [saving, setSaving] = useState(false)
   122→
   123→  const [rejectFor, setRejectFor] = useState<string | null>(null)
   124→  const [rejectNote, setRejectNote] = useState('')
   125→  const [approveFor, setApproveFor] = useState<string | null>(null)
   126→  const [replacementNote, setReplacementNote] = useState('')
   127→  const [overrideNote, setOverrideNote] = useState('')
   128→
   129→  const isManager = ['OM', 'GM', 'OWNER'].includes(ctx.user?.role || '')
   130→  const today = todayIso()
   131→
   132→  const load = () => {
   133→    api<LeavesData>('/api/leaves?capacity=1')
   134→      .then(setData)
   135→      .catch((e: any) => toast.error(e.message))
   136→      .finally(() => setLoading(false))
   137→  }
   138→  useEffect(() => {
   139→    load()
   140→  }, [])
   141→
   142→  const usersById = useMemo(() => new Map((data?.users || []).map((u) => [u.id, u])), [data])
   143→  const activeCount = (data?.users || []).filter((u) => u.active).length
   144→  const policy: LeavePolicy | null = data?.policy || null
   145→  const capacity = data?.capacity || {}
   146→
   147→  /* ───── تقویم ماه جلالی ───── */
   148→  const cells = useMemo(() => {
   149→    const len = jMonthLength(jy, jm)
   150→    const arr: ({ iso: string; d: number } | null)[] = Array(jMonthStartWeekday(jy, jm)).fill(null)
   151→    for (let d = 1; d <= len; d++) arr.push({ iso: jalaliToIso(jy, jm, d), d })
   152→    return arr
   153→  }, [jy, jm])
   154→
   155→  const approvedByDay = useMemo(() => {
   156→    const map = new Map<string, { userId: string; userName: string; color: string; type: string; toDate: string }[]>()
   157→    for (const l of data?.all || []) {
   158→      if (l.status !== 'APPROVED') continue
   159→      const color = usersById.get(l.userId)?.color || '#0e7a4a'
   160→      let iso = l.fromDate
   161→      let guard = 0
   162→      while (iso <= l.toDate && guard < 120) {
   163→        const arr = map.get(iso) || []
   164→        arr.push({ userId: l.userId, userName: l.userName, color, type: l.type, toDate: l.toDate })
   165→        map.set(iso, arr)
   166→        iso = addDaysIso(1, iso)
   167→        guard++
   168→      }
   169→    }
   170→    return map
   171→  }, [data, usersById])
   172→
   173→  const pendingByDay = useMemo(() => {
   174→    const map = new Map<string, { userId: string; userName: string; type: string }[]>()
   175→    for (const l of data?.all || []) {
   176→      if (l.status !== 'PENDING') continue
   177→      let iso = l.fromDate
   178→      let guard = 0
   179→      while (iso <= l.toDate && guard < 120) {
   180→        const arr = map.get(iso) || []
   181→        arr.push({ userId: l.userId, userName: l.userName, type: l.type })
   182→        map.set(iso, arr)
   183→        iso = addDaysIso(1, iso)
   184→        guard++
   185→      }
   186→    }
   187→    return map
   188→  }, [data])
   189→
   190→  const moveMonth = (dir: number) => {
   191→    let m = jm + dir
   192→    let y = jy
   193→    if (m > 12) { m = 1; y++ }
   194→    if (m < 1) { m = 12; y-- }
   195→    setJm(m)
   196→    setJy(y)
   197→  }
   198→  const goToday = () => {
   199→    const t = toJalaliParts(todayIso())
   200→    setJy(t.jy)
   201→    setJm(t.jm)
   202→    setSelectedDay(todayIso())
   203→  }
   204→
   205→  /* ───── جزئیات روز انتخاب‌شده ───── */
   206→  const dayOut = selectedDay ? approvedByDay.get(selectedDay) || [] : []
   207→  const dayPending = selectedDay ? pendingByDay.get(selectedDay) || [] : []
   208→  const freeCapacity = activeCount - dayOut.length
   209→  const dayHoliday = selectedDay ? holidays.get(selectedDay) : undefined
   210→
   211→  /* ───── ثبت درخواست ───── */
   212→  const resetForm = () => {
   213→    setType('DAILY')
   214→    setFromDate(todayIso())
   215→    setToDate(todayIso())
   216→    setFromHour('09:00')
   217→    setToHour('13:00')
   218→    setReason('')
   219→  }
   220→  const pickType = (t: string) => {
   221→    setType(t)
   222→    if (t === 'HOURLY') setToDate(fromDate)
   223→  }
   224→  const pickFrom = (iso: string) => {
   225→    setFromDate(iso)
   226→    if (type === 'HOURLY') setToDate(iso)
   227→    else if (toDate < iso) setToDate(iso)
   228→  }
   229→  const submit = async () => {
   230→    if (!fromDate) return toast.error('تاریخ شروع الزامی است')
   231→    if (type === 'HOURLY') {
   232→      if (!fromHour || !toHour) return toast.error('ساعت شروع و پایان الزامی است')
   233→      if (fromHour >= toHour) return toast.error('ساعت شروع باید قبل از ساعت پایان باشد')
   234→    } else {
   235→      if (!toDate) return toast.error('تاریخ پایان الزامی است')
   236→      if (fromDate > toDate) return toast.error('تاریخ پایان نمی‌تواند قبل از تاریخ شروع باشد')
   237→    }
   238→    setSaving(true)
   239→    try {
   240→      const res = await api<{ leave: Leave; conflict: string | null }>('/api/leaves', {
   241→        method: 'POST',
   242→        body: {
   243→          type,
   244→          fromDate,
   245→          toDate: type === 'HOURLY' ? fromDate : toDate,
   246→          fromHour: type === 'HOURLY' ? fromHour : '',
   247→          toHour: type === 'HOURLY' ? toHour : '',
   248→          reason,
   249→        },
   250→      })
   251→      toast.success('درخواست مرخصی ثبت شد — در انتظار تأیید مدیر')
   252→      if (res.conflict) toast.warning(res.conflict, { duration: 9000 })
   253→      setAddOpen(false)
   254→      resetForm()
   255→      load()
   256→    } catch (e: any) {
   257→      toast.error(e.message)
   258→    } finally {
   259→      setSaving(false)
   260→    }
   261→  }
   262→
   263→  /* ───── تأیید / رد / حذف ───── */
   264→  const act = async (l: Leave, action: 'approve' | 'reject', note = '', extra: Record<string, unknown> = {}) => {
   265→    try {
   266→      await api('/api/leaves', { method: 'PUT', body: { id: l.id, action, responseNote: note, ...extra } })
   267→      if (action === 'approve') toast.success('مرخصی تأیید شد — تقویم تیم به‌روز شد')
   268→      else toast.success('درخواست رد شد')
   269→      setRejectFor(null)
   270→      setRejectNote('')
   271→      setApproveFor(null)
   272→      setReplacementNote('')
   273→      setOverrideNote('')
   274→      load()
   275→    } catch (e: any) {
   276→      toast.error(e.message, { duration: 8000 })
   277→    }
   278→  }
   279→
   280→  /** تعداد مرخصی تأییدشدهٔ هم‌پوشان روز شروع (برای گیت سیاست قبل از تأیید) */
   281→  const overlapCountFor = (l: Leave) =>
   282→    (data?.all || []).filter((a) => a.status === 'APPROVED' && a.userId !== l.userId && a.fromDate <= l.toDate && a.toDate >= l.fromDate).length
   283→
   284→  /** گیت سیاست: اگر سقف پر باشد، فرم یادداشت جایگزین/استثنا باز می‌شود؛ وگرنه تأیید مستقیم */
   285→  const requestApprove = (l: Leave) => {
   286→    if (!policy) return act(l, 'approve')
   287→    const overlap = overlapCountFor(l)
   288→    const hardFull = overlap >= policy.hardCapPerDay
   289→    const needsNote = policy.requireReplacementNote ? overlap > 0 : overlap >= policy.maxPerDayWithoutReplacement
   290→    if (hardFull || needsNote) {
   291→      setApproveFor(l.id)
   292→      setReplacementNote('')
   293→      setOverrideNote('')
   294→      return
   295→    }
   296→    act(l, 'approve')
   297→  }
   298→
   299→  const submitApprove = (l: Leave) => {
   300→    const overlap = overlapCountFor(l)
   301→    const isExecUser = ['OWNER', 'GM', 'OM', 'ADMIN'].includes(ctx.user?.role || '')
   302→    const hardFull = overlap >= (policy?.hardCapPerDay ?? 4)
   303→    if (hardFull) {
   304→      if (!isExecUser) return toast.error('سقف روزانه مرخصی پر است — امکان تأیید نیست')
   305→      if (!overrideNote.trim()) return toast.error('سقف روزانه پر است — برای استثنا، یادداشت استثنا (مدیر ارشد) را بنویسید')
   306→      return act(l, 'approve', '', { overrideNote })
   307→    }
   308→    if (replacementNote.trim().length < 2) return toast.error('نام جایگزین را ثبت کنید')
   309→    act(l, 'approve', '', hardFull ? { replacementNote, overrideNote } : { replacementNote })
   310→  }
   311→  const remove = async (l: Leave) => {
   312→    if (!window.confirm('این درخواست مرخصی حذف شود؟')) return
   313→    try {
   314→      await api(`/api/leaves?id=${l.id}`, { method: 'DELETE' })
   315→      toast.success('درخواست حذف شد')
   316→      load()
   317→    } catch (e: any) {
   318→      toast.error(e.message)
   319→    }
   320→  }
   321→
   322→  /** هم‌پوشانی درخواست در نوبت با مرخصی‌های تأییدشدهٔ دیگران (نشان FCFS برای تأییدکننده) */
   323→  const conflictNamesFor = (l: Leave) =>
   324→    (data?.all || [])
   325→      .filter((a) => a.status === 'APPROVED' && a.userId !== l.userId && a.fromDate <= l.toDate && a.toDate >= l.fromDate)
   326→      .map((a) => a.userName)
   327→
   328→  if (loading && !data) {
   329→    return <div className="py-24 text-center text-sm font-bold text-muted-foreground">در حال بارگذاری مدیریت مرخصی…</div>
   330→  }
   331→  if (!data) return null
   332→
   333→  const teamOutNames = data.stats.teamOutToday.map((t) => t.userName).slice(0, 3).join('، ')
   334→
   335→  return (
   336→    <div className="space-y-5">
   337→      {/* سرصفحه */}
   338→      <div className="fade-in-up">
   339→        <h1 className="text-xl font-black text-foreground sm:text-2xl">مدیریت مرخصی و غیبت</h1>
   340→        <p className="mt-1 text-sm text-muted-foreground">
   341→          ثبت درخواست، تقویم شفاف تیم و تأیید نوبتی — اولین ثبت، اولویت دارد و برای همه قابل مشاهده است
   342→        </p>
   343→      </div>
   344→
   345→      {/* KPI */}
   346→      <div className={cn('grid grid-cols-2 gap-3', isManager ? 'sm:grid-cols-4' : 'sm:grid-cols-3')}>
   347→        <StatKpi
   348→          label="مرخصی‌های تأییدشدهٔ من (امسال)"
   349→          value={`${faNum(data.stats.myApprovedDays)} روز`}
   350→          tone="emerald"
   351→          icon={<CalendarCheck size={18} />}
   352→          hint="جمع روزهای تأییدشدهٔ سال جاری"
   353→        />
   354→        <StatKpi
   355→          label="در انتظار تأیید"
   356→          value={faNum(data.stats.myPending)}
   357→          tone="gold"
   358→          icon={<Clock size={18} />}
   359→          hint="درخواست‌های در نوبتِ خودتان"
   360→        />
   361→        <StatKpi
   362→          label="امروز غایب"
   363→          value={faNum(data.stats.teamOutToday.length)}
   364→          tone="terra"
   365→          icon={<Users size={18} />}
   366→          hint={teamOutNames || 'همهٔ همکاران حاضرند'}
   367→        />
   368→        {isManager && (
   369→          <StatKpi
   370→            label="صف تأیید"
   371→            value={faNum(data.stats.pendingCount)}
   372→            tone="rose"
   373→            icon={<Hourglass size={18} />}
   374→            hint="درخواست‌های منتظر تصمیم شما"
   375→          />
   376→        )}
   377→      </div>
   378→
   379→      {/* نوار ظرفیت مرخصی امروز — سیاست سقف روزانه */}
   380→      {policy && (
   381→        <div className="fade-in-up rounded-2xl border border-border/70 bg-card p-3">
   382→          {(() => {
   383→            const cap = capacity[today]
   384→            const maxCap = policy.maxPerDayWithoutReplacement
   385→            const outToday = cap ? cap.out : data.stats.teamOutToday.length
   386→            const remaining = Math.max(0, maxCap - outToday)
   387→            const pct = maxCap > 0 ? Math.min(100, Math.round((outToday / maxCap) * 100)) : 0
   388→            const barColor = remaining <= 0 ? '#b3372f' : pct > 60 ? '#c9a227' : '#0e7a4a'
   389→            return (
   390→              <div className="flex flex-wrap items-center justify-between gap-3">
   391→                <div className="min-w-[210px] flex-1">
   392→                  <p className="text-xs font-black">
   393→                    ظرفیت مرخصی امروز: <span style={{ color: barColor }}>{faNum(remaining)} نفر</span>
   394→                    <span className="mr-1.5 text-[10px] font-bold text-muted-foreground">
   395→                      ({faNum(outToday)} تأییدشده از سقف {faNum(maxCap)} — حداقل حاضرین شیفت: {faNum(policy.minPresentPerShift)})
   396→                    </span>
   397→                  </p>
   398→                  <div className="mt-1.5 h-2 overflow-hidden rounded-full bg-muted">
   399→                    <div className="h-full rounded-full transition-all" style={{ width: `${pct}%`, background: barColor }} />
   400→                  </div>
   401→                </div>
   402→                <div className="flex flex-wrap items-center gap-1.5 text-[10px] font-bold text-muted-foreground">
   403→                  <Pill label={`بدون جایگزین تا ${faNum(maxCap)} نفر`} color="#0e7a4a" bg="#0e7a4a10" />
   404→                  <Pill label={`سقف روزانه: ${faNum(policy.hardCapPerDay)}`} color="#b3372f" bg="#b3372f10" />
   405→                  {policy.blockHolidays && <Pill label="تعطیل رسمی: بسته" color="#8a6d10" bg="#fdf6dd" />}
   406→                  {policy.blockFridays && <Pill label="جمعه: بسته" color="#8a6d10" bg="#fdf6dd" />}
   407→                </div>
   408→              </div>
   409→            )
   410→          })()}
   411→        </div>
   412→      )}
   413→
   414→      <div className="grid gap-5 lg:grid-cols-3">
   415→        {/* تقویم تیم */}
   416→        <SectionCard
   417→          title="تقویم مرخصی تیم"
   418→          subtitle="مرخصی‌های تأییدشدهٔ همهٔ همکاران — برای جزئیات روی هر روز بزنید"
   419→          icon={<CalendarDays size={18} />}
   420→          className="lg:col-span-2"
   421→        >
   422→          <div className="mb-3 flex items-center justify-between gap-2">
   423→            <div className="flex items-center gap-1">
   424→              <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => moveMonth(1)} aria-label="ماه بعد">
   425→                <ChevronRight size={16} />
   426→              </Button>
   427→              <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => moveMonth(-1)} aria-label="ماه قبل">
   428→                <ChevronLeft size={16} />
   429→              </Button>
   430→            </div>
   431→            <div className="text-sm font-black text-foreground">
   432→              {J_MONTHS[jm - 1]} {faNum(jy)}
   433→            </div>
   434→            <button
   435→              onClick={goToday}
   436→              className="rounded-full border border-border px-3 py-1 text-[11px] font-bold text-foreground/80 transition hover:border-[#c9a227] hover:bg-[#c9a227]/10"
   437→            >
   438→              برو به امروز
   439→            </button>
   440→          </div>
   441→
   442→          <div className="mb-1.5 grid grid-cols-7 gap-1 text-center font-bold text-muted-foreground sm:gap-1.5" style={calFontStyle('calc(var(--cal-font, 13px) * 0.8)')}>
   443→            {J_WEEKDAYS_SHORT.map((d, i) => (
   444→              <span key={i} className={d === 'ج' ? 'text-[#b3372f]' : ''}>{d}</span>
   445→            ))}
   446→          </div>
   447→          <div className="grid grid-cols-7 gap-1 sm:gap-1.5" style={calFontStyle()}>
   448→            {cells.map((c, i) => {
   449→              if (!c) return <span key={`e${i}`} />
   450→              const iso = c.iso
   451→              const out = approvedByDay.get(iso) || []
   452→              const pend = pendingByDay.get(iso) || []
   453→              const isToday = iso === today

   821→                <JalaliDatePicker value={fromDate} onChange={pickFrom} holidays={holidays} minDate={addDaysIso(-1)} warnHoliday={false} />
   822→              </Labeled>
   823→              <Labeled label="تا تاریخ">
   824→                <JalaliDatePicker value={toDate} onChange={setToDate} holidays={holidays} minDate={fromDate} warnHoliday={false} />
   825→              </Labeled>
   826→            </>
   827→          )}
   828→
   829→          <Labeled label="دلیل (اختیاری)">
   830→            <textarea
   831→              value={reason}

   836→              className={cn(INPUT_CLS, 'resize-none')}
   837→            />
   838→          </Labeled>
   839→
   840→          <p className="rounded-xl bg-[#c9a227]/10 px-3 py-2 text-[11px] font-bold leading-5 text-[#8a6d10]">
   841→            نوبت‌بندی به‌صورت «اولین ثبت، اولویت» است؛ ثبت درخواست به معنای تأیید نیست و تصمیم نهایی با مدیر عملیات است.
   842→          </p>
   843→
   844→          <button
   845→            type="button"
   846→            disabled={saving}
   847→            onClick={submit}
   848→            className="w-full rounded-xl bg-[#0e7a4a] py-3 text-sm font-black text-white shadow-md transition hover:bg-[#0b6a40] disabled:opacity-60"
   849→          >
   850→            {saving ? 'در حال ثبت…' : 'ثبت درخواست'}
   851→          </button>
   852→        </Modal>
   853→      )}
   854→    </div>
   855→  )
   856→}
