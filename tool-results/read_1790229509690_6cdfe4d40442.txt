   292→function DangerTab() {
   293→  const [armed, setArmed] = useState<string | null>(null)
   294→  const [purge, setPurge] = useState<DangerCard | null>(null)
   295→  const [typed, setTyped] = useState('')
   296→  const [busy, setBusy] = useState(false)
   297→
   298→  useEffect(() => {
   299→    if (!armed) return
   300→    const t = setTimeout(() => setArmed(null), 6000)
   301→    return () => clearTimeout(t)
   302→  }, [armed])
   303→
   304→  const runAction = async (action: string, confirm?: string) => {
   305→    setBusy(true)
   306→    try {
   307→      const res = await api<{ ok: boolean; summary?: { deleted: Record<string, number> } }>('/api/admin/db', {
   308→        method: 'POST',
   309→        body: { action, confirm },
   310→      })
   311→      const del = res.summary?.deleted || {}
   312→      const keys = Object.keys(del)
   313→      if (keys.length) {
   314→        toast.success(`انجام شد — ${keys.map((k) => `${faNum(del[k])} «${TABLE_FA[k] || k}»`).join('، ')} حذف شد`)
   315→      } else {
   316→        toast.info('انجام شد — موردی برای حذف وجود نداشت')
   317→      }
   318→      setPurge(null)
   319→      setTyped('')
   320→      setArmed(null)
   321→    } catch (e) {
   322→      toast.error((e as Error).message)
   323→    } finally {
   324→      setBusy(false)
   325→    }
   326→  }
   327→
   328→  return (
   329→    <div className="space-y-5">

   451→const MAP_KEY = (kind: string) => `hz-import-map-${kind}`
   452→
   453→const normHeader = (s: string) => s.toLowerCase().replace(/\u200c/g, '').replace(/\s+/g, ' ').trim()
   454→
   455→function autoGuess(headers: string[], kind: string): Record<string, number> {
   456→  const aliases = GUESS[kind] || {}
   457→  const map: Record<string, number> = {}
   458→  const claimed = new Set<number>()
   459→  // گذر ۱ — تطبیق دقیق
   460→  for (const key of Object.keys(aliases)) {
   461→    for (const a of aliases[key]) {
   462→      const i = headers.findIndex((h, idx) => !claimed.has(idx) && normHeader(h) === normHeader(a))
   463→      if (i >= 0) { map[key] = i; claimed.add(i); break }
   464→    }
   465→  }
   466→  // گذر ۲ — تطبیق جزئی (شامل‌مشتق دوطرفه با حداقل طول)
   467→  for (const key of Object.keys(aliases)) {
   468→    if (map[key] !== undefined) continue
   469→    for (const a of aliases[key]) {
   470→      const needle = normHeader(a)
   471→      if (needle.length < 3) continue
   472→      const i = headers.findIndex((h, idx) => {
   473→        if (claimed.has(idx)) return false
   474→        const n = normHeader(h)
   475→        return n.length >= 3 && (n.includes(needle) || needle.includes(n))
   476→      })
   477→      if (i >= 0) { map[key] = i; claimed.add(i); break }
   478→    }
   479→  }
   480→  return map
   481→}

  1206→  label,
  1207→  armedLabel = 'مطمئنید؟ دوباره بزنید',
  1208→  onClick,
  1209→  disabled,
  1210→  className,
  1211→  tone = 'rose',
  1212→}: {
  1213→  label: string
  1214→  armedLabel?: string
  1215→  onClick: () => void
  1216→  disabled?: boolean
  1217→  className?: string
  1218→  tone?: 'rose' | 'gold'
  1219→}) {
  1220→  const [armed, setArmed] = useState(false)
  1221→  useEffect(() => {
  1222→    if (!armed) return

  1227→    <button

  1229→      disabled={disabled}
  1230→      onClick={() => {

  1232→          setArmed(false)
  1233→          onClick()
  1234→        } else {
  1235→          setArmed(true)
  1236→        }
  1237→      }}
  1238→      className={cn(
  1239→        'rounded-xl px-4 py-2.5 text-xs font-extrabold transition-all disabled:opacity-40',

  1245→        className

  1248→      {armed ? armedLabel : label}
