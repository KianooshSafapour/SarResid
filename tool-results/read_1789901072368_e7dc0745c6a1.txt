   125→    { k: 'R', v: r.rScore, color: '#0e7a4a', tip: `تازگی — ${faNum(r.rDays)} روز از آخرین فعالیت · نمرهٔ ${faNum(r.rScore)} از ۵` },

   127→    { k: 'M', v: r.mScore, color: '#c96f4a', tip: `ارزش — میانگین سبد ${faMoney(r.mAvg)} تومان · نمرهٔ ${faNum(r.mScore)} از ۵` },
   128→  ]
   129→  return (
   130→    <div className="flex flex-col gap-1" dir="ltr">
   131→      {rows.map((row) => (
   132→        <div key={row.k} className="flex items-center gap-1.5" title={row.tip}>
   133→          <span className="w-3 text-[9px] font-black text-muted-foreground">{row.k}</span>
   134→          <div className="h-1.5 w-20 overflow-hidden rounded-full bg-muted">
   135→            <div className="h-full rounded-full transition-all" style={{ width: `${(row.v / 5) * 100}%`, background: row.color }} />
   136→          </div>
   137→        </div>
   138→      ))}
   139→    </div>
   140→  )
   141→}

   148→function tagList(c: Pick<Customer, 'tags'>): string[] {

   153→    return []
