    98→function PlanogramMini({ planogram, productMap }: { planogram: PlanogramRow; productMap: ProductMap }) {
    99→  const rows = Array.isArray(planogram.layout) ? planogram.layout : []
   100→  let filled = 0
   101→  let oos = 0
   102→  let low = 0
   103→  for (const row of rows)
   104→    for (const pid of row || []) {
   105→      if (!pid) continue
   106→      filled++
   107→      const pr = productMap[pid]
   108→      if (!pr) continue
   109→      if (pr.stock <= 0) oos++
   110→      else if (pr.stock < Math.max(1, pr.reorderLevel)) low++
   111→    }
   112→  return (
   113→    <div className="space-y-2">
   114→      <div className="overflow-x-auto scroll-gold rounded-lg border border-border bg-[#faf7ee] p-1.5">
   115→        <div className="inline-grid gap-[2px]" style={{ gridTemplateColumns: `repeat(${Math.max(1, planogram.cols)}, 13px)` }}>
   116→          {rows.flatMap((row, ri) =>
   117→            (row || []).map((pid, ci) => {
   118→              const pr = pid ? productMap[pid] : undefined
   119→              return (
   120→                <div
   121→                  key={`${ri}-${ci}`}
   122→                  title={pr ? `${pr.name} — موجودی: ${faNum(pr.stock)}` : pid ? 'کالای ناشناس' : 'خالی'}
   123→                  className="h-[13px] w-[13px] rounded-[2px] border border-black/5"
   124→                  style={{ background: pr ? stockColor(pr.stock, pr.reorderLevel) : 'transparent' }}
   125→                />
   126→              )
   127→            }),
   128→          )}
   129→        </div>
   130→      </div>
   131→      <div className="flex flex-wrap gap-1.5 text-[10px] font-bold">
   132→        <span className="rounded-full bg-muted px-2 py-0.5">📦 {faNum(filled)} خانهٔ پر</span>
   133→        <span className="rounded-full bg-[#b3372f]/10 px-2 py-0.5 text-[#b3372f]">🔴 {faNum(oos)} تمام‌شده</span>
   134→        <span className="rounded-full bg-[#c9a227]/15 px-2 py-0.5 text-[#8a6d10]">🟡 {faNum(low)} رو به اتمام</span>
   135→      </div>
   136→    </div>
   137→  )
   138→}
