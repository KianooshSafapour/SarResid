   222→/* ───────────────── Smart reorder: low-stock → DRAFT orders per provider ───────────────── */
   223→
   224→type ProviderRow = { id: string; name: string }
   225→
   226→function SmartReorderModal({ products, onClose, onDone }: { products: Product[]; onClose: () => void; onDone: (created: number) => void }) {
   227→  const [providers, setProviders] = useState<ProviderRow[]>([])
   228→  const [qtys, setQtys] = useState<Record<string, number>>({})
   229→  const [excluded, setExcluded] = useState<Record<string, boolean>>({}) // by providerId
   230→  const [skipped, setSkipped] = useState<Record<string, boolean>>({}) // per product
   231→  const [busy, setBusy] = useState(false)
   232→
   233→  useEffect(() => {
   234→    api<{ providers: ProviderRow[] }>('/api/providers').then((d) => setProviders(d.providers)).catch(() => {})
   235→  }, [])
   236→
   237→  const low = useMemo(() => products.filter((p) => p.stock <= p.reorderLevel), [products])
   238→  const suggested = (p: Product) => Math.max(p.reorderLevel * 2 - p.stock, 1)
   239→
   240→  // initialize suggestion quantities once
   241→  useEffect(() => {
   242→    if (Object.keys(qtys).length === 0 && low.length > 0) {
   243→      const init: Record<string, number> = {}
   244→      for (const p of low) init[p.id] = suggested(p)
   245→      setQtys(init)
   246→    }
   247→  }, [low.length])
   248→
   249→  const groups = useMemo(() => {
   250→    const m = new Map<string, Product[]>()
   251→    for (const p of low) {
   252→      const key = p.providerId || '_none'
   253→      if (!m.has(key)) m.set(key, [])
   254→      m.get(key)!.push(p)
   255→    }
   256→    return [...m.entries()]
   257→  }, [low])
   258→
   259→  const activeGroups = groups.filter(([pid]) => !excluded[pid])
   260→  const activeItems = activeGroups.flatMap(([, ps]) => ps).filter((p) => !skipped[p.id])
   261→  const estTotal = activeItems.reduce((s, p) => s + (qtys[p.id] || 0) * p.buyPrice, 0)
   262→
   263→  const create = async () => {
   264→    if (activeItems.length === 0) return
   265→    setBusy(true)
   266→    const d = new Date()
   267→    d.setDate(d.getDate() + 2)
   268→    const deliveryDate = d.toISOString().slice(0, 10)
   269→    let created = 0
   270→    try {
   271→      for (const [pid, ps] of activeGroups) {
   272→        const items = ps.filter((p) => !skipped[p.id])
   273→        if (items.length === 0) continue
   274→        const prov = providers.find((x) => x.id === pid)
   275→        await api('/api/orders', {
   276→          method: 'POST',
   277→          body: {
   278→            providerId: pid,
   279→            providerName: prov?.name || 'بدون تأمین‌کننده',
   280→            status: 'DRAFT',
   281→            deliveryDate,
   282→            notes: 'پیش‌نویس هوشمند — بر اساس نقطه سفارش و موجودی فعلی',
   283→            items: items.map((p) => ({ productId: p.id, qty: qtys[p.id] || suggested(p), unitBuyPrice: p.buyPrice })),
   284→          },
   285→        })
   286→        created++
   287→      }
   288→      onDone(created)
   289→    } catch (e: any) {
   290→      toast.error(e.message)
   291→    } finally {
   292→      setBusy(false)
   293→    }
   294→  }
   295→
   296→  return (
   297→    <Modal title="پیش‌نویس هوشمند سفارش 🤖" onClose={onClose} wide>
   298→      <p className="mb-4 rounded-xl border border-[#77934a]/30 bg-[#e9f0e4]/60 p-3 text-[11px] leading-5">
   299→        سامانه {faNum(low.length)} کالای رسیده به نقطه سفارش را پیدا کرده و بر اساس «دو برابر نقطه سفارش منهای موجودی فعلی» پیشنهاد تعداد می‌دهد.
   300→        تعدادها را اصلاح کنید، تأمین‌کننده‌های غیرضروری را بردارید و سفارش‌های پیش‌نویس را یک‌جا بسازید — بعد از بخش سفارش‌ها ارسالشان کنید.
   301→      </p>
   302→
   303→      {low.length === 0 ? (
   304→        <EmptyState emoji="🌿" title="همه موجودی‌ها سالم‌اند" hint="فعلاً کالایی به نقطه سفارش نرسیده" />
   305→      ) : (
   306→        <div className="scroll-gold max-h-[52vh] space-y-3 overflow-y-auto pl-1">
   307→          {groups.map(([pid, ps]) => {
   308→            const prov = providers.find((x) => x.id === pid)
   309→            const isExcluded = !!excluded[pid]
   310→            const groupTotal = ps.reduce((s, p) => s + (qtys[p.id] || 0) * p.buyPrice, 0)
   311→            return (
   312→              <div key={pid} className={cn('rounded-2xl border p-3 transition', isExcluded ? 'border-border bg-muted/40 opacity-60' : 'glow-card border-[#77934a]/30 bg-white/80')}>
   313→                <div className="mb-2 flex flex-wrap items-center justify-between gap-2">
   314→                  <label className="flex cursor-pointer items-center gap-2">
   315→                    <input
   316→                      type="checkbox"
   317→                      checked={!isExcluded}
   318→                      onChange={(e) => setExcluded((x) => ({ ...x, [pid]: !e.target.checked }))}
   319→                      className="h-4 w-4 accent-[#77934a]"
   320→                    />
   321→                    <span className="text-sm font-black">🚚 {prov?.name || 'بدون تأمین‌کننده مشخص'}</span>
   322→                    <span className="rounded-full bg-secondary px-2 py-0.5 text-[10px] font-bold text-[#5c7236]">{faNum(ps.length)} قلم</span>
   323→                  </label>
   324→                  <span className="text-[11px] font-black text-[#8a6d10]">{faMoney(groupTotal)} تومان</span>
   325→                </div>
   326→                <div className="grid gap-1.5 sm:grid-cols-2">
   327→                  {ps.map((p) => (
   328→                    <div key={p.id} className={cn('flex items-center justify-between gap-2 rounded-xl border border-border/70 px-2.5 py-2', skipped[p.id] && 'opacity-40')}>
   329→                      <label className="flex min-w-0 cursor-pointer items-center gap-2">
   330→                        <input
   331→                          type="checkbox"
   332→                          checked={!skipped[p.id]}
   333→                          disabled={isExcluded}
   334→                          onChange={(e) => setSkipped((x) => ({ ...x, [p.id]: !e.target.checked }))}
   335→                          className="h-3.5 w-3.5 accent-[#0e7a4a]"
   336→                        />
   337→                        <span className="min-w-0">
   338→                          <span className="block truncate text-[11px] font-bold">{p.name}</span>
   339→                          <span className="block text-[9px] text-muted-foreground">
   340→                            موجودی {faNum(p.stock)} • نقطه سفارش {faNum(p.reorderLevel)} {p.stock === 0 ? '• 🔴 ناموجود' : ''}
   341→                          </span>
   342→                        </span>
   343→                      </label>
   344→                      <span className="flex shrink-0 items-center gap-1">
   345→                        <button
   346→                          disabled={isExcluded}
   347→                          onClick={() => setQtys((q) => ({ ...q, [p.id]: Math.max((q[p.id] || 1) - 1, 1) }))}
   348→                          className="stepper-btn !h-8 !w-8 text-sm disabled:opacity-30"
   349→                        >
   350→                          −
   351→                        </button>
   352→                        <span className="w-10 text-center text-xs font-black">{faNum(qtys[p.id] || 0)}</span>
   353→                        <button
   354→                          disabled={isExcluded}
   355→                          onClick={() => setQtys((q) => ({ ...q, [p.id]: (q[p.id] || 0) + 1 }))}
   356→                          className="stepper-btn !h-8 !w-8 text-sm disabled:opacity-30"
   357→                        >
   358→                          +
   359→                        </button>
   360→                      </span>
   361→                    </div>
   362→                  ))}
   363→                </div>
   364→              </div>
   365→            )
   366→          })}
   367→        </div>
   368→      )}
   369→
   370→      <div className="mt-4 flex flex-wrap items-center justify-between gap-3 border-t border-border pt-4">
   371→        <div className="text-[11px] font-bold text-muted-foreground">
   372→          {faNum(activeItems.length)} قلم در {faNum(activeGroups.length)} سفارش پیش‌نویس • برآورد خرید: <b className="text-[#8a6d10]">{faMoney(estTotal)} تومان</b>

   697→        byName.set(p.name.trim().toLowerCase(), p)
   698→      }
   699→      for (const r of mapped) {
   700→        const hit = (r.barcode && byBarcode.get(r.barcode)) || byName.get(r.name.toLowerCase())
   701→        if (hit) { r.duplicate = true; r.dupWith = hit.name }
   702→      }
   703→      setRows(mapped)
   704→      if (!mapped.length) toast.error('سطر قابل‌نگاشت پیدا نشد — ستون‌های «نام کالا» و «بارکد» را بررسی کنید')
   705→      else toast.success(`${faNum(mapped.length)} سطر خوانده شد، ${faNum(mapped.filter((x) => x.duplicate).length)} مورد تکراری تشخیص داده شد`)
   706→    } catch (err: any) {
   707→      toast.error('خواندن فایل ناموفق بود: ' + err.message)
   708→    }
   709→  }
   710→
   711→  const doImport = async () => {
   712→    setBusy(true)
   713→    try {
   714→      const res = await api<any>('/api/products/import', { method: 'POST', body: { items: rows, mode } })
   715→      setResult(res)
   716→      toast.success(`درون‌ریزی انجام شد: ${faNum(res.created)} جدید، ${faNum(res.updated)} به‌روزرسانی، ${faNum(res.skipped)} ردشد`)
   717→    } catch (e: any) { toast.error(e.message) } finally { setBusy(false) }
   718→  }
   719→
   720→  return (
   721→    <Modal title="ورود کالاها از فایل اکسل هلو" onClose={onClose} wide>
   722→      {!rows.length ? (
   723→        <div className="rounded-2xl border-2 border-dashed border-[#c9a227]/50 bg-[#fdf6dd]/40 p-8 text-center">
   724→          <FileSpreadsheet size={40} className="mx-auto mb-3 text-[#8a6d10]" />
   725→          <p className="text-sm font-black">فایل اکسل خروجی هلو را انتخاب کنید (.xls / .xlsx / .csv)</p>
   726→          <p className="mx-auto mt-1 max-w-md text-[11px] leading-5 text-muted-foreground">
   727→            ستون‌های رایج هلو (نام کالا، بارکد، کد، قیمت خرید، قیمت فروش، موجودی) به‌طور خودکار شناسایی می‌شوند و تکراری‌ها مشخص می‌گردند.
   728→          </p>
   729→          <label className="mt-4 inline-flex cursor-pointer items-center gap-2 rounded-xl bg-primary px-6 py-3 text-sm font-extrabold text-white">
   730→            📁 انتخاب فایل اکسل
   731→            <input type="file" accept=".xls,.xlsx,.csv" hidden onChange={onFile} />
   732→          </label>
   733→        </div>
   734→      ) : result ? (
   735→        <div className="space-y-3 text-center">
   736→          <p className="text-3xl">🎉</p>
   737→          <p className="text-sm font-black">درون‌ریزی کامل شد</p>
   738→          <div className="grid grid-cols-3 gap-2">
   739→            <KeyValue k="جدید" v={faNum(result.created)} />
   740→            <KeyValue k="به‌روزرسانی" v={faNum(result.updated)} />
   741→            <KeyValue k="ردشده" v={faNum(result.skipped)} />
   742→          </div>
   743→          {result.log?.length > 0 && (
   744→            <div className="scroll-gold max-h-40 overflow-y-auto rounded-xl bg-muted/40 p-3 text-right text-[11px]">
   745→              {result.log.map((l: string, i: number) => <p key={i}>• {l}</p>)}
   746→            </div>
   747→          )}
   748→          <button onClick={onDone} className="w-full rounded-xl bg-primary py-2.5 text-xs font-extrabold text-white">بستن</button>
   749→        </div>
   750→      ) : (
   751→        <>
   752→          <div className="mb-3 flex flex-wrap items-center justify-between gap-2">
   753→            <p className="text-xs font-bold">{faNum(rows.length)} سطر خوانده شد — {faNum(rows.filter((r) => r.duplicate).length)} تکراری</p>
   754→            <div className="flex gap-1.5">
   755→              <button onClick={() => setMode('update')} className={cn('rounded-lg px-3 py-1.5 text-[11px] font-bold', mode === 'update' ? 'bg-primary text-white' : 'border')}>به‌روزرسانی تکراری‌ها</button>
   756→              <button onClick={() => setMode('skip')} className={cn('rounded-lg px-3 py-1.5 text-[11px] font-bold', mode === 'skip' ? 'bg-primary text-white' : 'border')}>ردکردن تکراری‌ها</button>
   757→            </div>
   758→          </div>
   759→          <div className="scroll-gold max-h-72 overflow-y-auto rounded-xl border border-border">
   760→            <table className="w-full text-right text-[11px]">
   761→              <thead className="sticky top-0 bg-secondary">
   762→                <tr>
   763→                  <th className="p-2">کالا</th><th className="p-2">بارکد</th><th className="p-2">خرید</th><th className="p-2">فروش</th><th className="p-2">موجودی</th><th className="p-2">وضعیت</th>
   764→                </tr>
   765→              </thead>
   766→              <tbody>
   767→                {rows.map((r, i) => (
   768→                  <tr key={i} className={cn('border-t border-border/50', r.duplicate && 'bg-[#fef9c3]/50')}>
   769→                    <td className="max-w-40 truncate p-2 font-bold">{r.name}</td>
   770→                    <td className="p-2 font-mono" dir="ltr">{r.barcode || '—'}</td>
   771→                    <td className="p-2">{r.buyPrice ? faMoney(r.buyPrice) : '—'}</td>
   772→                    <td className="p-2">{r.sellPrice ? faMoney(r.sellPrice) : '—'}</td>
   773→                    <td className="p-2">{r.stock || '—'}</td>
   774→                    <td className="p-2">{r.duplicate ? <Pill label="تکراری" color="#a16207" /> : <Pill label="جدید" color="#0e7a4a" />}</td>
   775→                  </tr>
   776→                ))}
   777→              </tbody>
   778→            </table>
   779→          </div>
   780→          <button onClick={doImport} disabled={busy} className="mt-3 w-full rounded-xl bg-primary py-3 text-sm font-extrabold text-white disabled:opacity-50">
   781→            {busy ? 'در حال درون‌ریزی…' : `شروع درون‌ریزی (${faNum(rows.length)} کالا)`}
   782→          </button>
   783→        </>
   784→      )}
   785→    </Modal>
   786→  )
   787→}

   790→
   791→function MergeWizard({ products, onClose, onDone }: { products: Product[]; onClose: () => void; onDone: () => void }) {
   792→  const groups = useMemo(() => {
   793→    const byName = new Map<string, Product[]>()
   794→    for (const p of products) {
   795→      const key = p.name.replace(/\s+/g, ' ').trim().toLowerCase()
   796→      const norm = key.replace(/[يیكک]/g, (m) => (m === 'ي' || m === 'ی' ? 'ی' : 'ک'))
   797→      if (!byName.has(norm)) byName.set(norm, [])
   798→      byName.get(norm)!.push(p)
   799→    }
   800→    return [...byName.values()].filter((g) => g.length > 1)
   801→  }, [products])
   802→  const [primary, setPrimary] = useState<Record<string, string>>({})
   803→
   804→  const merge = async (groupName: string, ids: string[]) => {
   805→    const primaryId = primary[groupName] || ids[0]
   806→    try {
   807→      await api('/api/products/merge', { method: 'POST', body: { primaryId, duplicateIds: ids.filter((i) => i !== primaryId) } })
   808→      toast.success('ادغام انجام شد ✅ — موجودی و بارکد‌ها ترکیب شدند')
   809→      onDone()
   810→    } catch (e: any) { toast.error(e.message) }
   811→  }
   812→
   813→  return (
   814→    <Modal title="پاک‌سازی کالاهای تکراری" onClose={onClose} wide>
   815→      {groups.length === 0 ? (
   816→        <div className="py-6 text-center text-sm text-muted-foreground">
   817→          <p className="text-3xl">✨</p>
   818→          <p className="mt-2 font-bold">تکراری‌ای پیدا نشد — انبار تمیز است!</p>
   819→        </div>
   820→      ) : (
   821→        <div className="space-y-4">
   822→          <p className="rounded-xl bg-accent/60 p-3 text-[11px] leading-5">
   823→            برای هر گروه تکراری، کالای «اصلی» را انتخاب کنید. موجودی‌ها جمع و همه بارکدها زیر کالای اصلی نگه‌داری می‌شوند و تکراری‌ها غیرفعال می‌گردند.
   824→          </p>
   825→          <div className="scroll-gold max-h-80 space-y-3 overflow-y-auto">
   826→            {groups.map((g, gi) => {
   827→              const key = `g${gi}`
   828→              return (
   829→                <div key={key} className="rounded-2xl border border-border p-3">
   830→                  <p className="mb-2 text-xs font-black">{g[0].name.replace(/\s+/g, ' ').trim()} <span className="text-muted-foreground">({faNum(g.length)} مورد)</span></p>
   831→                  <div className="grid gap-1.5">
   832→                    {g.map((p) => (
   833→                      <label key={p.id} className="flex cursor-pointer items-center gap-2 rounded-xl bg-muted/40 p-2 text-xs">
   834→                        <input type="radio" name={key} defaultChecked={!primary[key] || primary[key] === p.id} onChange={() => setPrimary((s) => ({ ...s, [key]: p.id }))} />
   835→                        <span className="flex-1 font-bold">{p.name}</span>
   836→                        <span className="font-mono text-[10px] text-muted-foreground" dir="ltr">{p.barcodes.join(', ') || '—'}</span>
   837→                        <span>موجودی: {faNum(p.stock)}</span>
   838→                      </label>
   839→                    ))}
   840→                  </div>
   841→                  <button onClick={() => merge(key, g.map((p) => p.id))} className="mt-2 rounded-lg bg-[#8a5a2b] px-4 py-1.5 text-[11px] font-extrabold text-white">
   842→                    ادغام این گروه
   843→                  </button>
   844→                </div>
   845→              )
   846→            })}
   847→          </div>
   848→        </div>
   849→      )}
   850→    </Modal>
   851→  )
   852→}
