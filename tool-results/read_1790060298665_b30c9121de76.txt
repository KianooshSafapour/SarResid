   704→/* ───────────────── Smart reorder: low-stock → DRAFT orders per provider ───────────────── */
   705→
   706→type ProviderRow = { id: string; name: string }
   707→
   708→function SmartReorderModal({ onClose, onDone }: { onClose: () => void; onDone: (created: number) => void }) {
   709→  const [products, setProducts] = useState<Product[]>([])
   710→  const [loaded, setLoaded] = useState(false)
   711→  const [providers, setProviders] = useState<ProviderRow[]>([])
   712→  const [qtys, setQtys] = useState<Record<string, number>>({})
   713→  const [excluded, setExcluded] = useState<Record<string, boolean>>({}) // by providerId
   714→  const [skipped, setSkipped] = useState<Record<string, boolean>>({}) // per product
   715→  const [busy, setBusy] = useState(false)
   716→
   717→  // فهرست کم‌موجودی‌ها از سرور (پارامتر lowstock=1 — مقیاس‌پذیر برای ۱۰هزار کالا)
   718→  useEffect(() => {
   719→    api<{ products: Product[] }>('/api/products?lowstock=1&page=1&pageSize=500')
   720→      .then((d) => { setProducts(d.products); setLoaded(true) })
   721→      .catch(() => setLoaded(true))
   722→    api<{ providers: ProviderRow[] }>('/api/providers').then((d) => setProviders(d.providers)).catch(() => {})
   723→  }, [])
   724→
   725→  const low = useMemo(() => products.filter((p) => p.stock <= p.reorderLevel), [products])
   726→  const suggested = (p: Product) => Math.max(p.reorderLevel * 2 - p.stock, 1)
   727→
   728→  // initialize suggestion quantities once
   729→  useEffect(() => {
   730→    if (Object.keys(qtys).length === 0 && low.length > 0) {
   731→      const init: Record<string, number> = {}
   732→      for (const p of low) init[p.id] = suggested(p)
   733→      setQtys(init)
   734→    }
   735→  }, [low.length])
   736→
   737→  const groups = useMemo(() => {
   738→    const m = new Map<string, Product[]>()
   739→    for (const p of low) {
   740→      const key = p.providerId || '_none'
   741→      if (!m.has(key)) m.set(key, [])
   742→      m.get(key)!.push(p)
   743→    }
   744→    return [...m.entries()]
   745→  }, [low])
   746→
   747→  const activeGroups = groups.filter(([pid]) => !excluded[pid])
   748→  const activeItems = activeGroups.flatMap(([, ps]) => ps).filter((p) => !skipped[p.id])
   749→  const estTotal = activeItems.reduce((s, p) => s + (qtys[p.id] || 0) * p.buyPrice, 0)
   750→
   751→  const create = async () => {
   752→    if (activeItems.length === 0) return
   753→    setBusy(true)
   754→    const d = new Date()
   755→    d.setDate(d.getDate() + 2)
   756→    const deliveryDate = d.toISOString().slice(0, 10)
   757→    let created = 0
   758→    try {
   759→      for (const [pid, ps] of activeGroups) {
   760→        const items = ps.filter((p) => !skipped[p.id])
   761→        if (items.length === 0) continue
   762→        const prov = providers.find((x) => x.id === pid)
   763→        await api('/api/orders', {
   764→          method: 'POST',
   765→          body: {
   766→            providerId: pid,
   767→            providerName: prov?.name || 'بدون تأمین‌کننده',
   768→            status: 'DRAFT',
   769→            deliveryDate,
   770→            notes: 'پیش‌نویس هوشمند — بر اساس نقطه سفارش و موجودی فعلی',
   771→            items: items.map((p) => ({ productId: p.id, qty: qtys[p.id] || suggested(p), unitBuyPrice: p.buyPrice })),
   772→          },
   773→        })
   774→        created++
   775→      }
   776→      onDone(created)
   777→    } catch (e: any) {
   778→      toast.error(e.message)
   779→    } finally {
   780→      setBusy(false)
   781→    }
   782→  }
   783→
   784→  return (
   785→    <Modal title="پیش‌نویس هوشمند سفارش 🤖" onClose={onClose} wide>
   786→      <p className="mb-4 rounded-xl border border-[#77934a]/30 bg-[#e9f0e4]/60 p-3 text-[11px] leading-5">
   787→        سامانه {faNum(low.length)} کالای رسیده به نقطه سفارش را پیدا کرده و بر اساس «دو برابر نقطه سفارش منهای موجودی فعلی» پیشنهاد تعداد می‌دهد.
   788→        تعدادها را اصلاح کنید، تأمین‌کننده‌های غیرضروری را بردارید و سفارش‌های پیش‌نویس را یک‌جا بسازید — بعد از بخش سفارش‌ها ارسالشان کنید.
   789→      </p>
   790→
   791→      {!loaded ? (
   792→        <div className="space-y-2">
   793→          {Array.from({ length: 4 }).map((_, i) => <div key={i} className="h-12 animate-pulse rounded-xl bg-muted/60" />)}
   794→        </div>
   795→      ) : low.length === 0 ? (
   796→        <EmptyState emoji="🌿" title="همه موجودی‌ها سالم‌اند" hint="فعلاً کالایی به نقطه سفارش نرسیده" />
   797→      ) : (
   798→        <div className="scroll-gold max-h-[52vh] space-y-3 overflow-y-auto pl-1">
   799→          {groups.map(([pid, ps]) => {
   800→            const prov = providers.find((x) => x.id === pid)
   801→            const isExcluded = !!excluded[pid]
   802→            const groupTotal = ps.reduce((s, p) => s + (qtys[p.id] || 0) * p.buyPrice, 0)
   803→            return (
   804→              <div key={pid} className={cn('rounded-2xl border p-3 transition', isExcluded ? 'border-border bg-muted/40 opacity-60' : 'glow-card border-[#77934a]/30 bg-white/80')}>
   805→                <div className="mb-2 flex flex-wrap items-center justify-between gap-2">
   806→                  <label className="flex cursor-pointer items-center gap-2">
   807→                    <input
   808→                      type="checkbox"
   809→                      checked={!isExcluded}
   810→                      onChange={(e) => setExcluded((x) => ({ ...x, [pid]: !e.target.checked }))}
   811→                      className="h-4 w-4 accent-[#77934a]"
   812→                    />
   813→                    <span className="text-sm font-black">🚚 {prov?.name || 'بدون تأمین‌کننده مشخص'}</span>
   814→                    <span className="rounded-full bg-secondary px-2 py-0.5 text-[10px] font-bold text-[#5c7236]">{faNum(ps.length)} قلم</span>
   815→                  </label>
   816→                  <span className="text-[11px] font-black text-[#8a6d10]">{faMoney(groupTotal)} تومان</span>
   817→                </div>
   818→                <div className="grid gap-1.5 sm:grid-cols-2">
   819→                  {ps.map((p) => (
   820→                    <div key={p.id} className={cn('flex items-center justify-between gap-2 rounded-xl border border-border/70 px-2.5 py-2', skipped[p.id] && 'opacity-40')}>
   821→                      <label className="flex min-w-0 cursor-pointer items-center gap-2">
   822→                        <input
   823→                          type="checkbox"
   824→                          checked={!skipped[p.id]}
   825→                          disabled={isExcluded}
   826→                          onChange={(e) => setSkipped((x) => ({ ...x, [p.id]: !e.target.checked }))}
   827→                          className="h-3.5 w-3.5 accent-[#0e7a4a]"
   828→                        />
   829→                        <span className="min-w-0">
   830→                          <span className="block truncate text-[11px] font-bold">{p.name}</span>
   831→                          <span className="block text-[9px] text-muted-foreground">
   832→                            موجودی {faNum(p.stock)} • نقطه سفارش {faNum(p.reorderLevel)} {p.stock === 0 ? '• 🔴 ناموجود' : ''}
   833→                          </span>
   834→                        </span>
   835→                      </label>
   836→                      <span className="flex shrink-0 items-center gap-1">
   837→                        <button
   838→                          disabled={isExcluded}
   839→                          onClick={() => setQtys((q) => ({ ...q, [p.id]: Math.max((q[p.id] || 1) - 1, 1) }))}
   840→                          className="stepper-btn !h-8 !w-8 text-sm disabled:opacity-30"
   841→                        >
   842→                          −
   843→                        </button>
   844→                        <span className="w-10 text-center text-xs font-black">{faNum(qtys[p.id] || 0)}</span>
   845→                        <button
   846→                          disabled={isExcluded}
   847→                          onClick={() => setQtys((q) => ({ ...q, [p.id]: (q[p.id] || 0) + 1 }))}
   848→                          className="stepper-btn !h-8 !w-8 text-sm disabled:opacity-30"
   849→                        >
   850→                          +
   851→                        </button>
   852→                      </span>
   853→                    </div>
   854→                  ))}
   855→                </div>
   856→              </div>
   857→            )
   858→          })}
   859→        </div>
   860→      )}
   861→
   862→      <div className="mt-4 flex flex-wrap items-center justify-between gap-3 border-t border-border pt-4">
   863→        <div className="text-[11px] font-bold text-muted-foreground">
   864→          {faNum(activeItems.length)} قلم در {faNum(activeGroups.length)} سفارش پیش‌نویس • برآورد خرید: <b className="text-[#8a6d10]">{faMoney(estTotal)} تومان</b>
   865→        </div>
   866→        <div className="flex gap-2">
   867→          <button onClick={onClose} className="rounded-xl border px-4 py-2.5 text-xs font-bold">انصراف</button>
   868→          <button
   869→            onClick={create}
   870→            disabled={busy || activeItems.length === 0}

   872→          >
   873→            <Sparkles size={14} /> {busy ? 'در حال ساخت…' : `ساخت ${faNum(activeGroups.length)} سفارش پیش‌نویس`}
