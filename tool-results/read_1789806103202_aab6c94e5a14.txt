   142→/** برچسب ناوبری چندزبانه — در نبود ترجمه، برچسب فارسی اصلی */
   143→function navLabel(key: string, faLabel: string, lang: string): string {
   144→  if (lang === 'fa') return faLabel
   145→  const v = tI18n(`nav.${key}`)
   146→  return v.startsWith('nav.') ? faLabel : v
   147→}
   148→function groupTitle(faTitle: string, lang: string): string {
   149→  if (lang === 'fa') return faTitle
   150→  const v = tI18n(GROUP_TITLE_KEYS[faTitle] || '')
   151→  return v.startsWith('nav.') ? faTitle : v
   152→}

   154→/** تاریخ امروز به‌همراه تعطیلی (اگر تعطیل است) — چیپ زیبای نوار بالا */
   155→function TopbarDateChip() {
   156→  const holidays = useHolidays()
   157→  const today = todayIso()
   158→  const holidayName = holidays.get(today)
   159→  const isWeekend = [4, 5].includes(new Date().getDay()) // پنجشنبه/جمعه
   160→  return (
   161→    <div
   162→      className={cn(
   163→        'hidden items-center gap-2 rounded-xl border px-3 py-1.5 md:flex',
   164→        holidayName
   165→          ? 'border-[#b3372f]/30 bg-[#fee2e2]/50'
   166→          : isWeekend
   167→            ? 'border-[#c9a227]/40 bg-[#fdf6dd]/60'
   168→            : 'border-border bg-card'
   169→      )}
   170→      title={holidayName || undefined}
   171→    >
   172→      <CalendarDays size={15} className={holidayName || isWeekend ? 'text-[#b3372f]' : 'text-[#0e7a4a]'} />
   173→      <span className="text-[11px] font-extrabold text-foreground">{formatJalaliFull(todayIso())}</span>
   174→      {holidayName && (
   175→        <span className="rounded-lg bg-[#b3372f] px-2 py-0.5 text-[10px] font-black text-white">تعطیل — {holidayName}</span>
   176→      )}
   177→    </div>
   178→  )
   179→}

   181→function parseHash(): { view: string; param: string } {
   182→  const h = window.location.hash.replace(/^#\/?/, '')
   183→  const [view, param] = h.split('/')
   184→  return { view: view || 'dashboard', param: param || '' }

   209→  const fireDesktopNotifs = useCallback((list: { id: string; severity: string; icon: string; title: string; detail: string; count?: number }[]) => {
   210→    try {
   211→      if (!('Notification' in window) || Notification.permission !== 'granted') return
   212→      const enabled = localStorage.getItem('hz-desktop-notifs') === '1'
   213→      const seen: Record<string, number> = JSON.parse(localStorage.getItem('hz-desktop-seen') || '{}')
   214→      const hidden = document.visibilityState === 'hidden'
   215→      for (const n of list) {
   216→        const cur = n.count ?? 1
   217→        const wasSeen = seen[n.id]
   218→        const isNew = wasSeen === undefined || cur > wasSeen
   219→        if (enabled && hidden && isNew && n.severity === 'critical') {
   220→          try {
   221→            const notif = new Notification('هایپر زیتون 🫒', {
   222→              body: `${n.icon} ${n.title}\n${n.detail.slice(0, 110)}`,
   223→              tag: n.id,
   224→            })
   225→            notif.onclick = () => {
   226→              window.focus()
   227→              notif.close()
   228→            }
   229→          } catch {
   230→            /* some engines throw on constructor — ignore */
   231→          }
   232→        }
   233→        seen[n.id] = cur
   234→      }
   235→      // prune keys that no longer exist
   236→      const live = new Set(list.map((n) => n.id))
   237→      for (const k of Object.keys(seen)) if (!live.has(k)) delete seen[k]
   238→      localStorage.setItem('hz-desktop-seen', JSON.stringify(seen))
   239→    } catch {
   240→      /* storage unavailable — silent */
   241→    }
   242→  }, [])

   248→      localStorage.setItem('hz-desktop-notifs', '0')

   250→      toast.info('اعلان دسکتاپ خاموش شد — اعلان‌ها فقط داخل سامانه می‌آیند')
