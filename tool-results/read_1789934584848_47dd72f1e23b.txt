   110→/** کلید لاتین برای نقش سفارشی — از نام اگر لاتین داشت، وگرنه خالی (کاربر باید وارد کند) */
   111→function latinKey(s: string): string {
   112→  return s
   113→    .trim()
   114→    .toUpperCase()
   115→    .replace(/[^A-Z0-9_]+/g, '_')
   116→    .replace(/^_+|_+$/g, '')
   117→}

   119→function Chip({ label, checked, color, onClick, disabled }: { label: React.ReactNode; checked: boolean; color: string; onClick: () => void; disabled?: boolean }) {
   120→  return (
   121→    <button
   122→      type="button"
   123→      disabled={disabled}
   124→      onClick={onClick}
   125→      className={cn(
   126→        'rounded-xl border px-3 py-2 text-xs font-bold transition-all active:scale-95 disabled:opacity-50',
   127→        checked ? 'border-transparent text-white shadow-md' : 'border-border bg-card text-muted-foreground hover:border-[#0e7a4a]/40'
   128→      )}
   129→      style={checked ? { background: color } : undefined}
   130→    >
   131→      {checked && <CheckIcon />} {label}
   132→    </button>
   133→  )
   134→}

   154→
   155→  const user = ctx.user
   156→  const rawRoleIds = (user as unknown as { roleIds?: string | string[] } | null)?.roleIds
   157→  const myRoleIds: string[] = Array.isArray(rawRoleIds)

   161→          try {

   199→      if (!m.has(cat)) m.set(cat, [])
   200→      m.get(cat)!.push(r)
   201→    }

   205→  const managed = manageId ? users.find((u) => u.id === manageId) || null : null

   208→  const saveRole = async () => {
   209→    if (!editor) return
   210→    if (!editor.name.trim()) {
   211→      toast.error('نام نقش الزامی است')
   212→      return
   213→    }
   214→    setSaving(true)
   215→    const payload = {
   216→      name: editor.name.trim(),
   217→      category: editor.category.trim() || 'سفارشی',
   218→      color: editor.color,
   219→      description: editor.description.trim(),
   220→      views: editor.views,
   221→      caps: editor.caps,
   222→    }

   225→        await api('/api/roles', { method: 'PATCH', body: { id: editor.id, ...payload, active: editor.active } })

   228→        // کلید لاتین لازم است (نام فارسی به‌تنهایی کلید نمی‌شود) — از فیلد اختصاصی یا حروف لاتینِ نام

   232→          setSaving(false)
   233→          return

   235→        await api('/api/roles', { method: 'POST', body: { ...payload, key } })
   236→        toast.success(`نقش جدید ساخته شد 🌿 (custom_${key})`)
   237→      }
   238→      setEditor(null)
   239→      await load()
   240→    } catch (e) {
   241→      toast.error((e as Error).message)

   285→      toast.success('کاربر به‌روزرسانی شد ✓')
   286→      setManageId(null)
   287→      await load()
   288→    } catch (e) {
   289→      toast.error((e as Error).message)
   290→    } finally {
   291→      setSaving(false)
   292→    }
   293→  }
   294→

   304→

   324→        icon={<ShieldCheck size={20} />}

   326→          <button

   332→            className="flex min-h-[44px] items-center gap-1.5 rounded-xl bg-primary px-4 py-2.5 text-sm font-bold text-white shadow-md transition hover:bg-primary/90 active:scale-95"
   333→          >
   334→            <Plus size={16} /> نقش جدید

   336→        }
   337→      >
   338→        <div className="flex items-start gap-2 rounded-xl border border-[#e9b90c]/50 bg-[#fdf6dd] px-3.5 py-3 text-xs font-bold leading-relaxed text-[#8a6d10]">
   339→          <Info size={15} className="mt-0.5 shrink-0" />
   340→          مالک، مدیر کل، مدیر عملیات و مدیر سامانه همیشه به همهٔ بخش‌ها دسترسی دارند و محدود نمی‌شوند.

   389→                        description: r.description,
   390→                        views: r.permissions?.views || [],
   391→                        caps: r.permissions?.caps || [],
   392→                        active: r.active,

   598→
