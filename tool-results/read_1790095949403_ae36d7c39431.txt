   169→/** برچسب ناوبری چندزبانه — در نبود ترجمه، برچسب فارسی اصلی */
   170→function navLabel(key: string, faLabel: string, lang: string): string {
   171→  if (lang === 'fa') return faLabel
   172→  const v = tI18n(`nav.${key}`)
   173→  return v.startsWith('nav.') ? faLabel : v
   174→}
   175→function groupTitle(faTitle: string, lang: string): string {
   176→  if (lang === 'fa') return faTitle
   177→  const v = tI18n(GROUP_TITLE_KEYS[faTitle] || '')
   178→  return v.startsWith('nav.') ? faTitle : v
   179→}

   224→  const fireDesktopNotifs = useCallback((list: { id: string; severity: string; icon: string; title: string; detail: string; count?: number }[]) => {
   225→    try {
   226→      if (!('Notification' in window) || Notification.permission !== 'granted') return

   228→      const seen: Record<string, number> = JSON.parse(localStorage.getItem('hz-desktop-seen') || '{}')
   229→      const hidden = document.visibilityState === 'hidden'
   230→      for (const n of list) {
   231→        const cur = n.count ?? 1
   232→        const wasSeen = seen[n.id]
   233→        const isNew = wasSeen === undefined || cur > wasSeen

   235→          try {
   236→            const notif = new Notification(`${businessName} • سررسید`, {
   237→              body: `${n.icon} ${n.title}\n${n.detail.slice(0, 110)}`,

   241→              window.focus()
   242→              notif.close()
   243→            }
   244→          } catch {
   245→            /* some engines throw on constructor — ignore */
   246→          }
   247→        }
   248→        seen[n.id] = cur
   249→      }
   250→      // prune keys that no longer exist
   251→      const live = new Set(list.map((n) => n.id))
   252→      for (const k of Object.keys(seen)) if (!live.has(k)) delete seen[k]
   253→      localStorage.setItem('hz-desktop-seen', JSON.stringify(seen))
   254→    } catch {
   255→      /* storage unavailable — silent */
   256→    }
