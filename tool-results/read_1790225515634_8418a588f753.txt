   114→  const tabs: { key: Tab; label: string; sub: string; icon: React.ReactNode }[] = [

   117→  ]
   118→  return (
   119→    <div role="tablist" aria-label="حالت نمایش داشبورد" className="flex justify-center">
   120→      <div className="glow-card inline-flex w-full gap-1 rounded-2xl bg-card p-1.5 sm:w-auto">
   121→        {tabs.map((t) => {
   122→          const active = tab === t.key
   123→          return (
   124→            <button
   125→              key={t.key}
   126→              role="tab"
   127→              aria-selected={active}
   128→              onClick={() => onChange(t.key)}
   129→              className={cn(
   130→                'flex min-h-11 flex-1 items-center justify-center gap-2 rounded-xl px-4 transition-all sm:flex-none sm:px-5',
   131→                active ? 'bg-primary text-primary-foreground shadow-sm' : 'text-muted-foreground hover:bg-secondary/70'
   132→              )}
   133→            >
   134→              {t.icon}
   135→              <span className="text-xs font-black">{t.label}</span>
   136→              <span className={cn('hidden text-[10px] font-bold md:inline', active ? 'text-primary-foreground/70' : 'text-muted-foreground/80')}>
   137→                · {t.sub}
   138→              </span>
   139→            </button>
   140→          )
   141→        })}
   142→      </div>
   143→    </div>
   144→  )
   145→}

   179→/** چیپ فاصلهٔ زمانی تا سررسید — دیرکرد قرمز، امروز/فردا طلایی */
   180→function DueChip({ days }: { days: number }) {
   181→  if (days < 0) return <Pill label={`${faNum(Math.abs(days))} روز دیرکرد`} color="#b3372f" />
   182→  if (days === 0) return <Pill label="امروز" color="#8a6d10" />
   183→  if (days === 1) return <Pill label="فردا" color="#a16207" />
   184→  return <Pill label={`${faNum(days)} روز مانده`} color={days <= 5 ? '#77934a' : '#6d7a6e'} />
   185→}
