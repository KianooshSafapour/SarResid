   119→   119→function NewOrderWizard({ ctx, onDone }: { ctx: AppCtx; onDone: (id: string) => void }) {
   120→   120→  const holidays = useHolidays()

   122→   122→  const [providers, setProviders] = useState<Provider[]>([])
   123→   123→  const [products, setProducts] = useState<Product[]>([])
   124→   124→  const [providerId, setProviderId] = useState('')
   125→   125→  const [company, setCompany] = useState('')
   126→   126→  const [draft, setDraft] = useState<Record<string, number>>({})
   127→   127→  const [deliveryDate, setDeliveryDate] = useState('')
   128→   128→  const [payMethod, setPayMethod] = useState('CASH')
   129→   129→  const [notes, setNotes] = useState('')
   130→   130→  const [detailFor, setDetailFor] = useState<Product | null>(null)
   131→   131→  const [busy, setBusy] = useState(false)
   132→   132→  const [margins, setMargins] = useState<Record<string, { marginPct: number; buyPrice: number; sellPrice: number; orderCode: string; at: string }>>({})
   133→   133→
   134→   134→  function add(days: number) { const d = new Date(); d.setDate(d.getDate() + days); return d.toISOString().slice(0, 10) }
   135→   135→
   136→   136→  useEffect(() => {
   137→   137→    api<{ providers: Provider[] }>('/api/providers').then((d) => setProviders(d.providers))
   138→   138→    api<{ products: Product[] }>('/api/products').then((d) => setProducts(d.products))
   139→   139→    setDeliveryDate(add(1))
   140→   140→  }, [])
   141→   141→
   142→   142→  // حاشیه سود آخرین خریدها از همین تأمین‌کننده — «همان کالا از منابع مختلف، حاشیهٔ هر منبع»
   143→   143→  useEffect(() => {
   144→   144→    if (!providerId) { setMargins({}); return }
   145→   145→    api<{ margins: Record<string, { marginPct: number; buyPrice: number; sellPrice: number; orderCode: string; at: string }> }>(`/api/orders/margins?providerId=${providerId}`)
   146→   146→      .then((d) => setMargins(d.margins || {}))
   147→   147→      .catch(() => setMargins({}))
   148→   148→  }, [providerId])
   149→   149→
   150→   150→  const marginTone = (pct: number) => (pct < 10 ? { c: '#b3372f', l: 'کم' } : pct < 25 ? { c: '#a16207', l: 'متوسط' } : { c: '#166534', l: 'خوب' })
   151→   151→
   152→   152→  const provider = providers.find((p) => p.id === providerId)
   153→   153→  const isGM = ctx.user!.role === 'GM' || ctx.user!.role === 'OM'
   154→   154→  const prodsOfCompany = useMemo(() => {
   155→   155→    if (!provider) return []
   156→   156→    return products.filter((p) => p.providerId === providerId && (!company || p.brand === company))
   157→   157→  }, [products, provider, company, providerId])
   158→   158→  const draftItems: DraftItem[] = Object.entries(draft).filter(([, q]) => q > 0).map(([pid, qty]) => {
   159→   159→    const p = products.find((x) => x.id === pid)!
   160→   160→    return { productId: pid, name: p.name, qty, unitBuyPrice: p.buyPrice }
   161→   161→  })
   162→   162→  const total = draftItems.reduce((s, i) => s + i.qty * i.unitBuyPrice, 0)
   163→   163→
   164→   164→  const submit = async (approve: boolean) => {
   165→   165→    if (!provider || draftItems.length === 0 || !deliveryDate) return
   166→   166→    setBusy(true)
   167→   167→    try {
   168→   168→      const res = await api<{ order: any }>('/api/orders', {
   169→   169→        method: 'POST',
   170→   170→        body: {
   171→   171→          providerId, providerName: provider.name,
   172→   172→          items: draftItems,
   173→   173→          deliveryDate, payMethod, notes,
   174→   174→          status: approve ? 'APPROVED' : 'SUBMITTED',
   175→   175→        },
   176→   176→      })
   177→   177→      toast.success(approve ? 'سفارش ثبت و تأیید شد ✅' : 'سفارش ثبت شد — در انتظار تأیید مدیر کل')
   178→   178→      onDone(res.order.id)
   179→   179→    } catch (e: any) {
   180→   180→      toast.error(e.message)
   181→   181→    } finally { setBusy(false) }
   182→   182→  }
   183→   183→
   184→   184→  return (
   185→   185→    <div className="space-y-4">
   186→   186→      {/* steps */}
   187→   187→      <div className="flex items-center gap-2 rounded-2xl border border-border bg-card p-3">
   188→   188→        {[1, 2, 3, 4].map((s) => (
   189→   189→          <div key={s} className="flex flex-1 items-center gap-2">
   190→   190→            <span className={cn('flex h-8 w-8 shrink-0 items-center justify-center rounded-full text-xs font-black', step >= s ? 'bg-primary text-white' : 'bg-muted text-muted-foreground')}>{faNum(s)}</span>
   191→   191→            <span className={cn('hidden text-xs font-bold sm:block', step >= s ? 'text-foreground' : 'text-muted-foreground')}>
   192→   192→              {s === 1 ? 'تأمین‌کننده' : s === 2 ? 'انتخاب کالا' : s === 3 ? 'زمان و پرداخت' : 'بازبینی و ثبت'}
   193→   193→            </span>
   194→   194→            {s < 4 && <span className={cn('h-0.5 flex-1 rounded', step > s ? 'bg-primary' : 'bg-border')} />}
   195→   195→          </div>
   196→   196→        ))}
   197→   197→      </div>
   198→   198→
   199→   199→      {step === 1 && (
   200→   200→        <SectionCard title="انتخاب تأمین‌کننده / ویزیتور" subtitle="شرکت‌هایی که هر پخش‌کننده کار می‌کند را ببینید" icon={<Plus size={18} />}>
   201→   201→          <div className="grid gap-2.5 sm:grid-cols-2 lg:grid-cols-3">
   202→   202→            {providers.map((p) => (
   203→   203→              <button key={p.id} onClick={() => { setProviderId(p.id); setCompany(''); setStep(2) }}
   204→   204→                className={cn('glow-card rounded-2xl bg-white/70 p-4 text-right', providerId === p.id && 'ring-2 ring-primary')}>
   205→   205→                <div className="flex items-center justify-between">
   206→   206→                  <span className="text-sm font-black">{p.name}</span>
   207→   207→                  <Pill label={p.type === 'DIRECT' ? 'کارخانه' : p.type === 'VISITOR' ? 'ویزیتور' : 'پخش'} color={p.type === 'DIRECT' ? '#0e7a4a' : p.type === 'VISITOR' ? '#c96f4a' : '#8a6d10'} />
   208→   208→                </div>
   209→   209→                <p className="mt-1 text-xs text-muted-foreground">{p.personName} • {faNum(p.phone)}</p>
   210→   210→                <div className="mt-2 flex flex-wrap gap-1">
   211→   211→                  {p.companyNames.map((c) => <Pill key={c} label={c} color="#0e7a4a" className="!bg-secondary" />)}

   213→   213→              </button>
   214→   214→            ))}
   215→   215→          </div>
   216→   216→        </SectionCard>
   217→   217→      )}
   218→   218→
   219→   219→      {step === 2 && provider && (
   220→   220→        <SectionCard
   221→   221→          title={`کالاهای ${provider.name}`}
   222→   222→          subtitle="روی دکمه + بزنید تا به سفارش اضافه شود — رنگ نشانگر، وضعیت انبار است"
   223→   223→          icon={<Plus size={18} />}
   224→   224→          actions={<button onClick={() => setStep(1)} className="rounded-lg border px-3 py-1.5 text-xs font-bold">← تغییر تأمین‌کننده</button>}

   226→   226→          {provider.companyNames.length > 1 && (
   227→   227→            <div className="mb-3 flex flex-wrap gap-1.5">
   228→   228→              <FilterChip active={!company} label="همه شرکت‌ها" onClick={() => setCompany('')} />

   233→   233→            {prodsOfCompany.map((p) => {

   696→   696→function CheckoutModal({ order, items, onClose, onDone }: { order: any; items: any[]; onClose: () => void; onDone: () => void }) {
   697→   697→  const holidays = useHolidays()
   698→   698→  const [rows, setRows] = useState(() =>
   699→   699→    items.map((it) => ({
   700→   700→      id: it.id as string,
   701→   701→      productName: it.productName as string,
   702→   702→      qty: it.qty as number,
   703→   703→      unitBuyPrice: it.unitBuyPrice as number,
   704→   704→      receivedQty: it.receivedQty ?? it.qty,
   705→   705→      missingQty: it.missingQty || 0,
   706→   706→      returned: !!it.returned,
   707→   707→      returnReason: it.returnReason || '',
   708→   708→      printedPrice: it.printedPrice ?? '',
   709→   709→      expiryDate: it.expiryDate || '',
   710→   710→      note: it.note || '',
   711→   711→    }))
   712→   712→  )
   713→   713→  const [checkNote, setCheckNote] = useState('')
   714→   714→  const [busy, setBusy] = useState(false)
   715→   715→
   716→   716→  const patchRow = (id: string, patch: Partial<(typeof rows)[number]>) =>
   717→   717→    setRows((rs) => rs.map((r) => (r.id === id ? { ...r, ...patch } : r)))
   718→   718→
   719→   719→  // جمع‌ها: سفارش اولیه vs نهایی (موثر = بدون مرجوعی و منهای کسری)
   720→   720→  const initialTotal = rows.reduce((s, r) => s + r.qty * r.unitBuyPrice, 0)
   721→   721→  const finalTotal = rows.reduce((s, r) => s + (r.returned ? 0 : Math.max(0, Number(r.receivedQty ?? r.qty) - r.missingQty)) * r.unitBuyPrice, 0)
   722→   722→  const returnedCount = rows.filter((r) => r.returned).length
   723→   723→  const missingCount = rows.filter((r) => !r.returned && r.missingQty > 0).length
   724→   724→  const diffColor = finalTotal === initialTotal ? '#0e7a4a' : '#b3372f'
   725→   725→
   726→   726→  const submit = async () => {
   727→   727→    setBusy(true)
   728→   728→    try {
   729→   729→      await api(`/api/orders/${order.id}`, {
   730→   730→        method: 'PATCH',
   731→   731→        body: {
   732→   732→          action: 'checkout',
   733→   733→          checkNote,
   734→   734→          items: rows.map((r) => ({
   735→   735→            id: r.id,
   736→   736→            receivedQty: Number(r.receivedQty) || 0,
   737→   737→            missingQty: Number(r.missingQty) || 0,
   738→   738→            returned: r.returned,
   739→   739→            returnReason: r.returnReason,
   740→   740→            printedPrice: r.printedPrice === '' ? undefined : Number(r.printedPrice),
   741→   741→            expiryDate: r.expiryDate || undefined,
   742→   742→            note: r.note,
   743→   743→          })),
   744→   744→        },
   745→   745→      })
   746→   746→      toast.success('تحویل نهایی شد و در حسابداری ثبت گردید ✓')
   747→   747→      onDone()
   748→   748→    } catch (e: any) { toast.error(e.message) } finally { setBusy(false) }
   749→   749→  }
   750→   750→
   751→   751→  return (
   752→   752→    <Modal title={`تحویل و نهایی‌سازی ${order.code} — بازبینی حسابدار`} onClose={onClose} wide>
   753→   753→      <p className="rounded-xl bg-[#fdf6dd]/70 px-3 py-2 text-[11px] leading-5 text-[#8a5a2b]">
   754→   754→        کالای دریافتی را با فاکتور مطابقت دهید: قیمت چاپی اشتباه، انقضای کوتاه، کسری و مرجوعی را همین‌جا اصلاح کنید. مبلغ نهایی به‌صورت خودکار بازمحاسبه می‌شود.
   755→   755→      </p>
   756→   756→      <div className="space-y-2">
   757→   757→        {rows.map((r) => (
   758→   758→          <div key={r.id} className={cn('rounded-2xl border p-3', r.returned ? 'border-[#b3372f]/40 bg-[#b3372f]/5' : 'border-border bg-white/70')}>
   759→   759→            <div className="flex flex-wrap items-center justify-between gap-2">
   760→   760→              <p className="text-xs font-black">{r.productName}</p>
   761→   761→              <div className="flex items-center gap-2 text-[10.5px] font-bold text-muted-foreground">
   762→   762→                <span>سفارش: {faNum(r.qty)} عدد</span>
   763→   763→                <span>• واحد: {faMoney(r.unitBuyPrice)}</span>
   764→   764→                <label className="flex cursor-pointer items-center gap-1 rounded-lg bg-[#b3372f]/10 px-2 py-1 text-[10.5px] font-black text-[#b3372f]">
   765→   765→                  <input type="checkbox" checked={r.returned} onChange={(e) => patchRow(r.id, { returned: e.target.checked })} className="accent-[#b3372f]" />
   766→   766→                  مرجوع
   767→   767→                </label>
   768→   768→              </div>
   769→   769→            </div>
   770→   770→            {r.returned && (
   771→   771→              <Labeled label="دلیل مرجوعی (حداقل ۳ حرف) *">
   772→   772→                <input value={r.returnReason} onChange={(e) => patchRow(r.id, { returnReason: e.target.value })} placeholder="مثلاً: قیمت چاپی با فاکتور نمی‌خواند" className="w-full rounded-xl border border-input p-2 text-xs" />
   773→   773→              </Labeled>
   774→   774→            )}
   775→   775→            <div className="mt-2 grid grid-cols-2 gap-2 sm:grid-cols-4">
   776→   776→              <Labeled label="دریافتی">
   777→   777→                <input type="number" min={0} dir="ltr" value={r.receivedQty} onChange={(e) => patchRow(r.id, { receivedQty: Number(e.target.value) })} className="w-full rounded-xl border border-input p-2 text-center text-xs font-bold" />
   778→   778→              </Labeled>
   779→   779→              <Labeled label="کسری">
   780→   780→                <input type="number" min={0} dir="ltr" value={r.missingQty} onChange={(e) => patchRow(r.id, { missingQty: Number(e.target.value) || 0 })} className="w-full rounded-xl border border-input p-2 text-center text-xs font-bold" />
   781→   781→              </Labeled>
   782→   782→              <Labeled label="قیمت چاپی">
   783→   783→                <input type="number" min={0} dir="ltr" value={r.printedPrice} onChange={(e) => patchRow(r.id, { printedPrice: e.target.value === '' ? '' : Number(e.target.value) })} placeholder="—" className="w-full rounded-xl border border-input p-2 text-center text-xs font-bold" />
   784→   784→              </Labeled>
   785→   785→              <Labeled label="انقضا">
   786→   786→                <JalaliDatePicker value={r.expiryDate} onChange={(iso) => patchRow(r.id, { expiryDate: iso })} holidays={holidays} compact />
   787→   787→              </Labeled>
   788→   788→            </div>
   789→   789→            <Labeled label="یادداشت قلم">
   790→   790→              <input value={r.note} onChange={(e) => patchRow(r.id, { note: e.target.value })} className="mt-1 w-full rounded-xl border border-input p-2 text-xs" placeholder="مثلاً: ۲ عدد اضافه آمد — مرجوع شد" />
   791→   791→            </Labeled>
   792→   792→          </div>
   793→   793→        ))}
   794→   794→      </div>
   795→   795→
   796→   796→      {/* جمع‌های زنده */}
   797→   797→      <div className="grid grid-cols-2 gap-2 rounded-2xl bg-muted/40 p-3 text-center sm:grid-cols-4">
   798→   798→        <div><p className="text-sm font-black">{faMoney(initialTotal)}</p><p className="text-[10px] font-bold text-muted-foreground">سفارش اولیه</p></div>
   799→   799→        <div><p className="text-sm font-black" style={{ color: diffColor }}>{faMoney(finalTotal)}</p><p className="text-[10px] font-bold text-muted-foreground">مبلغ نهایی تحویل</p></div>
   800→   800→        <div><p className="text-sm font-black text-[#b3372f]">{faNum(returnedCount)}</p><p className="text-[10px] font-bold text-muted-foreground">قلم مرجوع</p></div>
   801→   801→        <div><p className="text-sm font-black text-[#a16207]">{faNum(missingCount)}</p><p className="text-[10px] font-bold text-muted-foreground">قلم دارای کسری</p></div>
   802→   802→      </div>
   803→   803→      <Labeled label="یادداشت بازبینی (checkNote)">
   804→   804→        <textarea value={checkNote} onChange={(e) => setCheckNote(e.target.value)} rows={2} className="w-full rounded-xl border border-input p-3 text-sm" placeholder="مثلاً: فاکتور با نرم‌افزار تطبیق داده شد؛ مغایرت قیمت قلم دوم مرجوع شد" />

   808→   808→      </button>

   813→   813→export function Modal({ title, children, onClose, wide }: { title: string; children: React.ReactNode; onClose: () => void; wide?: boolean }) {
   814→   814→  // الگوی WAI-ARIA dialog: بستن با کلید Escape — یک‌بار اینجا، برای همهٔ مودال‌های سامانه
   815→   815→  useEffect(() => {
   816→   816→    const onKey = (e: KeyboardEvent) => {
   817→   817→      if (e.key === 'Escape') onClose()
   818→   818→    }
   819→   819→    window.addEventListener('keydown', onKey)
   820→   820→    return () => window.removeEventListener('keydown', onKey)
   821→   821→  }, [onClose])
   822→   822→  return (
   823→   823→    <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
   824→   824→      <div className="absolute inset-0 bg-black/50" onClick={onClose} />
   825→   825→      <div className={cn('scroll-gold glow-card max-h-[85vh] w-full overflow-y-auto rounded-2xl bg-card p-5', wide ? 'max-w-2xl' : 'max-w-md')}>
   826→   826→        <div className="mb-4 flex items-center justify-between">
   827→   827→          <h3 className="text-sm font-black">{title}</h3>
   828→   828→          <button onClick={onClose} className="rounded-lg p-1 hover:bg-muted"><X size={18} /></button>
   829→   829→        </div>
   830→   830→        <div className="space-y-3">{children}</div>
   831→   831→      </div>
   832→   832→    </div>
   833→   833→  )
   834→   834→}
