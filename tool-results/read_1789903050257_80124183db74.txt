    19→export default function OrdersView({ ctx }: { ctx: AppCtx }) {
    20→  // param can be 'new' or an order id — mode is derived, not state
    21→  const mode: 'list' | 'new' | 'detail' = ctx.param === 'new' ? 'new' : ctx.param ? 'detail' : 'list'
    22→
    23→  if (mode === 'new') return <NewOrderWizard ctx={ctx} onDone={(id) => ctx.navigate('orders', id)} />
    24→  if (mode === 'detail') return <OrderDetail ctx={ctx} orderId={ctx.param} onBack={() => ctx.navigate('orders')} />
    25→  return <OrdersList ctx={ctx} onOpen={(id) => ctx.navigate('orders', id)} />
    26→}

    30→function OrdersList({ ctx, onOpen }: { ctx: AppCtx; onOpen: (id: string) => void }) {
    31→  const [orders, setOrders] = useState<OrderRow[]>([])
    32→  const [status, setStatus] = useState('')
    33→  const [q, setQ] = useState('')
    34→  const [loading, setLoading] = useState(true)
    35→
    36→  const load = () => api<{ orders: OrderRow[] }>('/api/orders').then((d) => setOrders(d.orders)).finally(() => setLoading(false))
    37→  useEffect(() => { load() }, [])
    38→
    39→  const filtered = useMemo(
    40→    () => orders.filter((o) => (!status || o.status === status) && (!q || o.code.includes(q) || o.providerName.includes(q))),
    41→    [orders, status, q]
    42→  )
    43→
    44→  const canCreate = ['GM', 'PM', 'OM'].includes(ctx.user!.role)
    45→
    46→  return (
    47→    <div className="space-y-4">
    48→      <SectionCard
    49→        title="سفارش‌های تأمین‌کننده"
    50→        subtitle="از ثبت تا تحویل، تأیید انبار و حسابداری — همه در یک مسیر شفاف"
    51→        icon={<ClipboardList size={18} />}
    52→        actions={
    53→          canCreate && (
    54→            <button onClick={() => ctx.navigate('orders', 'new')} className="flex items-center gap-1.5 rounded-xl bg-gradient-to-l from-[#0e7a4a] to-[#12905a] px-4 py-2.5 text-xs font-extrabold text-white shadow-lg transition hover:shadow-xl">
    55→              <Plus size={15} /> سفارش جدید
    56→            </button>
    57→          )
    58→        }
    59→      >
    60→        <div className="mb-4 flex flex-wrap items-center gap-2">
    61→          <SearchInput value={q} onChange={setQ} placeholder="جستجوی کد سفارش یا تأمین‌کننده…" className="w-full sm:w-72" />
    62→          <div className="flex flex-wrap gap-1.5">
    63→            <FilterChip active={!status} label="همه" onClick={() => setStatus('')} />
    64→            {['DRAFT', 'SUBMITTED', 'APPROVED', 'RECEIVING', 'RECEIVED', 'VERIFIED', 'ACCOUNTED', 'DONE', 'CANCELLED'].map((s) => (
    65→              <FilterChip key={s} active={status === s} label={ORDER_STATUSES[s].label} color={ORDER_STATUSES[s].color} onClick={() => setStatus(s)} />
    66→            ))}
    67→          </div>
    68→        </div>
    69→
    70→        {loading ? (
    71→          <div className="space-y-2">{Array.from({ length: 4 }).map((_, i) => <div key={i} className="h-16 animate-pulse rounded-xl bg-muted" />)}</div>
    72→        ) : filtered.length === 0 ? (
    73→          <EmptyState emoji="📦" title="سفارشی یافت نشد" hint="با دکمه «سفارش جدید» اولین سفارش را ثبت کنید" />
    74→        ) : (
    75→          <div className="grid gap-2.5">
    76→            {filtered.map((o) => {
    77→              const st = ORDER_STATUSES[o.status]
    78→              return (
    79→                <button key={o.id} onClick={() => onOpen(o.id)} className="glow-card group rounded-2xl bg-white/70 p-3.5 text-right sm:p-4">
    80→                  <div className="flex flex-wrap items-center gap-3">
    81→                    <span className="flex h-11 w-11 items-center justify-center rounded-xl text-xl" style={{ background: st.bg }}>🚚</span>
    82→                    <div className="min-w-0 flex-1">
    83→                      <div className="flex flex-wrap items-center gap-2">
    84→                        <span className="text-sm font-black">{o.code}</span>
    85→                        <Pill label={st.label} color={st.color} bg={st.bg} />
    86→                        {o.isOverdue && <Pill label="سرآمده ⚠️" color="#b3372f" bg="#fee2e2" />}
    87→                        {o.payMethod === 'CHEQUE' && <Pill label="چکی" color="#8a6d10" bg="#fdf6dd" />}
    88→                      </div>
    89→                      <p className="mt-0.5 text-xs text-muted-foreground">
    90→                        {o.providerName} • {faNum(o.itemsCount)} قلم • ثبت: {o.createdByName} • تحویل: {formatJalaliFull(o.deliveryDate)}
    91→                      </p>
    92→                    </div>
    93→                    <span className="text-sm font-black text-[#8a6d10]">{faMoney(o.totalAmount)}</span>
    94→                  </div>
    95→                </button>
    96→              )
    97→            })}
    98→          </div>
    99→        )}
   100→      </SectionCard>
   101→    </div>
   102→  )
   103→}

   110→      style={active ? { background: color || '#0e7a4a' } : {}}

   119→function NewOrderWizard({ ctx, onDone }: { ctx: AppCtx; onDone: (id: string) => void }) {
   120→  const holidays = useHolidays()
   121→  const [step, setStep] = useState(1)
   122→  const [providers, setProviders] = useState<Provider[]>([])
   123→  const [products, setProducts] = useState<Product[]>([])
   124→  const [providerId, setProviderId] = useState('')
   125→  const [company, setCompany] = useState('')
   126→  const [draft, setDraft] = useState<Record<string, number>>({})
   127→  const [deliveryDate, setDeliveryDate] = useState('')
   128→  const [payMethod, setPayMethod] = useState('CASH')
   129→  const [notes, setNotes] = useState('')
   130→  const [detailFor, setDetailFor] = useState<Product | null>(null)
   131→  const [busy, setBusy] = useState(false)
   132→  const [margins, setMargins] = useState<Record<string, { marginPct: number; buyPrice: number; sellPrice: number; orderCode: string; at: string }>>({})
   133→
   134→  function add(days: number) { const d = new Date(); d.setDate(d.getDate() + days); return d.toISOString().slice(0, 10) }
   135→
   136→  useEffect(() => {
   137→    api<{ providers: Provider[] }>('/api/providers').then((d) => setProviders(d.providers))
   138→    api<{ products: Product[] }>('/api/products').then((d) => setProducts(d.products))
   139→    setDeliveryDate(add(1))
   140→  }, [])
   141→
   142→  // حاشیه سود آخرین خریدها از همین تأمین‌کننده — «همان کالا از منابع مختلف، حاشیهٔ هر منبع»
   143→  useEffect(() => {
   144→    if (!providerId) { setMargins({}); return }
   145→    api<{ margins: Record<string, { marginPct: number; buyPrice: number; sellPrice: number; orderCode: string; at: string }> }>(`/api/orders/margins?providerId=${providerId}`)
   146→      .then((d) => setMargins(d.margins || {}))
   147→      .catch(() => setMargins({}))
   148→  }, [providerId])
   149→
   150→  const marginTone = (pct: number) => (pct < 10 ? { c: '#b3372f', l: 'کم' } : pct < 25 ? { c: '#a16207', l: 'متوسط' } : { c: '#166534', l: 'خوب' })
   151→
   152→  const provider = providers.find((p) => p.id === providerId)
   153→  const isGM = ctx.user!.role === 'GM' || ctx.user!.role === 'OM'
   154→  const prodsOfCompany = useMemo(() => {
   155→    if (!provider) return []
   156→    return products.filter((p) => p.providerId === providerId && (!company || p.brand === company))
   157→  }, [products, provider, company, providerId])
   158→  const draftItems: DraftItem[] = Object.entries(draft).filter(([, q]) => q > 0).map(([pid, qty]) => {
   159→    const p = products.find((x) => x.id === pid)!
   160→    return { productId: pid, name: p.name, qty, unitBuyPrice: p.buyPrice }
   161→  })
   162→  const total = draftItems.reduce((s, i) => s + i.qty * i.unitBuyPrice, 0)
   163→
   164→  const submit = async (approve: boolean) => {
   165→    if (!provider || draftItems.length === 0 || !deliveryDate) return
   166→    setBusy(true)
   167→    try {
   168→      const res = await api<{ order: any }>('/api/orders', {
   169→        method: 'POST',
   170→        body: {
   171→          providerId, providerName: provider.name,
   172→          items: draftItems,
   173→          deliveryDate, payMethod, notes,
   174→          status: approve ? 'APPROVED' : 'SUBMITTED',
   175→        },
   176→      })
   177→      toast.success(approve ? 'سفارش ثبت و تأیید شد ✅' : 'سفارش ثبت شد — در انتظار تأیید مدیر کل')
   178→      onDone(res.order.id)
   179→    } catch (e: any) {
   180→      toast.error(e.message)
   181→    } finally { setBusy(false) }
   182→  }
   183→
   184→  return (
   185→    <div className="space-y-4">
   186→      {/* steps */}
   187→      <div className="flex items-center gap-2 rounded-2xl border border-border bg-card p-3">
   188→        {[1, 2, 3, 4].map((s) => (
   189→          <div key={s} className="flex flex-1 items-center gap-2">
   190→            <span className={cn('flex h-8 w-8 shrink-0 items-center justify-center rounded-full text-xs font-black', step >= s ? 'bg-primary text-white' : 'bg-muted text-muted-foreground')}>{faNum(s)}</span>
   191→            <span className={cn('hidden text-xs font-bold sm:block', step >= s ? 'text-foreground' : 'text-muted-foreground')}>
   192→              {s === 1 ? 'تأمین‌کننده' : s === 2 ? 'انتخاب کالا' : s === 3 ? 'زمان و پرداخت' : 'بازبینی و ثبت'}
   193→            </span>
   194→            {s < 4 && <span className={cn('h-0.5 flex-1 rounded', step > s ? 'bg-primary' : 'bg-border')} />}
   195→          </div>
   196→        ))}
   197→      </div>
   198→
   199→      {step === 1 && (
   200→        <SectionCard title="انتخاب تأمین‌کننده / ویزیتور" subtitle="شرکت‌هایی که هر پخش‌کننده کار می‌کند را ببینید" icon={<Plus size={18} />}>
   201→          <div className="grid gap-2.5 sm:grid-cols-2 lg:grid-cols-3">
   202→            {providers.map((p) => (
   203→              <button key={p.id} onClick={() => { setProviderId(p.id); setCompany(''); setStep(2) }}
   204→                className={cn('glow-card rounded-2xl bg-white/70 p-4 text-right', providerId === p.id && 'ring-2 ring-primary')}>
   205→                <div className="flex items-center justify-between">
   206→                  <span className="text-sm font-black">{p.name}</span>
   207→                  <Pill label={p.type === 'DIRECT' ? 'کارخانه' : p.type === 'VISITOR' ? 'ویزیتور' : 'پخش'} color={p.type === 'DIRECT' ? '#0e7a4a' : p.type === 'VISITOR' ? '#c96f4a' : '#8a6d10'} />
   208→                </div>
   209→                <p className="mt-1 text-xs text-muted-foreground">{p.personName} • {faNum(p.phone)}</p>
   210→                <div className="mt-2 flex flex-wrap gap-1">
   211→                  {p.companyNames.map((c) => <Pill key={c} label={c} color="#0e7a4a" className="!bg-secondary" />)}
   212→                </div>
   213→              </button>
   214→            ))}
   215→          </div>
   216→        </SectionCard>
   217→      )}
   218→
   219→      {step === 2 && provider && (
   220→        <SectionCard
   221→          title={`کالاهای ${provider.name}`}
   222→          subtitle="روی دکمه + بزنید تا به سفارش اضافه شود — رنگ نشانگر، وضعیت انبار است"
   223→          icon={<Plus size={18} />}
   224→          actions={<button onClick={() => setStep(1)} className="rounded-lg border px-3 py-1.5 text-xs font-bold">← تغییر تأمین‌کننده</button>}
   225→        >
   226→          {provider.companyNames.length > 1 && (
   227→            <div className="mb-3 flex flex-wrap gap-1.5">
   228→              <FilterChip active={!company} label="همه شرکت‌ها" onClick={() => setCompany('')} />
   229→              {provider.companyNames.map((c) => <FilterChip key={c} active={company === c} label={c} onClick={() => setCompany(c)} />)}
   230→            </div>
   231→          )}
   232→          <div className="scroll-gold grid max-h-[52vh] grid-cols-2 gap-3 overflow-y-auto pl-1 sm:grid-cols-3 lg:grid-cols-4">
   233→            {prodsOfCompany.map((p) => {
   234→              const st = stockStatus(p.stock, p.reorderLevel)
   235→              const qty = draft[p.id] || 0
   236→              return (
   237→                <div key={p.id} className="glow-card relative flex flex-col rounded-2xl bg-white/80 p-3">
   238→                  <button onClick={() => setDetailFor(p)} className="mb-2 flex h-20 items-center justify-center rounded-xl bg-gradient-to-br from-secondary to-muted text-3xl">
   239→                    {p.imageUrl ? <img src={p.imageUrl} alt={p.name} className="h-full w-full rounded-xl object-cover" /> : CATEGORY_EMOJI[p.category] || '📦'}
   240→                  </button>
   241→                  <p className="line-clamp-2 min-h-9 text-xs font-extrabold leading-4">{p.name}</p>
   242→                  {margins[p.id] && margins[p.id].marginPct >= 0 && (() => { const t = marginTone(margins[p.id].marginPct); return (
   243→                    <span className="mt-1 inline-flex w-fit items-center gap-1 rounded-full px-2 py-0.5 text-[9px] font-black" style={{ background: `${t.c}18`, color: t.c }} title={`آخرین خرید از همین تأمین‌کننده — سفارش ${margins[p.id].orderCode}`}>
   244→                      حاشیه اخیر: {faNum(margins[p.id].marginPct)}٪ ({t.l})
   245→                    </span>
   246→                  ) })()}
   247→                  <div className="mt-1 flex items-center justify-between text-[10px]">
   248→                    <span className="text-muted-foreground">{p.brand}</span>
   249→                    <span className="font-black" style={{ color: st.color }}>{st.label} {faNum(p.stock)}</span>
   250→                  </div>
   251→                  <div className="mt-2 flex items-center justify-between">
   252→                    <span className="text-[11px] font-bold text-[#8a6d10]">{faMoney(p.buyPrice)}</span>
   253→                    <div className="flex items-center gap-1">
   254→                      {qty > 0 && (
   255→                        <>
   256→                          <button className="stepper-btn !h-8 !w-8 !text-base" onClick={() => setDraft((d) => ({ ...d, [p.id]: Math.max(0, (d[p.id] || 0) - 1) }))}>−</button>
   257→                          <span className="w-8 text-center text-sm font-black">{faNum(qty)}</span>
   258→                        </>
   259→                      )}
   260→                      <button className="stepper-btn !h-8 !w-8 !text-base !border-primary !bg-primary !text-white" onClick={() => setDraft((d) => ({ ...d, [p.id]: (d[p.id] || 0) + 1 }))}>+</button>
   261→                    </div>
   262→                  </div>
   263→                  <button onClick={() => setDetailFor(p)} className="mt-1.5 rounded-lg bg-muted/70 py-1 text-[10px] font-bold text-muted-foreground hover:bg-muted">
   264→                    📊 سفارش و فروش ماه اخیر
   265→                  </button>
   266→                </div>
   267→              )
   268→            })}
   269→          </div>
   270→          <div className="mt-4 flex items-center justify-between rounded-xl bg-secondary/60 p-3">
   271→            <span className="text-xs font-bold">{faNum(draftItems.length)} قلم انتخاب شده</span>
   272→            <button disabled={!draftItems.length} onClick={() => setStep(3)} className="rounded-xl bg-primary px-6 py-2.5 text-xs font-extrabold text-white disabled:opacity-40">
   273→              ادامه →
   274→            </button>
   275→          </div>
   276→        </SectionCard>
   277→      )}
   278→
   279→      {step === 3 && (
   280→        <SectionCard title="زمان تحویل و روش پرداخت" subtitle="تقویم شمسی با نشان‌دادن تعطیلات رسمی" icon={<Plus size={18} />}>
   281→          <div className="grid gap-4 sm:grid-cols-2">
   282→            <Labeled label="تاریخ تحویل مورد انتظار">
   283→              <JalaliDatePicker value={deliveryDate} onChange={setDeliveryDate} holidays={holidays} minDate={todayIso()} warnHoliday />
   284→            </Labeled>
   285→            <Labeled label="روش پرداخت">
   286→              <div className="grid grid-cols-2 gap-2">
   287→                {[['CASH', 'نقدی هنگام تحویل 💵'], ['CHEQUE', 'چک 🧾']].map(([v, l]) => (
   288→                  <button key={v} onClick={() => setPayMethod(v)} className={cn('rounded-xl border p-3 text-xs font-bold transition', payMethod === v ? 'border-primary bg-primary/10 text-primary' : 'border-border bg-card hover:bg-secondary')}>
   289→                    {l}
   290→                  </button>
   291→                ))}
   292→              </div>
   293→            </Labeled>
   294→          </div>
   295→          <div className="mt-4">
   296→            <Labeled label="یادداشت سفارش (اختیاری)">
   297→              <textarea value={notes} onChange={(e) => setNotes(e.target.value)} rows={2} className="w-full rounded-xl border border-input bg-white/90 p-3 text-sm outline-none focus:border-primary" placeholder="مثلاً: تحویل قبل ساعت ۹ صبح" />
   298→            </Labeled>
   299→          </div>
   300→          <div className="mt-4 flex justify-between">
   301→            <button onClick={() => setStep(2)} className="rounded-xl border px-5 py-2.5 text-xs font-bold">→ بازگشت</button>
   302→            <button disabled={!deliveryDate} onClick={() => setStep(4)} className="rounded-xl bg-primary px-6 py-2.5 text-xs font-extrabold text-white disabled:opacity-40">بازبینی نهایی ←</button>
   303→          </div>
   304→        </SectionCard>
   305→      )}
   306→
   307→      {step === 4 && (
   308→        <SectionCard title="بازبینی و ثبت سفارش" subtitle="قبل از ثبت، همه چیز را چک کنید" icon={<CheckCircle2 size={18} />}>
   309→          <div className="mb-4 grid gap-2 sm:grid-cols-3">
   310→            <KeyValue k="تأمین‌کننده" v={provider?.name} />
   311→            <KeyValue k="تاریخ تحویل" v={formatJalaliFull(deliveryDate)} />
   312→            <KeyValue k="مبلغ کل" v={`${faMoney(total)} تومان`} />
   313→          </div>
   314→          <ul className="scroll-gold mb-4 max-h-56 space-y-1.5 overflow-y-auto">
   315→            {draftItems.map((i) => (
   316→              <li key={i.productId} className="flex items-center justify-between rounded-lg bg-muted/50 px-3 py-2 text-xs">
   317→                <span className="font-bold">{i.name}</span>
   318→                <span>{faNum(i.qty)} × {faMoney(i.unitBuyPrice)} = <b>{faMoney(i.qty * i.unitBuyPrice)}</b></span>
   319→              </li>
   320→            ))}
   321→          </ul>
   322→          <div className="flex flex-wrap justify-between gap-2">
   323→            <button onClick={() => setStep(3)} className="rounded-xl border px-5 py-2.5 text-xs font-bold">→ بازگشت</button>
   324→            <div className="flex gap-2">
   325→              <button disabled={busy} onClick={() => submit(false)} className="rounded-xl border border-primary bg-secondary px-5 py-2.5 text-xs font-extrabold text-primary">
   326→                ثبت و ارسال برای تأیید {isGM ? '' : '(مدیر کل)'}
   327→              </button>
   328→              {isGM && (
   329→                <button disabled={busy} onClick={() => submit(true)} className="rounded-xl bg-gradient-to-l from-[#0e7a4a] to-[#12905a] px-5 py-2.5 text-xs font-extrabold text-white shadow-lg">
   330→                  ثبت و تأیید نهایی ✅
   331→                </button>
   332→              )}
   333→            </div>
   334→          </div>
   335→        </SectionCard>
   336→      )}
   337→
   338→      {/* product history modal */}
   339→      {detailFor && <ProductHistoryModal product={detailFor} onClose={() => setDetailFor(null)} onOrder={(q) => { setDraft((d) => ({ ...d, [detailFor.id]: (d[detailFor.id] || 0) + q })); setDetailFor(null) }} />}
   340→    </div>
   341→  )
