    30→const WEEKDAY_SHORT = ['ش', 'ی', 'د', 'س', 'چ', 'پ', 'ج']
    31→function wdShort(iso: string): string {
    32→  const d = new Date(iso + 'T12:00:00').getDay()
    33→  return WEEKDAY_SHORT[d === 6 ? 0 : d + 1]
    34→}
    35→/** compact money for calendar cells: ۸۵م / ۱٫۲م / ۹۵۰ه */
    36→function faMoneyShort(v: number): string {
    37→  if (v >= 1_000_000_000) return `${faNum((v / 1_000_000_000).toFixed(1))} میلیارد`
    38→  if (v >= 1_000_000) return `${faNum(Math.round(v / 1_000_000))}م`
    39→  if (v >= 1000) return `${faNum(Math.round(v / 1000))}ه`
    40→  return faNum(v)
    41→}

   309→/** month calendar with per-day amounts + click-to-detail */
   310→function ChequeCalendar({ byDate, holidays, flexRules, jy, jm, setJy, setJm, selDay, onPick }: any) {
   311→  const cells: (number | null)[] = []
   312→  const len = jMonthLength(jy, jm)
   313→  const start = jMonthStartWeekday(jy, jm)
   314→  for (let i = 0; i < start; i++) cells.push(null)
   315→  for (let d = 1; d <= len; d++) cells.push(d)
   316→  const move = (dir: number) => {
   317→    let m = jm + dir, y = jy
   318→    if (m > 12) { m = 1; y++ }
   319→    if (m < 1) { m = 12; y-- }
   320→    setJm(m); setJy(y)
   321→  }
   322→  return (
   323→    <div className="w-full max-w-[340px]">
   324→      <div className="mb-2 flex items-center justify-between">
   325→        <button onClick={() => move(1)} className="rounded-lg px-2 py-1 hover:bg-muted">›</button>
   326→        <span className="text-sm font-extrabold">{J_MONTHS[jm - 1]} {faNum(jy)}</span>
   327→        <button onClick={() => move(-1)} className="rounded-lg px-2 py-1 hover:bg-muted">‹</button>
   328→      </div>
   329→      <div className="grid grid-cols-7 gap-1 text-center text-[9px] font-bold text-muted-foreground">
   330→        {WEEKDAY_SHORT.map((d, i) => <span key={i} className={cn(i === 6 && 'text-[#b3372f]')}>{d}</span>)}
   331→      </div>
   332→      <div className="mt-1 grid grid-cols-7 gap-1">
   333→        {cells.map((d, i) => {
   334→          if (d === null) return <span key={i} />
   335→          const iso = jalaliToIso(jy, jm, d)
   336→          const cheques = byDate.get(iso) || []
   337→          const hol = holidays.get(iso)
   338→          const fri = new Date(iso + 'T12:00:00').getDay() === 5
   339→          const dayTotal = cheques.reduce((s: number, c: Cheque) => s + c.amount, 0)
   340→          const isSel = selDay === iso
   341→          let hijriTiny = ''
   342→          try { hijriTiny = faNum(isoToHijri(iso).hd) } catch { /* ignore */ }
   343→          return (
   344→            <button
   345→              key={i}
   346→              type="button"
   347→              onClick={() => onPick(iso)}
   348→              title={[
   349→                formatJalaliFull(iso),
   350→                formatHijri(iso) && `هجری: ${formatHijri(iso)}`,
   351→                hol && `تعطیل رسمی: ${hol}`,
   352→                cheques.map((c: Cheque) => `${c.recipientName}: ${faMoney(c.amount)}`).join('\n'),
   353→              ].filter(Boolean).join(' • ')}
   354→              className={cn(
   355→                'relative flex h-11 w-full flex-col items-center justify-center rounded-lg text-[11px] font-bold transition',
   356→                isSel ? 'ring-2 ring-[#c9a227]' : '',
   357→                cheques.length ? 'bg-[#c9a227]/20 ring-1 ring-[#c9a227]/70' : hol ? 'bg-[#b3372f]/8 text-[#b3372f]' : 'hover:bg-muted',
   358→                (hol || fri) && !cheques.length && 'text-[#b3372f]/70',
   359→              )}
   360→            >
   361→              <span>{faNum(d)}</span>
   362→              {hijriTiny && (
   363→                <span className="absolute left-1 top-0.5 text-[6.5px] font-bold leading-none text-muted-foreground/70">{hijriTiny}</span>
   364→              )}
   365→              {cheques.length > 0 && <span className="text-[8px] font-black leading-none text-[#8a5a2b]">{faMoneyShort(dayTotal)}</span>}
   366→              {cheques.length === 0 && hol && <span className="absolute bottom-0.5 h-1 w-1 rounded-full bg-[#b3372f]" />}
