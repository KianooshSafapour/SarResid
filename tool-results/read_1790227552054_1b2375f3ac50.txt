   341→/* ═══════════════════════════ Tab 2: صف صندوق ═══════════════════════════ */
   342→function CashierQueue({ ctx }: { ctx: AppCtx }) {
   343→  const [list, setList] = useState<PreOrder[]>([])
   344→  const [loading, setLoading] = useState(true)
   345→  const [openId, setOpenId] = useState<string | null>(null)
   346→  const [drafts, setDrafts] = useState<Record<string, PItem[]>>({})
   347→  const [printing, setPrinting] = useState<PreOrder | null>(null)
   348→  const [busy, setBusy] = useState(false)
   349→
   350→  const load = async () => {
   351→    try {
   352→      const d = await api<{ preOrders: PreOrder[] }>('/api/preorders?scope=cashier')
   353→      setList(d.preOrders)
   354→    } catch (e: any) {
   355→      toast.error(e.message)
   356→    } finally {
   357→      setLoading(false)
   358→    }
   359→  }
   360→  useEffect(() => { load() }, [])
   361→
   362→  const toggle = (po: PreOrder) => {
   363→    if (openId === po.id) return setOpenId(null)
   364→    setOpenId(po.id)
   365→    setDrafts((d) => ({ ...d, [po.id]: po.items.map((i) => ({ ...i })) }))
   366→  }
   367→
   368→  const setQty = (poId: string, idx: number, qty: number) =>
   369→    setDrafts((d) => ({
   370→      ...d,
   371→      [poId]: (d[poId] || []).map((it, i) => (i === idx ? { ...it, qty: Math.max(1, qty) } : it)),
   372→    }))
   373→
   374→  const sendToPos = async (po: PreOrder) => {
   375→    setBusy(true)
   376→    try {
   377→      const items = drafts[po.id] || po.items
   378→      const res = await api<{ preOrder: PreOrder }>(`/api/preorders/${po.id}`, {
   379→        method: 'PATCH',
   380→        body: { action: 'cashier_edit', items },
   381→      })
   382→      toast.success('آماده پردازش در هلو — رسید چاپ شد 🖨️')
   383→      setOpenId(null)
   384→      setPrinting({ ...res.preOrder, items })
   385→      await load()
   386→      ctx.refreshNotifications()
   387→    } catch (e: any) {
   388→      toast.error(e.message)
   389→    } finally {
   390→      setBusy(false)
   391→    }
   392→  }
   393→
   394→  const cancel = async (po: PreOrder) => {
   395→    setBusy(true)
   396→    try {
   397→      await api(`/api/preorders/${po.id}`, { method: 'PATCH', body: { action: 'cancel' } })
   398→      toast.success(`پیش‌فاکتور ${po.code} لغو شد`)
   399→      setOpenId(null)
   400→      await load()
   401→    } catch (e: any) {
   402→      toast.error(e.message)
   403→    } finally {
   404→      setBusy(false)
   405→    }
   406→  }
   407→
   408→  return (
   409→    <>
   410→      <SectionCard
   411→        title="صف صندوق 💳"
   412→        subtitle="پیش‌فاکتورهای ثبت‌شدهٔ فروشندگان — ویرایش تعداد، ارسال به هلو و چاپ رسید"
   413→        icon={<CreditCard size={18} />}
   414→      >
   415→        {loading ? (
   416→          <p className="py-8 text-center text-sm text-muted-foreground">در حال بارگذاری صف…</p>
   417→        ) : list.length === 0 ? (

   633→/* ═══════════════════════════ Tab 4: درخواست‌های مشتری ═══════════════════════════ */
   634→function RequestsSection({ canDelete }: { canDelete: boolean }) {
   635→  const [reqs, setReqs] = useState<CReq[]>([])
   636→  const [loading, setLoading] = useState(true)
   637→  const [name, setName] = useState('')
   638→  const [busy, setBusy] = useState(false)
   639→
   640→  const load = async () => {
   641→    try {
   642→      const d = await api<{ requests: CReq[] }>('/api/customer-requests')
   643→      setReqs(d.requests)
   644→    } catch (e: any) {
   645→      toast.error(e.message)
   646→    } finally {
   647→      setLoading(false)
   648→    }
   649→  }
   650→  useEffect(() => { load() }, [])
   651→
   652→  const add = async () => {
   653→    if (!name.trim()) return toast.error('نام کالا را وارد کنید')
   654→    setBusy(true)
   655→    try {
   656→      const res = await api<{ repeated: boolean }>('/api/customer-requests', { method: 'POST', body: { productName: name.trim() } })
   657→      toast.success(res.repeated ? 'تعداد درخواست این کالا افزایش یافت ✅' : 'درخواست مشتری ثبت شد 📝')
   658→      setName('')
   659→      await load()
   660→    } catch (e: any) {
   661→      toast.error(e.message)
   662→    } finally {
   663→      setBusy(false)
   664→    }
   665→  }
   666→
   667→  const del = async (r: CReq) => {
   668→    try {
   669→      await api(`/api/customer-requests?id=${r.id}`, { method: 'DELETE' })
   670→      toast.success('حذف شد')
   671→      await load()
   672→    } catch (e: any) {
   673→      toast.error(e.message)
   674→    }
   675→  }
   676→
   677→  return (
   678→    <SectionCard
   679→      title="درخواست‌های مشتری برای کالاهای ناموجود"
   680→      subtitle="هر بار مشتری دنبال کالایی بود که نداشتیم، اینجا ثبت کنید تا در تصمیم خرید دیده شود"
   681→      icon={<NotebookPen size={18} />}
   682→    >
   683→      <div className="mb-3 flex gap-2">
   684→        <input
   685→          value={name}
   686→          onChange={(e) => setName(e.target.value)}
   687→          onKeyDown={(e) => e.key === 'Enter' && add()}
   688→          placeholder="نام کالایی که مشتری خواست…"
   689→          className="flex-1 rounded-xl border border-input bg-white/90 p-3 text-sm"
   690→        />
   691→        <button
   692→          onClick={add}
   693→          disabled={busy}
   694→          className="flex items-center gap-1.5 rounded-xl bg-primary px-5 py-3 text-xs font-extrabold text-white disabled:opacity-40"
   695→        >
   696→          <Plus size={14} /> ثبت درخواست
   697→        </button>
   698→      </div>
   699→      {loading ? (
   700→        <p className="py-8 text-center text-sm text-muted-foreground">در حال بارگذاری…</p>
   701→      ) : reqs.length === 0 ? (
   702→        <EmptyState emoji="📝" title="هنوز درخواستی ثبت نشده" hint="اولین درخواست را همین بالا ثبت کنید" />
   703→      ) : (
   704→        <div className="scroll-gold max-h-[42vh] space-y-1.5 overflow-y-auto pl-1">
   705→          {reqs.map((r) => (
   706→            <div key={r.id} className="flex items-center justify-between gap-2 rounded-xl border border-border bg-white/80 p-3">
   707→              <div className="flex min-w-0 items-center gap-2.5">
   708→                <span className="flex h-9 shrink-0 items-center rounded-xl bg-[#c9a227]/15 px-2.5 text-xs font-black text-[#8a6d10]">
   709→                  ×{faNum(r.count)}
   710→                </span>
   711→                <div className="min-w-0">
   712→                  <p className="truncate text-xs font-bold">{r.productName}</p>

   714→                </div>
   715→              </div>
   716→              {canDelete && (
   717→                <button
   718→                  onClick={() => del(r)}
   719→                  className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-[#b3372f]/10 text-[#b3372f]"

   729→    </SectionCard>
