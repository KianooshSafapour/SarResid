   188→
   189→      {step === 1 && (

   193→              <button key={p.id} onClick={() => { setProviderId(p.id); setCompany(''); setStep(2) }}

   329→function ProductHistoryModal({ product, onClose, onOrder }: { product: Product; onClose: () => void; onOrder: (q: number) => void }) {

   333→  }, [product.id])
   334→  return (
   335→    <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
   336→      <div className="absolute inset-0 bg-black/50" onClick={onClose} />
   337→      <div className="scroll-gold glow-card max-h-[80vh] w-full max-w-md overflow-y-auto rounded-2xl bg-card p-5">

   340→          <button onClick={onClose}><X size={18} /></button>
   341→        </div>
   342→        {history ? (
   343→          <>
   344→            <div className="mb-3 grid grid-cols-2 gap-2">
   345→              <KeyValue k="سفارش ماه اخیر" v={`${faNum(history.orderedLastMonth)} عدد`} />

   349→            </div>
   350→            <p className="mb-2 text-xs font-black text-muted-foreground">آخرین سفارش‌ها:</p>
   351→            {history.lastOrders.length === 0 ? (
   352→              <p className="text-xs text-muted-foreground">در ماه اخیر سفارشی ثبت نشده</p>
   353→            ) : (
   354→              <ul className="mb-4 space-y-1">
   355→                {history.lastOrders.map((o: any, i: number) => (
   356→                  <li key={i} className="flex items-center justify-between rounded-lg bg-muted/50 px-3 py-1.5 text-[11px]">
   357→                    <span className="font-bold">{o.code}</span>
   358→                    <span>{faNum(o.qty)} عدد</span>
   359→                    <span className="text-muted-foreground">{formatJalaliShort(o.at)}</span>
   360→                  </li>
   361→                ))}
   362→              </ul>
   363→            )}
   364→          </>
   365→        ) : (
   366→          <div className="py-6 text-center text-xs text-muted-foreground">در حال بارگذاری…</div>
   367→        )}
   368→        <div className="flex gap-2">
   369→          <button onClick={() => onOrder(1)} className="flex-1 rounded-xl bg-primary py-2.5 text-xs font-extrabold text-white">+ افزودن ۱ عدد به سفارش</button>
   370→          <button onClick={() => onOrder(6)} className="rounded-xl bg-secondary px-4 py-2.5 text-xs font-extrabold text-primary">+{faNum(6)}</button>
   371→          <button onClick={() => onOrder(12)} className="rounded-xl bg-secondary px-4 py-2.5 text-xs font-extrabold text-primary">+{faNum(12)}</button>
   372→        </div>
   373→      </div>
   374→    </div>
   375→  )
   376→}

   380→function OrderDetail({ ctx, orderId, onBack }: { ctx: AppCtx; orderId: string; onBack: () => void }) {
   381→  const holidays = useHolidays()
   382→  const [order, setOrder] = useState<any>(null)
   383→  const [items, setItems] = useState<any[]>([])
   384→  const [editOpen, setEditOpen] = useState(false)
   385→  const [corrOpen, setCorrOpen] = useState(false)
   386→  const [corrTitle, setCorrTitle] = useState('')
   387→  const [corrDetail, setCorrDetail] = useState('')
   388→  const [editDate, setEditDate] = useState('')
   389→  const [editNotes, setEditNotes] = useState('')
   390→
   391→  const load = () =>
   392→    api<{ order: any; items: any[] }>(`/api/orders/${orderId}`).then((d) => {
   393→      setOrder(d.order)
   394→      setItems(d.items)
   395→    })
   396→  useEffect(() => { load() }, [orderId])
   397→
   398→  if (!order) return <div className="py-16 text-center text-sm text-muted-foreground">در حال بارگذاری…</div>
   399→
   400→  const st = ORDER_STATUSES[order.status]
   401→  const role = ctx.user!.role
   402→  const act = async (body: any, successMsg: string) => {
   403→    try {
   404→      await api(`/api/orders/${orderId}`, { method: 'PATCH', body })
   405→      toast.success(successMsg)
   406→      await load()
   407→      ctx.refreshNotifications()
   408→    } catch (e: any) { toast.error(e.message) }
   409→  }
   410→
   411→  const canApprove = order.status === 'SUBMITTED' && ['GM', 'OM'].includes(role)
   412→  const canReceive = ['APPROVED', 'RECEIVING'].includes(order.status) && ['GM', 'OM', 'SK'].includes(role) || (['APPROVED', 'RECEIVING'].includes(order.status) && ctx.user!.secondaryRoles.includes('RECEIVER'))
   413→  const canVerify = order.status === 'RECEIVED' && ['SK', 'GM', 'OM'].includes(role)
   414→  const canEdit = !['RECEIVED', 'VERIFIED', 'ACCOUNTED', 'DONE', 'CANCELLED'].includes(order.status) && ['GM', 'PM', 'OM'].includes(role)
   415→  const canCancel = !['DONE', 'CANCELLED'].includes(order.status) && ['GM', 'OM'].includes(role)
   416→
   417→  return (
   418→    <div className="space-y-4">
   419→      <SectionCard
   420→        title={`${order.code} — ${order.providerName}`}
   421→        subtitle={`ثبت‌کننده: ${order.createdByName} • ${formatJalaliFull(order.createdAt)}`}
   422→        icon={<ClipboardList size={18} />}
   423→        actions={<button onClick={onBack} className="rounded-lg border px-3 py-1.5 text-xs font-bold">→ همه سفارش‌ها</button>}
   424→      >
   425→        {/* status pipeline */}
   426→        <div className="mb-5 flex flex-wrap items-center gap-1.5">
   427→          {ORDER_FLOW.map((s, i) => {
   428→            const idx = ORDER_FLOW.indexOf(order.status)
   429→            const done = idx >= 0 && i < idx
   430→            const current = order.status === s
   431→            return (
   432→              <div key={s} className="flex items-center gap-1.5">
   433→                <span className={cn('rounded-full px-2.5 py-1 text-[10px] font-black', current ? 'text-white shadow' : done ? 'bg-[#0e7a4a]/15 text-[#0e7a4a]' : 'bg-muted text-muted-foreground')} style={current ? { background: ORDER_STATUSES[s].color } : {}}>
   434→                  {ORDER_STATUSES[s].label}
   435→                </span>
   436→                {i < ORDER_FLOW.length - 1 && <span className={cn('h-0.5 w-4 rounded', done ? 'bg-[#0e7a4a]' : 'bg-border')} />}
   437→              </div>
   438→            )
   439→          })}
   440→          {order.status === 'CANCELLED' && <Pill label="لغو شده" color="#b3372f" bg="#fee2e2" />}
   441→        </div>
   442→
   443→        <div className="mb-4 grid gap-2 sm:grid-cols-4">
   444→          <KeyValue k="تاریخ تحویل" v={formatJalaliFull(order.deliveryDate)} />
   445→          <KeyValue k="مبلغ کل" v={`${faMoney(order.totalAmount)} تومان`} />
   446→          <KeyValue k="پرداخت" v={order.payMethod === 'CHEQUE' ? 'چک' : 'نقدی'} />
   447→          <KeyValue k="وضعیت" v={<Pill label={st.label} color={st.color} bg={st.bg} />} />
   448→        </div>
   449→
   450→        {order.isOverdue && (
   451→          <div className="mb-4 flex items-center gap-2 rounded-xl border border-[#b3372f]/30 bg-[#fee2e2]/60 p-3 text-xs font-bold text-[#b3372f]">
   452→            <AlertTriangle size={16} /> این سفارش از تاریخ تحویل گذشته و هنوز دریافت نشده — با تأمین‌کننده فالوآپ کنید.
   453→          </div>
   454→        )}
   455→        {order.notes && <p className="mb-4 rounded-xl bg-accent/60 p-3 text-xs">📝 {order.notes}</p>}
   456→
   457→        {/* items */}
   458→        <div className="overflow-x-auto">
   459→          <table className="w-full text-right text-xs">
   460→            <thead>
   461→              <tr className="border-b border-border text-[10px] text-muted-foreground">
   462→                <th className="p-2">کالا</th>
   463→                <th className="p-2">بارکد</th>
   464→                <th className="p-2">سفارش</th>
   465→                <th className="p-2">قیمت واحد</th>
   466→                <th className="p-2">جمع</th>
   467→                <th className="p-2">دریافتی</th>
   468→              </tr>
   469→            </thead>
   470→            <tbody>
   471→              {items.map((it) => (
   472→                <tr key={it.id} className="border-b border-border/40">
   473→                  <td className="p-2 font-bold">{it.productName}</td>
   474→                  <td className="p-2 font-mono text-[10px] text-muted-foreground">{it.barcode ? faNum(it.barcode) : '—'}</td>
   475→                  <td className="p-2">{faNum(it.qty)}</td>

   478→                  <td className="p-2">

   480→                      <Pill label={`${faNum(it.receivedQty)} / ${faNum(it.qty)}`} color={it.receivedQty < it.qty ? '#a16207' : '#166534'} />
   481→                    ) : it.status === 'REJECTED' ? <Pill label="رد شده" color="#b3372f" /> : it.status === 'SHORT' ? <Pill label="کسری" color="#a16207" /> : '—'}
   482→                  </td>
   483→                </tr>
   484→              ))}
   485→            </tbody>
   486→          </table>
   487→        </div>
   488→
   489→        {/* actions */}
   490→        <div className="mt-5 flex flex-wrap gap-2">

   493→              <CheckCircle2 size={15} /> تأیید و ارسال به تأمین‌کننده
   494→            </button>
   495→          )}
   496→          {canReceive && (
   497→            <button onClick={() => ctx.navigate('receiving', order.id)} className="flex items-center gap-1.5 rounded-xl bg-gradient-to-l from-[#c96f4a] to-[#d98a5f] px-4 py-2.5 text-xs font-extrabold text-white">
   498→              🚚 دریافت مرسوله
   499→            </button>
   500→          )}
   501→          {canVerify && (
   502→            <button onClick={() => ctx.navigate('verify', order.id)} className="flex items-center gap-1.5 rounded-xl bg-gradient-to-l from-[#77934a] to-[#8fa85e] px-4 py-2.5 text-xs font-extrabold text-white">

   512→            <button onClick={() => setCorrOpen(true)} className="flex items-center gap-1.5 rounded-xl border px-4 py-2.5 text-xs font-extrabold text-muted-foreground">
   513→              <History size={14} /> ثبت اصلاحیه
   514→            </button>
   515→          )}

   518→              <Ban size={14} /> لغو سفارش
   519→            </button>
   520→          )}
   521→        </div>
   522→      </SectionCard>
   523→

   526→        <ol className="relative space-y-3 border-r-2 border-dashed border-[#c9a227]/40 pr-5">

   528→            <li key={i} className="relative">

   556→        <Modal title="ثبت اصلاحیه (بدون تغییر تاریخچه)" onClose={() => setCorrOpen(false)}>
   557→          <Labeled label="عنوان اصلاحیه"><input value={corrTitle} onChange={(e) => setCorrTitle(e.target.value)} className="w-full rounded-xl border border-input p-3 text-sm" placeholder="مثلاً: اصلاح تعداد شیر پگاه" /></Labeled>
   558→          <Labeled label="توضیح"><textarea value={corrDetail} onChange={(e) => setCorrDetail(e.target.value)} rows={3} className="w-full rounded-xl border border-input p-3 text-sm" /></Labeled>

   560→            onClick={async () => {

   564→            }}
   565→            className="mt-2 w-full rounded-xl bg-primary py-2.5 text-xs font-extrabold text-white"
   566→          >
   567→            ثبت اصلاحیه
   568→          </button>
   569→        </Modal>
   570→      )}
   571→    </div>
   572→  )
   573→}
