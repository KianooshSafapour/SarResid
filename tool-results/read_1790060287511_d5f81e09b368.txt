   115→/** ورودی عدد کوچک (دقیقه/شمارش) — ارقام فارسی/عربی/لاتین می‌پذیرد */
   116→function SmallNum({ value, onChange, disabled, max = 720, w = 'w-14' }: { value: number; onChange: (v: number) => void; disabled?: boolean; max?: number; w?: string }) {
   117→  return (
   118→    <input
   119→      type="text" inputMode="numeric" dir="ltr" disabled={disabled}
   120→      value={value === 0 ? '' : faNum(value)}
   121→      placeholder="۰"
   122→      onChange={(e) => {
   123→        const cleaned = enDigits(e.target.value).replace(/[^\d]/g, '')
   124→        const n = cleaned === '' ? 0 : Math.min(max, Number(cleaned))
   125→        onChange(isNaN(n) ? 0 : n)
   126→      }}
   127→      className={cn(w, 'rounded-lg border border-input bg-white/90 px-1 py-1 text-center text-xs font-bold tabular-nums outline-none focus:border-primary disabled:opacity-50')}
   128→    />
   129→  )
   130→}

   155→  const canManage = !!data?.viewer.isManager
   156→  const canCommission = !!data?.viewer.canCommission
   157→
   158→  const load = useCallback(async (pjy = jy, pjm = jm) => {
   159→    try {
   160→      const d = await api<Resp>(`/api/payroll?jy=${pjy}&jm=${pjm}`)
   161→      setData(d)
   162→    } catch (e) {
   163→      toast.error((e as Error).message)
   164→    } finally {
   165→      setLoading(false)
   166→    }
   167→  }, [jy, jm])
   168→
   169→  useEffect(() => { setLoading(true); load() }, [load])

   176→  const totals = useMemo(() => {
   177→    const acc = { gross: 0, net: 0, insurance: 0, tax: 0, ot: 0, confirmed: 0, paid: 0 }

   179→      acc.gross += r.computed.gross; acc.net += r.computed.net
   180→      acc.insurance += r.computed.insurance; acc.tax += r.computed.tax
   181→      acc.ot += r.computed.otHours + r.computed.holidayOtHours
   182→      if (r.month.status === 'CONFIRMED') acc.confirmed += 1
   183→      if (r.month.status === 'PAID') acc.paid += 1
   184→    }

   191→    if (njm > 12) { njm = 1; njy += 1 }

   199→    { key: 'self', label: 'فیش من', icon: <Eye size={14} /> },

   211→          .no-print { display: none !important; }
   212→        }
   213→      `}</style>
   214→
   215→      {/* سرصفحهٔ ماه */}
   216→      <SectionCard
   217→        title="حقوق و دستمزد"
   218→        subtitle={`ماه ${J_MONTHS[(data?.jm || jm) - 1]} ${faNum(data?.jy || jy)} — مبالغ جدول به تومان · مطابق قانون کار سال حقوقی ${faNum(data?.salaryYear || 1405)} (مبالغ موتور: ریال)`}
   219→        icon={<Landmark size={20} />}
   220→        actions={
   221→          <>
   222→            <button onClick={() => switchMonth(-1)} className="flex items-center gap-1 rounded-xl border border-border bg-card px-3 py-2 text-sm font-bold hover:border-primary"><ChevronRight size={16} /> ماه قبل</button>
   223→            <button onClick={() => { setJy(now.jy); setJm(now.jm) }} className="rounded-xl border border-border bg-card px-3 py-2 text-sm font-bold hover:border-primary" title="ماه جاری"><CalendarDays size={16} /></button>

   229→        <div className="mb-4 flex flex-wrap gap-2">

   231→            <button key={t.key} onClick={() => setTab(t.key)}

   233→                tab === t.key ? 'bg-primary text-white shadow' : 'border border-border bg-card text-muted-foreground hover:border-primary')}>
   234→              {t.icon}{t.label}
   235→            </button>
   236→          ))}
   237→        </div>
   238→
   239→        {loading ? (
   240→          <div className="space-y-2">{[0, 1, 2].map((i) => <div key={i} className="h-12 animate-pulse rounded-xl bg-muted/60" />)}</div>
   241→        ) : tab === 'month' && (
   242→          <MonthTab data={data} rows={rows} totals={totals} onOpenEditor={setEditorId} onOpenSlip={setSlipId} onOpenYearly={(pid) => { setYearlySel({ personnelId: pid, jy: now.jy }); setYearly(null); setYearlyMeta(null); setTab('yearly') }} />
   243→        )}
   244→        {tab === 'yearly' && (
   245→          <YearlyTab
   246→            rows={rows} canManage={canManage} selfPid={data?.self?.personnelId || null}
   247→            sel={yearlySel} setSel={setYearlySel} yearly={yearly} setYearly={setYearly} meta={yearlyMeta} setMeta={setYearlyMeta}

   250→        {tab === 'self' && <SelfTab row={selfRow} onOpenSlip={setSlipId} />}
   251→        {/* key = سال حقوقی: با تغییر سال، پیش‌نویس تنظیمات از نو بر پایهٔ جدول قانونی تازه ساخته می‌شود */}

   255→      {/* ویرایشگر ماه پرسنل */}

   284→              const type = confirmBox.type; const note = confirmBox.note

   287→                await api('/api/payroll', { method: 'POST', body: { action: type, personnelId: editorRow.id, jy: data?.jy || jy, jm: data?.jm || jm, note } })

   290→              } catch (e) { toast.error((e as Error).message) }
   291→            }} className="flex-1 rounded-xl bg-primary py-2.5 text-sm font-black text-white hover:opacity-90">
   292→              بله، انجام بده
   293→            </button>
   294→            <button onClick={() => setConfirmBox(null)} className="rounded-xl border border-border px-4 py-2.5 text-sm font-bold text-muted-foreground">انصراف</button>
   295→          </div>
   296→        </Modal>
   297→      )}
   298→
   299→      {/* فیش حقوقی (چاپ) */}
   300→      {slipRow && data && (
   301→        <Modal title={`فیش حقوقی — ${slipRow.name}`} wide onClose={() => setSlipId(null)}>
   302→          <Payslip row={slipRow} data={data} generatedBy={ctx.user?.name || 'سامانه'} />
   303→        </Modal>
   304→      )}
   305→    </div>
   306→  )
   307→}

   311→function MonthTab({ rows, totals, onOpenEditor, onOpenSlip, onOpenYearly, data }: {

   322→        <StatCard label="جمع خالص پرداختی ماه (تومان)" value={T(totals.net)} tone="emerald" icon={<Banknote size={18} />} hint={`${faNum(rows.length)} پرسنل`} />

   324→        <StatCard label="بیمهٔ سهم کارگر (تومان)" value={T(totals.insurance)} tone="olive" icon={<PiggyBank size={18} />} hint="۷٪ مشمول بیمه" />
   325→        <StatCard label="مالیات حقوق (تومان)" value={T(totals.tax)} tone="rose" icon={<Percent size={18} />} hint={`تأییدشده: ${faNum(totals.confirmed)} · پرداخت‌شده: ${faNum(totals.paid)}`} />
   326→      </div>
   327→
   328→      <div className="overflow-x-auto rounded-2xl border border-border">
   329→        <table className="w-full min-w-[820px] text-right text-sm">
   330→          <thead>
   331→            <tr className="border-b border-border bg-muted/50 text-xs text-muted-foreground">
   332→              <th className="px-3 py-2.5 font-bold">پرسنل</th>
   333→              <th className="px-3 py-2.5 font-bold">کارکرد</th>

   877→function YearlyTab({ rows, canManage, selfPid, sel, setSel, yearly, setYearly, meta, setMeta }: {

   879→  sel: { personnelId: string; jy: number } | null

   895→    } catch (e) { toast.error((e as Error).message) } finally { setBusy(false) }
   896→  }
   897→
   898→  if (!rows.length) return <EmptyState emoji="🫒" title="پرسنلی یافت نشد" />
   899→
   900→  return (
   901→    <div className="space-y-4">

   917→          className="rounded-xl bg-primary px-4 py-2.5 text-sm font-black text-white disabled:opacity-40">

  1005→function TomanInput({ rial, onRial, className, disabled, placeholder, ariaLabel }: {
  1006→  rial: number; onRial: (rial: number) => void; className?: string; disabled?: boolean; placeholder?: string; ariaLabel?: string
  1007→}) {
  1008→  const toTxt = (r: number) => {
  1009→    if (!r) return ''
  1010→    const t = r / 10

  1012→  }

  1014→  const [focused, setFocused] = useState(false)
  1015→  // resync when the prop changes from outside (e.g. save/reload) — official render-phase adjustment pattern
  1016→  const [prev, setPrev] = useState(rial)
  1017→  if (prev !== rial) {
  1018→    setPrev(rial)
  1019→    if (!focused) setText(toTxt(rial))
  1020→  }
  1021→  return (
  1022→    <input
  1023→      dir="ltr" inputMode="decimal" disabled={disabled} aria-label={ariaLabel}
  1024→      value={text} placeholder={placeholder ?? '۰'}
  1025→      onFocus={() => setFocused(true)}
  1026→      onBlur={() => { setFocused(false); setText(toTxt(rial)) }}
  1027→      onChange={(e) => {
  1028→        const cleaned = enDigits(e.target.value).replace(/[^\d.]/g, '').slice(0, 18)
  1029→        setText(faNum(cleaned))

  1032→      }}

  1034→    />
  1035→  )
