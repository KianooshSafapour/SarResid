    36→function loadTax(rows: SettingRow[], year = 1405): TaxConfig {
    37→  const statutory = getStatutory(year).tax as unknown as TaxConfig
    38→  const raw = rows.find((r) => r.key === S_TAX)
    39→  const t = raw ? safeParse<TaxConfig>(raw.value, statutory) : statutory
    40→  const exempt = Number(t?.exempt)
    41→  const bands = Array.isArray(t?.bands) ? t.bands : []
    42→  if (!isFinite(exempt) || exempt < 0 || !bands.length) return statutory
    43→  const clean = bands
    44→    .map((b) => ({ upTo: b.upTo == null ? null : Number(b.upTo), rate: Number(b.rate) }))
    45→    .filter((b) => isFinite(b.rate) && b.rate > 0 && b.rate <= 1 && (b.upTo == null || b.upTo >= 0))
    46→  return clean.length ? { exempt, bands: clean } : statutory
    47→}

    87→/** عدد معتبر نامنفی؛ NaN/منفی/بی‌نهایت/boolean → null */
    88→function reqNum(v: unknown, opts: { int?: boolean; max?: number } = {}): number | null {
    89→  if (typeof v === 'boolean' || v == null || v === '') return null
    90→  const n = Number(v)
    91→  if (!isFinite(n) || isNaN(n) || n < 0) return null
    92→  if (opts.int && Math.round(n) !== n) return null
    93→  return opts.max != null && n > opts.max ? null : n
    94→}

   152→async function computeDayKinds(personnelId: string, monthStartIso: string, monthEndIso: string, daysIso: string[]): Promise<Map<string, string>> {
   153→  const map = new Map<string, string>()
   154→  const [holidays, users, leaves] = await Promise.all([
   155→    db.holiday.findMany({ where: { date: { gte: monthStartIso, lte: monthEndIso }, kind: 'HOLIDAY' }, select: { date: true } }),
   156→    db.user.findMany({ where: { personnelId }, select: { id: true } }),
   157→    db.leaveRequest.findMany({
   158→      where: { status: 'APPROVED', fromDate: { lte: monthEndIso }, toDate: { gte: monthStartIso } },
   159→      select: { userId: true, fromDate: true, toDate: true },
   160→    }),
   161→  ])
   162→  const holidaySet = new Set(holidays.map((h) => h.date.slice(0, 10)))
   163→  const userIds = new Set(users.map((u) => u.id))
   164→  const leaveDates = new Set<string>()
   165→  for (const l of leaves) {
   166→    if (!userIds.has(l.userId)) continue
   167→    for (const d of daysIso) {
   168→      if (d >= l.fromDate.slice(0, 10) && d <= l.toDate.slice(0, 10)) leaveDates.add(d)
   169→    }
   170→  }
   171→  for (const d of daysIso) {
   172→    const wd = new Date(d + 'T12:00:00').getDay()
   173→    map.set(d, holidaySet.has(d) ? 'HOLIDAY' : wd === 5 ? 'FRIDAY' : leaveDates.has(d) ? 'LEAVE' : 'WORK')
   174→  }
   175→  return map
   176→}

   197→/** محاسبهٔ مجدد جمع‌های ماه و ذخیره در SalaryMonth (پس از هر نوشتن) */
   198→async function recomputeMonth(personnelId: string, personnelName: string, jy: number, jm: number, profile: SalaryProfileLike, policy: PayrollPolicy, tax: TaxConfig, personnelActive: boolean) {
   199→  const month = await ensureMonthRow(personnelId, personnelName, jy, jm)
   200→  const [days, adjusts] = await Promise.all([
   201→    db.salaryDay.findMany({ where: { monthId: month.id }, orderBy: { forDate: 'asc' } }),
   202→    db.salaryAdjust.findMany({ where: { monthId: month.id }, orderBy: { createdAt: 'asc' } }),
   203→  ])
   204→  const engineDays = days.map((d) => ({ forDate: d.forDate, otMinutes: d.otMinutes, holidayOtMinutes: d.holidayOtMinutes, nightMinutes: d.nightMinutes, absence: d.absence as 'NONE', dayKind: d.dayKind as 'WORK', note: d.note }))
   205→  const engineAdjusts = adjusts.map((a) => ({ kind: a.kind as AdjustKind, amount: a.amount, ownerOnly: a.ownerOnly }))
   206→  const computed = computeMonth({ profile, days: engineDays, adjusts: engineAdjusts, monthDays: jMonthDays(jy, jm), tax, policy, jy, jm, statutoryYear: jy, personnelActive })
   207→  const updated = await db.salaryMonth.update({
   208→    where: { id: month.id },
   209→    data: {
   210→      workDays: computed.workDays, overtimeHours: computed.otHours, holidayOtHours: computed.holidayOtHours,
   211→      nightHours: computed.nightHours, gross: computed.gross, insurance: computed.insurance,
   212→      tax: computed.tax, otherDeductions: computed.otherDeductions, net: computed.net,
   213→      seniorityAccrual: computed.seniorityAccrual,
   214→    },
   215→  })
   216→  return { month: updated, days, adjusts, computed }
   217→}
   218→
   219→function engineDaysOf(days: Array<{ forDate: string; otMinutes: number; holidayOtMinutes: number; nightMinutes: number; absence: string; dayKind: string; note: string }>) {
   220→  return days.map((d) => ({ forDate: d.forDate, otMinutes: d.otMinutes, holidayOtMinutes: d.holidayOtMinutes, nightMinutes: d.nightMinutes, absence: d.absence as 'NONE', dayKind: d.dayKind as 'WORK', note: d.note }))
   221→}
   222→function engineAdjustsOf(list: Array<{ kind: string; amount: number; ownerOnly?: boolean }>) {
   223→  return list.map((a) => ({ kind: a.kind as AdjustKind, amount: a.amount, ownerOnly: a.ownerOnly }))
   224→}

   228→export async function GET(req: Request) {
   229→  const me = await getSessionUser(req)
   230→  if (!me) return fail('ابتدا وارد شوید', 401)
   231→
   232→  const url = new URL(req.url)
   233→  const cur = monthKeyOf(todayIso())
   234→  const jy = Math.round(Number(url.searchParams.get('jy')) || cur.jy)
   235→  const jm = Math.round(Number(url.searchParams.get('jm')) || cur.jm)
   236→  if (!(jm >= 1 && jm <= 12) || !(jy >= 1380 && jy <= 1460)) return fail('ماه شمسی نامعتبر است')
   237→
   238→  const [isManager, isViewer, canSettings] = await Promise.all([
   239→    hasCap(me, 'payroll.manage'),
   240→    canViewApi(me, 'payroll'),
   241→    hasCap(me, 'settings.manage'),
   242→  ])
   243→  const isOwner = me.role === 'OWNER' || me.role === 'ADMIN' // دیدن/تعیین پورسانت — فقط مالک و روت
   244→  const meRow = await db.user.findUnique({ where: { id: me.id }, select: { personnelId: true } })
   245→  const selfPersonnelId = meRow?.personnelId || ''
   246→
   247→  const rows = await db.setting.findMany({ where: { key: { in: [S_DEFAULTS, S_POLICY, S_TAX, S_YEAR, S_POSITION_BASE] } } })
   248→  const defaults = loadDefaults(rows)
   249→  const policy = loadPolicy(rows)
   250→  const py = payrollYear(rows)
   251→  const tax = loadTax(rows, py)
   252→  const positionBase = loadPositionBase(rows)
   253→
   254→  if (!isViewer && !selfPersonnelId) {
   255→    return json({
   256→      jy, jm, isManager: false, isOwner: false, self: null, personnel: [],
   257→      defaults: { ...defaultProfileFor(py), ...defaults }, policy, tax, statutory: getStatutory(py), salaryYear: py, positionBase,
   258→      viewer: { isManager: false, isOwner: false, canCommission: false, canSettings },
   259→      hint: 'به پروندهٔ پرسنلی متصل نیستید — برای مشاهدهٔ فیش خود با منابع انسانی تماس بگیرید',
   260→    })
   261→  }
   262→
   263→  const { md, start, end } = await monthRange(jy, jm)
   264→  const daysIso: string[] = []
   265→  for (let i = 0; i < md; i++) daysIso.push(jalaliToIso(jy, jm, i + 1))
   266→
   267→  const all = await db.personnel.findMany({ orderBy: [{ active: 'desc' }, { lastName: 'asc' }] })
   268→  const scope = isViewer ? all : all.filter((p) => p.id === selfPersonnelId)
   269→
   270→  const out: unknown[] = []
   271→  for (const p of scope) {
   272→    const profileRow = await ensureProfile(p, defaults, positionBase)
   273→    const profile = toEngineProfile(profileRow)
   274→    const name = `${p.firstName} ${p.lastName}`.trim()
   275→    const { month, days, adjusts, computed } = await recomputeMonth(p.id, name, jy, jm, profile, policy, tax, p.active)
   276→
   277→    const visibleAdjusts = isOwner ? adjusts : adjusts.filter((a) => !a.ownerOnly)
   278→    const hiddenCount = adjusts.length - visibleAdjusts.length
   279→
   280→    // غیرمالک: محاسبهٔ نمایشی بدون اقلام پورسانت تا مبلغ محرمانه نشت نکند
   281→    const computedShown = hiddenCount === 0 ? computed : computeMonth({
   282→      profile, days: engineDaysOf(days), adjusts: engineAdjustsOf(visibleAdjusts),
   283→      monthDays: md, tax, policy, jy, jm, statutoryYear: jy, personnelActive: p.active,
   284→    })
   285→
   286→    out.push({
   287→      id: p.id,
   288→      name,
   289→      personnelNo: personnelNo(p.id),
   290→      jobTitle: p.jobTitle,
   291→      nationalId: p.nationalId,
   292→      bankCard: p.bankCard,
   293→      shaba: p.shaba,
   294→      hireDate: p.hireDate, // شمسی
   295→      active: p.active,
   296→      childrenCount: safeParse<unknown[]>(p.children, []).length,
   297→      profile,
   298→      month: {
   299→        id: month.id, status: month.status, payslipNo: month.payslipNo,
   300→        confirmedByName: month.confirmedByName, confirmedAt: month.confirmedAt, paidAt: month.paidAt,
   301→        note: month.note,
   302→        gross: hiddenCount === 0 ? month.gross : computedShown.gross,
   303→        net: hiddenCount === 0 ? month.net : computedShown.net,
   304→      },
   305→      days,
   306→      adjusts: visibleAdjusts,
   307→      hiddenCommissionCount: hiddenCount,
   308→      computed: computedShown,
   309→    })
   310→  }
   311→
   312→  return json({
   313→    jy, jm, monthDays: md, employer: EMPLOYER,
   314→    isManager, isOwner,
   315→    self: selfPersonnelId ? { personnelId: selfPersonnelId } : null,
   316→    defaults: { ...defaultProfileFor(py), ...defaults },
   317→    policy, tax, statutory: getStatutory(py), salaryYear: py, positionBase,
   318→    viewer: { isManager, isOwner, canCommission: isOwner, canSettings },
   319→    personnel: out,
   320→  })
   321→}

   325→export async function POST(req: Request) {
   326→  const me = await getSessionUser(req)
   327→  if (!me) return fail('ابتدا وارد شوید', 401)
   328→  const body = (await req.json().catch(() => ({}))) as Record<string, unknown>
   329→  const action = String(body.action || '')
   330→  const isOwner = me.role === 'OWNER' || me.role === 'ADMIN'
   331→
   332→  /* ── set-tax — جدول مالیات (settings.manage یا نقش اجرایی) ── */
   333→  if (action === 'set-tax') {
   334→    const g = await guardCap(me, 'settings.manage')
   335→    if (g) return fail(g, 403)
   336→    const exempt = reqNum(body.exempt, { max: 1e15 })
   337→    if (exempt == null) return fail('مبلغ معافیت مالیات نامعتبر است (عدد نامنفی ریال)')
   338→    const bandsRaw = Array.isArray(body.bands) ? body.bands : []
   339→    if (!bandsRaw.length) return fail('حداقل یک پلهٔ نرخ مالیات لازم است')
   340→    const bands: TaxConfig['bands'] = []
   341→    for (const b of bandsRaw as Array<{ upTo?: unknown; rate?: unknown }>) {
   342→      const rate = Number(b.rate)
   343→      if (!isFinite(rate) || rate <= 0 || rate > 1) return fail('نرخ هر پله باید بین ۰ و ۱۰۰ درصد باشد')
   344→      const upTo = b.upTo == null || b.upTo === '' ? null : reqNum(b.upTo, { max: 1e15 })
   345→      if (b.upTo != null && b.upTo !== '' && upTo == null) return fail('سقف پلهٔ مالیاتی نامعتبر است')
   346→      bands.push({ upTo, rate })
   347→    }
   348→    const payload = { exempt, bands }
   349→    await db.setting.upsert({
   350→      where: { key: S_TAX },
   351→      update: { value: JSON.stringify(payload) },
   352→      create: { key: S_TAX, value: JSON.stringify(payload) },
   353→    })
   354→    await logActivity(me, 'به‌روزرسانی جدول مالیات حقوق', 'payroll', S_TAX, JSON.stringify(payload))
   355→    return json({ ok: true, tax: payload })
   356→  }
   357→
   358→  /* ── set-defaults — پیش‌فرض‌های پروفایل + سیاست (settings.manage یا نقش اجرایی) ── */
   359→  if (action === 'set-defaults') {
   360→    const g = await guardCap(me, 'settings.manage')
   361→    if (g) return fail(g, 403)
   362→    const rows = await db.setting.findMany({ where: { key: { in: [S_DEFAULTS, S_POLICY] } } })
   363→    const nextDefaults: Record<string, number | boolean> = { ...loadDefaults(rows) }
   364→    for (const f of ['baseSalary', 'housing', 'foodAllowance', 'maritalAllowance', 'childAllowance', 'seniorityBase'] as const) {
   365→      if (body[f] !== undefined) {
   366→        const n = reqNum(body[f], { max: 1e15 })
   367→        if (n == null) return fail('مقادیر عددی باید نامنفی و معتبر باشند')
   368→        nextDefaults[f] = n
   369→      }
   370→    }
   371→    if (body.overtimeDivisor !== undefined) {
   372→      const n = Number(body.overtimeDivisor)
   373→      if (!isFinite(n) || n < 1 || n > 400) return fail('مخرج مزد ساعتی باید عددی بین ۱ تا ۴۰۰ باشد')

   675→  }

   677→  /* ── pay ── */
   678→  if (action === 'pay') {
   679→    if (month.status === 'PAID') return json({ ok: true, already: true })
   680→    if (month.status !== 'CONFIRMED') return fail('ابتدا ماه را تأیید کنید، سپس پرداخت را ثبت نمایید')
   681→    const payslipNo = month.payslipNo || `HZ-${jy}${String(jm).padStart(2, '0')}-${p.id.slice(-6).toUpperCase()}`
   682→    await db.salaryMonth.update({ where: { id: month.id }, data: { status: 'PAID', paidAt: new Date().toISOString(), payslipNo } })
   683→    await logActivity(me, 'ثبت پرداخت حقوق', 'SalaryMonth', month.id, `${name} — ${jy}/${jm} — خالص ${month.net.toLocaleString('en-US')} ریال — فیش ${payslipNo}`)
   684→    return json({ ok: true, payslipNo })
   685→  }
   686→
   687→  /* ── reopen ── */
   688→  if (action === 'reopen') {
   689→    if (month.status === 'DRAFT') return json({ ok: true, already: true })
   690→    const note = String(body.note || '').trim()
   691→    if (month.status === 'PAID' && month.paidAt && !note) {
   692→      return fail('بازگشایی ماه پرداخت‌شده مستلزم ذکر دلیل است — یادداشت الزامی است')
   693→    }
   694→    await db.salaryMonth.update({
   695→      where: { id: month.id },
   696→      data: { status: 'DRAFT', confirmedById: '', confirmedByName: '', confirmedAt: '', paidAt: '' },
   697→    })
   698→    await logActivity(me, 'بازگشایی ماه حقوق (حسابرسی)', 'SalaryMonth', month.id, `${name} — ${jy}/${jm} — وضعیت قبلی: ${month.status}${note ? ` — دلیل: ${note}` : ''}`)
   699→    return json({ ok: true })
   700→  }
   701→
   702→  return fail('اکشن نامعتبر است')
   703→}
