    75→function parseZones(z: unknown): string[] {
    76→  if (Array.isArray(z)) return z.filter((x): x is string => typeof x === 'string')
    77→  if (typeof z === 'string') {
    78→    try {
    79→      const v = JSON.parse(z || '[]')
    80→      return Array.isArray(v) ? v.filter((x): x is string => typeof x === 'string') : []
    81→    } catch {
    82→      return []
    83→    }
    84→  }
    85→  return []
    86→}

   508→function ZoneAssign({ users, zones, onSaved }: { users: UserRow[]; zones: string[]; onSaved: () => void }) {
   509→  const [busy, setBusy] = useState<string | null>(null)
   510→  const toggle = async (u: UserRow, z: string) => {
   511→    const current = parseZones(u.zones)
   512→    const next = current.includes(z) ? current.filter((x) => x !== z) : [...current, z]
   513→    setBusy(u.id)
   514→    try {
   515→      await api(`/api/users/${u.id}`, { method: 'PATCH', body: { zones: next } })
   516→      toast.success(`زون‌های ${u.name} به‌روزرسانی شد`)

   518→    } catch (e: unknown) {
   519→      toast.error((e as Error)?.message || 'خطا در ذخیرهٔ زون‌ها')
   520→    } finally {
   521→      setBusy(null)
   522→    }
   523→  }
   524→  if (!users.length)
   525→    return <EmptyState emoji="👥" title="کاربر فعالی پیدا نشد" hint="از بخش مدیریت کاربران، همکاران را فعال کنید." />
