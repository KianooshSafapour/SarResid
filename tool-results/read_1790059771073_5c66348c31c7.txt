    15→/** کارت آماری فشرده دوم line — مقدار + برچسب؛ جزئیات در tooltip */
    16→function KpiMini({
    17→  label,
    18→  value,
    19→  hint,
    20→  tone = 'emerald',
    21→  icon,
    22→  onClick,
    23→}: {
    24→  label: string
    25→  value: React.ReactNode
    26→  hint?: string
    27→  tone?: 'emerald' | 'gold' | 'terra' | 'rose' | 'olive'
    28→  icon: React.ReactNode
    29→  onClick?: () => void
    30→}) {
    31→  const tones: Record<string, string> = {
    32→    emerald: 'bg-[#0e7a4a]/10 text-[#0e7a4a]',
    33→    gold: 'bg-[#c9a227]/15 text-[#8a6d10]',
    34→    terra: 'bg-[#c96f4a]/15 text-[#a04c2a]',
    35→    rose: 'bg-[#b3372f]/10 text-[#b3372f]',
    36→    olive: 'bg-[#77934a]/15 text-[#5c7236]',
    37→  }
    38→  return (
    39→    <button
    40→      type="button"
    41→      onClick={onClick}
    42→      title={hint}
    43→      className={cn('glow-card flex min-w-0 items-center gap-2.5 rounded-xl bg-card p-2.5 text-right transition-all sm:p-3', onClick && 'cursor-pointer hover:-translate-y-0.5')}
    44→    >
    45→      <span className={cn('flex h-8 w-8 shrink-0 items-center justify-center rounded-lg', tones[tone])}>{icon}</span>
    46→      <span className="min-w-0">
    47→        <span className="block truncate text-lg font-black leading-6 text-foreground">{value}</span>
    48→        <span className="block truncate text-[11px] font-bold text-muted-foreground">{label}</span>
    49→      </span>
    50→    </button>
    51→  )
    52→}

    66→  desc: string
    67→  tone: string

    75→        <p className="truncate text-xs font-black text-foreground">{title}</p>
