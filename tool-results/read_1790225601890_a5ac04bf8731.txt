    41→/** جدول قانونی سال + اصلاحات دستی مدیر (هر فیلد عددی معتبر نامنفی override می‌شود) */
    42→function statutoryOf(rows: StatutoryRow[], year: number) {
    43→  const base = getStatutory(year)
    44→  const ovr = loadStatutoryOvr(rows)[String(year)] || {}
    45→  const merged: Record<string, unknown> = { ...base }
    46→  for (const [k, v] of Object.entries(ovr)) {
    47→    if (!(STATUTORY_FIELDS as readonly string[]).includes(k)) continue
    48→    const n = Number(v)
    49→    if (isFinite(n) && n >= 0) merged[k] = n
    50→  }
    51→  return merged as unknown as ReturnType<typeof getStatutory>
    52→}

    65→function loadTax(rows: SettingRow[], year = 1405): TaxConfig {
    66→  const statutory = getStatutory(year).tax as unknown as TaxConfig
    67→  const raw = rows.find((r) => r.key === S_TAX)
    68→  const t = raw ? safeParse<TaxConfig>(raw.value, statutory) : statutory
    69→  const exempt = Number(t?.exempt)
    70→  const bands = Array.isArray(t?.bands) ? t.bands : []
    71→  if (!isFinite(exempt) || exempt < 0 || !bands.length) return statutory
    72→  const clean = bands
    73→    .map((b) => ({ upTo: b.upTo == null ? null : Number(b.upTo), rate: Number(b.rate) }))
    74→    .filter((b) => isFinite(b.rate) && b.rate > 0 && b.rate <= 1 && (b.upTo == null || b.upTo >= 0))
    75→  return clean.length ? { exempt, bands: clean } : statutory
    76→}

   116→/** عدد معتبر نامنفی؛ NaN/منفی/بی‌نهایت/boolean → null */
   117→function reqNum(v: unknown, opts: { int?: boolean; max?: number } = {}): number | null {
   118→  if (typeof v === 'boolean' || v == null || v === '') return null
   119→  const n = Number(v)
   120→  if (!isFinite(n) || isNaN(n) || n < 0) return null
   121→  if (opts.int && Math.round(n) !== n) return null
   122→  return opts.max != null && n > opts.max ? null : n
   123→}

   179→/* ── نوع روز: جمعه / تعطیل رسمی / مرخصی تأییدشده / کار ── */

   181→async function computeDayKinds(personnelId: string, monthStartIso: string, monthEndIso: string, daysIso: string[]): Promise<Map<string, string>> {
   182→  const map = new Map<string, string>()
   183→  const [holidays, users, leaves] = await Promise.all([
   184→    db.holiday.findMany({ where: { date: { gte: monthStartIso, lte: monthEndIso }, kind: 'HOLIDAY' }, select: { date: true } }),
   185→    db.user.findMany({ where: { personnelId }, select: { id: true } }),
   186→    db.leaveRequest.findMany({
   187→      where: { status: 'APPROVED', fromDate: { lte: monthEndIso }, toDate: { gte: monthStartIso } },
   188→      select: { userId: true, fromDate: true, toDate: true },
   189→    }),
   190→  ])
   191→  const holidaySet = new Set(holidays.map((h) => h.date.slice(0, 10)))
   192→  const userIds = new Set(users.map((u) => u.id))
   193→  const leaveDates = new Set<string>()
   194→  for (const l of leaves) {
   195→    if (!userIds.has(l.userId)) continue
   196→    for (const d of daysIso) {
   197→      if (d >= l.fromDate.slice(0, 10) && d <= l.toDate.slice(0, 10)) leaveDates.add(d)
   198→    }
   199→  }
   200→  for (const d of daysIso) {
   201→    const wd = new Date(d + 'T12:00:00').getDay()
   202→    map.set(d, holidaySet.has(d) ? 'HOLIDAY' : wd === 5 ? 'FRIDAY' : leaveDates.has(d) ? 'LEAVE' : 'WORK')
   203→  }
   204→  return map
   205→}

   226→/** محاسبهٔ مجدد جمع‌های ماه و ذخیره در SalaryMonth (پس از هر نوشتن) */
   227→async function recomputeMonth(personnelId: string, personnelName: string, jy: number, jm: number, profile: SalaryProfileLike, policy: PayrollPolicy, tax: TaxConfig, personnelActive: boolean, minMonthlyWage?: number) {
   228→  const month = await ensureMonthRow(personnelId, personnelName, jy, jm)
   229→  const [days, adjusts] = await Promise.all([
   230→    db.salaryDay.findMany({ where: { monthId: month.id }, orderBy: { forDate: 'asc' } }),
   231→    db.salaryAdjust.findMany({ where: { monthId: month.id }, orderBy: { createdAt: 'asc' } }),
   232→  ])
   233→  const engineDays = days.map((d) => ({ forDate: d.forDate, otMinutes: d.otMinutes, holidayOtMinutes: d.holidayOtMinutes, nightMinutes: d.nightMinutes, absence: d.absence as 'NONE', dayKind: d.dayKind as 'WORK', note: d.note }))
   234→  const engineAdjusts = adjusts.map((a) => ({ kind: a.kind as AdjustKind, amount: a.amount, ownerOnly: a.ownerOnly }))
   235→  const computed = computeMonth({ profile, days: engineDays, adjusts: engineAdjusts, monthDays: jMonthDays(jy, jm), tax, policy, jy, jm, statutoryYear: jy, personnelActive, minMonthlyWage })
   236→  const updated = await db.salaryMonth.update({
   237→    where: { id: month.id },
   238→    data: {
   239→      workDays: computed.workDays, overtimeHours: computed.otHours, holidayOtHours: computed.holidayOtHours,
   240→      nightHours: computed.nightHours, gross: computed.gross, insurance: computed.insurance,
   241→      tax: computed.tax, otherDeductions: computed.otherDeductions, net: computed.net,
   242→      seniorityAccrual: computed.seniorityAccrual,
   243→    },
   244→  })
   245→  return { month: updated, days, adjusts, computed }
   246→}
   247→
   248→function engineDaysOf(days: Array<{ forDate: string; otMinutes: number; holidayOtMinutes: number; nightMinutes: number; absence: string; dayKind: string; note: string }>) {
   249→  return days.map((d) => ({ forDate: d.forDate, otMinutes: d.otMinutes, holidayOtMinutes: d.holidayOtMinutes, nightMinutes: d.nightMinutes, absence: d.absence as 'NONE', dayKind: d.dayKind as 'WORK', note: d.note }))
   250→}
   251→function engineAdjustsOf(list: Array<{ kind: string; amount: number; ownerOnly?: boolean }>) {
   252→  return list.map((a) => ({ kind: a.kind as AdjustKind, amount: a.amount, ownerOnly: a.ownerOnly }))
   253→}

   257→export async function GET(req: Request) {
   258→  const me = await getSessionUser(req)
   259→  if (!me) return fail('ابتدا وارد شوید', 401)
   260→
   261→  const url = new URL(req.url)
   262→  const cur = monthKeyOf(todayIso())
   263→  const jy = Math.round(Number(url.searchParams.get('jy')) || cur.jy)
   264→  const jm = Math.round(Number(url.searchParams.get('jm')) || cur.jm)
   265→  if (!(jm >= 1 && jm <= 12) || !(jy >= 1380 && jy <= 1460)) return fail('ماه شمسی نامعتبر است')
   266→
   267→  const [isManager, isViewer, canSettings] = await Promise.all([
   268→    hasCap(me, 'payroll.manage'),
   269→    canViewApi(me, 'payroll'),
   270→    hasCap(me, 'settings.manage'),
   271→  ])
   272→  const isOwner = me.role === 'OWNER' || me.role === 'ADMIN' // دیدن/تعیین پورسانت — فقط مالک و روت
   273→  const meRow = await db.user.findUnique({ where: { id: me.id }, select: { personnelId: true } })
   274→  const selfPersonnelId = meRow?.personnelId || ''
   275→
   276→  const rows = await db.setting.findMany({ where: { key: { in: [S_DEFAULTS, S_POLICY, S_TAX, S_YEAR, S_POSITION_BASE, S_STATUTORY] } } })
   277→  const defaults = loadDefaults(rows)
   278→  const policy = loadPolicy(rows)
   279→  const py = payrollYear(rows)
   280→  const tax = loadTax(rows, py)
   281→  const positionBase = loadPositionBase(rows)
   282→
   283→  if (!isViewer && !selfPersonnelId) {
   284→    return json({
   285→      jy, jm, isManager: false, isOwner: false, self: null, personnel: [],
   286→      defaults: { ...defaultProfileFor(py), ...defaults }, policy, tax, statutory: statutoryOf(rows, py), statutoryOvr: loadStatutoryOvr(rows)[String(py)] || {}, salaryYear: py, positionBase,
   287→      viewer: { isManager: false, isOwner: false, canCommission: false, canSettings },
   288→      hint: 'به پروندهٔ پرسنلی متصل نیستید — برای مشاهدهٔ فیش خود با منابع انسانی تماس بگیرید',
   289→    })
   290→  }
   291→
   292→  const { md, start, end } = await monthRange(jy, jm)
   293→  const daysIso: string[] = []
   294→  for (let i = 0; i < md; i++) daysIso.push(jalaliToIso(jy, jm, i + 1))
   295→
   296→  const all = await db.personnel.findMany({ orderBy: [{ active: 'desc' }, { lastName: 'asc' }] })
   297→  const scope = isViewer ? all : all.filter((p) => p.id === selfPersonnelId)
   298→
   299→  const out: unknown[] = []
   300→  for (const p of scope) {
   301→    const profileRow = await ensureProfile(p, defaults, positionBase)
   302→    const profile = toEngineProfile(profileRow)
   303→    const name = `${p.firstName} ${p.lastName}`.trim()
   304→    const { month, days, adjusts, computed } = await recomputeMonth(p.id, name, jy, jm, profile, policy, tax, p.active, statutoryOf(rows, jy).monthlyMinWage)
   305→
   306→    const visibleAdjusts = isOwner ? adjusts : adjusts.filter((a) => !a.ownerOnly)
   307→    const hiddenCount = adjusts.length - visibleAdjusts.length
   308→
   309→    // غیرمالک: محاسبهٔ نمایشی بدون اقلام پورسانت تا مبلغ محرمانه نشت نکند
   310→    const computedShown = hiddenCount === 0 ? computed : computeMonth({
   311→      profile, days: engineDaysOf(days), adjusts: engineAdjustsOf(visibleAdjusts),
   312→      monthDays: md, tax, policy, jy, jm, statutoryYear: jy, personnelActive: p.active,
   313→    })
   314→
   315→    out.push({
   316→      id: p.id,
   317→      name,
   318→      personnelNo: personnelNo(p.id),
   319→      jobTitle: p.jobTitle,
   320→      nationalId: p.nationalId,
   321→      bankCard: p.bankCard,
   322→      shaba: p.shaba,
   323→      hireDate: p.hireDate, // شمسی
   324→      active: p.active,
   325→      childrenCount: safeParse<unknown[]>(p.children, []).length,
   326→      profile,
   327→      month: {
   328→        id: month.id, status: month.status, payslipNo: month.payslipNo,
   329→        confirmedByName: month.confirmedByName, confirmedAt: month.confirmedAt, paidAt: month.paidAt,
   330→        note: month.note,
   331→        gross: hiddenCount === 0 ? month.gross : computedShown.gross,
   332→        net: hiddenCount === 0 ? month.net : computedShown.net,
   333→      },
   334→      days,
   335→      adjusts: visibleAdjusts,
   336→      hiddenCommissionCount: hiddenCount,
   337→      computed: computedShown,
   338→    })
   339→  }
   340→
   341→  return json({
   342→    jy, jm, monthDays: md, employer: EMPLOYER,
   343→    isManager, isOwner,
   344→    self: selfPersonnelId ? { personnelId: selfPersonnelId } : null,
   345→    defaults: { ...defaultProfileFor(py), ...defaults },
   346→    policy, tax, statutory: statutoryOf(rows, py), statutoryOvr: loadStatutoryOvr(rows)[String(py)] || {}, salaryYear: py, positionBase,
   347→    viewer: { isManager, isOwner, canCommission: isOwner, canSettings },
   348→    personnel: out,
   349→  })
   350→}

   354→export async function POST(req: Request) {
   355→  const me = await getSessionUser(req)
   356→  if (!me) return fail('ابتدا وارد شوید', 401)
   357→  const body = (await req.json().catch(() => ({}))) as Record<string, unknown>
   358→  const action = String(body.action || '')
   359→  const isOwner = me.role === 'OWNER' || me.role === 'ADMIN'
   360→
   361→  /* ── set-tax — جدول مالیات (settings.manage یا نقش اجرایی) ── */
   362→  if (action === 'set-tax') {
   363→    const g = await guardCap(me, 'settings.manage')
   364→    if (g) return fail(g, 403)
   365→    const exempt = reqNum(body.exempt, { max: 1e15 })
   366→    if (exempt == null) return fail('مبلغ معافیت مالیات نامعتبر است (عدد نامنفی ریال)')
   367→    const bandsRaw = Array.isArray(body.bands) ? body.bands : []
   368→    if (!bandsRaw.length) return fail('حداقل یک پلهٔ نرخ مالیات لازم است')
   369→    const bands: TaxConfig['bands'] = []

   686→    return json({ ok: true, dayKind, computed: after.computed, month: { id: month.id, status: 'DRAFT' }, warnings })
   687→  }
   688→
   689→  if (!(jm >= 1 && jm <= 12) || !(jy >= 1380 && jy <= 1460)) return fail('ماه شمسی نامعتبر است (jy/jm)')
   690→  const month = await ensureMonthRow(p.id, name, jy, jm)
   691→
   692→  /* ── set-adjust ── */
   693→  if (action === 'set-adjust') {
   694→    const kind = String(body.kind || '') as AdjustKind
   695→    if (!ADJUST_KINDS.includes(kind)) return fail('نوع قلم نامعتبر است (تشویقی/جریمه/پورسانت/مساعده/سایر)')
   696→    const amount = reqNum(body.amount, { max: 1e15 })
   697→    if (amount == null || amount <= 0) return fail('مبلغ باید عددی نامنفی، معتبر و بزرگ‌تر از صفر باشد')
   698→    const note = String(body.note || '').trim()
   699→
   700→    // پورسانت — فقط مالک/روت؛ بقیه ۴۰۳
   701→    if (kind === 'COMMISSION' && !isOwner) {
   702→      return fail('تعیین پورسانت فقط از سوی مالک مجاز است', 403)
   703→    }
   704→
   705→    if (month.status === 'PAID') return fail('ماه پرداخت‌شده قفل است', 409)
   706→    if (month.status === 'CONFIRMED') return fail('ماه تأییدشده قفل است — ابتدا بازگشایی کنید', 409)
   707→
   708→    if (kind === 'FINE') {
   709→      if (note.length < 5) return fail('جریمه مستلزم شرح قانونی و مستند است — یادداشت حداقل ۵ نویسه الزامی است')
   710→      // سقف جریمه: درصدی از جمع حقوق و مزایای همان ماه (پیش‌فرض ۲۵٪) — قبل از افزودن قلم سنجیده می‌شود
   711→      const before = await recomputeMonth(p.id, name, jy, jm, profile, policy, tax, p.active, statMinWageOf(jy))
   712→      const cap = (before.computed.gross * policy.fineCapPct) / 100
   713→      if (policy.fineCapPct > 0 && amount > cap) {
   714→        return fail(`سقف جریمهٔ ماهانه ${policy.fineCapPct}٪ جمع حقوق و مزایاست (حداکثر ${Math.round(cap).toLocaleString('en-US')} ریال) — مبلغ واردشده فراتر است`)
   715→      }
   716→    }
   717→
   718→    const created = await db.salaryAdjust.create({
   719→      data: {
   720→        monthId: month.id, kind, amount, note,
   721→        ownerOnly: kind === 'COMMISSION',
   722→        createdById: me.id, createdByName: me.name,
   723→      },
   724→    })
   725→    const after = await recomputeMonth(p.id, name, jy, jm, profile, policy, tax, p.active, statMinWageOf(jy))
   726→    await logActivity(me, `ثبت قلم ${kind}`, 'SalaryAdjust', created.id, `${name} — ${jy}/${jm} — ${amount.toLocaleString('en-US')} ریال${note ? ` — ${note}` : ''}`)
   727→    return json({ ok: true, adjust: created, computed: after.computed })
   728→  }
   729→
   730→  /* ── confirm ── */
   731→  if (action === 'confirm') {
   732→    if (month.status === 'PAID') return fail('ماه پرداخت‌شده است', 409)
   733→    if (month.status === 'CONFIRMED') return json({ ok: true, already: true })
   734→    const cur = monthKeyOf(todayIso())
   735→    if (jy > cur.jy || (jy === cur.jy && jm > cur.jm)) {
   736→      return fail('ماه آینده قابل تأیید نیست — ثبت پیش‌نویس آزاد است، تأیید پس از پایان ماه انجام می‌شود')
   737→    }
   738→    const { computed } = await recomputeMonth(p.id, name, jy, jm, profile, policy, tax, p.active, statMinWageOf(jy))
   739→    if (computed.gross <= 0) return fail('برای تأیید، جمع حقوق و مزایا باید بزرگ‌تر از صفر باشد')
   740→    await db.salaryMonth.update({
   741→      where: { id: month.id },
   742→      data: { status: 'CONFIRMED', confirmedById: me.id, confirmedByName: me.name, confirmedAt: new Date().toISOString() },
   743→    })
   744→    await logActivity(me, 'تأیید ماه حقوق', 'SalaryMonth', month.id, `${name} — ${jy}/${jm} — خالص ${computed.net.toLocaleString('en-US')} ریال${p.active ? '' : ' — هشدار: پرسنل غیرفعال'}`)
   745→    return json({ ok: true, warning: p.active ? undefined : 'این پرسنل غیرفعال است — پیش از پرداخت بررسی کنید' })
   746→  }
   747→
   748→  /* ── pay ── */
   749→  if (action === 'pay') {
   750→    if (month.status === 'PAID') return json({ ok: true, already: true })
   751→    if (month.status !== 'CONFIRMED') return fail('ابتدا ماه را تأیید کنید، سپس پرداخت را ثبت نمایید')
   752→    const payslipNo = month.payslipNo || `HZ-${jy}${String(jm).padStart(2, '0')}-${p.id.slice(-6).toUpperCase()}`
   753→    await db.salaryMonth.update({ where: { id: month.id }, data: { status: 'PAID', paidAt: new Date().toISOString(), payslipNo } })
   754→    await logActivity(me, 'ثبت پرداخت حقوق', 'SalaryMonth', month.id, `${name} — ${jy}/${jm} — خالص ${month.net.toLocaleString('en-US')} ریال — فیش ${payslipNo}`)
   755→    return json({ ok: true, payslipNo })
   756→  }
   757→
   758→  /* ── reopen ── */
   759→  if (action === 'reopen') {
   760→    if (month.status === 'DRAFT') return json({ ok: true, already: true })
   761→    const note = String(body.note || '').trim()
   762→    if (month.status === 'PAID' && month.paidAt && !note) {
   763→      return fail('بازگشایی ماه پرداخت‌شده مستلزم ذکر دلیل است — یادداشت الزامی است')
   764→    }
   765→    await db.salaryMonth.update({
   766→      where: { id: month.id },
   767→      data: { status: 'DRAFT', confirmedById: '', confirmedByName: '', confirmedAt: '', paidAt: '' },
   768→    })
   769→    await logActivity(me, 'بازگشایی ماه حقوق (حسابرسی)', 'SalaryMonth', month.id, `${name} — ${jy}/${jm} — وضعیت قبلی: ${month.status}${note ? ` — دلیل: ${note}` : ''}`)
   770→    return json({ ok: true })
   771→  }
   772→
   773→  return fail('اکشن نامعتبر است')
   774→}
