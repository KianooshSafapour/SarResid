   651→/** برگه شمارش: ورود تعداد شمارش‌شده + بستن با اصلاح خودکار موجودی */
   652→function CountModal({ count, onClose, onChanged }: { count: Count; onClose: () => void; onChanged: () => void }) {
   653→  const [items, setItems] = useState<CountItem[]>(count.items)
   654→  const [saving, setSaving] = useState(false)
   655→  const counted = items.filter((i) => i.countedQty !== null).length
   656→  const diffs = items.filter((i) => i.countedQty !== null && i.diff !== 0)
   657→  const acc = counted ? Math.round(((counted - diffs.length) / counted) * 100) : null
   658→
   659→  const setQty = (productId: string, v: number | '') => {
   660→    setItems((arr) =>
   661→      arr.map((it) => (it.productId === productId ? { ...it, countedQty: v === '' ? null : v, diff: v === '' ? 0 : (v as number) - it.systemQty } : it))
   662→    )
   663→  }
   664→
   665→  const save = async (close: boolean) => {
   666→    setSaving(true)
   667→    try {
   668→      await api('/api/cyclecount', {
   669→        method: 'PUT',
   670→        body: {
   671→          id: count.id,
   672→          close,
   673→          updates: items.filter((i) => i.countedQty !== null).map((i) => ({ productId: i.productId, countedQty: i.countedQty })),
   674→        },
   675→      })
   676→      toast.success(close ? 'برگه بسته شد و موجودی‌ها اصلاح شد ✅' : 'شمارش ذخیره شد')
   677→      onChanged()
   678→      if (close) onClose()
   679→    } catch (e: any) {
   680→      toast.error(e.message)
   681→    } finally {
   682→      setSaving(false)
   683→    }
   684→  }
   685→
   686→  return (
   687→    <Modal title={`برگه شمارش — ${count.title}`} onClose={onClose}>
   688→      <div className="mb-3 flex flex-wrap items-center gap-2 text-xs">
   689→        <Pill label={`${faNum(counted)}/${faNum(items.length)} شمرده‌شده`} color="#0e7a4a" />
   690→        {acc !== null && <Pill label={`دقت: ${faNum(acc)}٪`} color={acc >= 95 ? '#0e7a4a' : acc >= 80 ? '#c9a227' : '#b3372f'} />}
   691→        {diffs.length > 0 && <Pill label={`${faNum(diffs.length)} مغایرت`} color="#b3372f" />}
   692→        <span className="text-muted-foreground">کالای مغایرت را با مقدار واقعی پر کنید؛ در بستن برگه، موجودی اصلاح می‌شود</span>
   693→      </div>
   694→      <div className="scroll-gold max-h-[55vh] space-y-2 overflow-y-auto pl-1">
   695→        {items.map((it) => (
   696→          <div key={it.productId} className={cn('flex items-center gap-2 rounded-xl border p-2.5', it.diff !== 0 && it.countedQty !== null ? 'border-[#b3372f]/40 bg-[#fee2e2]/30' : 'border-border/70 bg-muted/20')}>
   697→            <div className="min-w-0 flex-1">
   698→              <p className="truncate text-xs font-bold">{it.productName}</p>
   699→              <p className="text-[10px] text-muted-foreground">سیستم: {faNum(it.systemQty)} {it.unit}</p>
   700→            </div>
   701→            {it.countedQty !== null && it.diff !== 0 && (
   702→              <Pill label={`${it.diff > 0 ? '+' : ''}${faNum(it.diff)}`} color={it.diff > 0 ? '#8a6d10' : '#b3372f'} />
   703→            )}
   704→            <FaPriceInput value={it.countedQty ?? ''} onChange={(v) => setQty(it.productId, v)} className="w-24 rounded-xl border border-input bg-white px-2 py-2" ariaLabel={`شمارش ${it.productName}`} />
   705→          </div>
   706→        ))}
   707→      </div>
   708→      <div className="mt-4 flex flex-wrap justify-end gap-2">
   709→        <button onClick={onClose} className="rounded-xl bg-muted px-4 py-2.5 text-sm font-bold">بستن پنجره</button>
   710→        <button onClick={() => save(false)} disabled={saving} className="rounded-xl bg-secondary px-5 py-2.5 text-sm font-black text-primary shadow">ذخیره شمارش</button>
   711→        <button onClick={() => save(true)} disabled={saving || counted === 0} className="rounded-xl bg-[#0e7a4a] px-5 py-2.5 text-sm font-black text-white shadow-lg disabled:opacity-50">
   712→          بستن برگه + اصلاح موجودی
   713→        </button>
   714→      </div>
   715→    </Modal>
   716→  )
   717→}
