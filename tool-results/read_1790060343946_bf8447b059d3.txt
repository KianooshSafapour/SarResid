   715→   715→                            className="rounded-xl border border-border px-4 py-2 text-xs font-bold text-muted-foreground transition hover:bg-muted"
   716→   716→                          >
   717→   717→                            انصراف
   718→   718→                          </button>
   719→   719→                        </div>
   720→   720→                      </div>
   721→   721→                    ) : rejectFor === l.id ? (
   722→   722→                      <div className="mt-2 flex flex-wrap gap-2">
   723→   723→                        <input
   724→   724→                          value={rejectNote}
   725→   725→                          onChange={(e) => setRejectNote(e.target.value)}

   727→   727→                          className="min-w-[160px] flex-1 rounded-xl border border-input bg-white/90 px-3 py-2 text-xs shadow-sm outline-none focus:border-primary focus:ring-2 focus:ring-primary/20"
   728→   728→                        />
   729→   729→                        <button
   730→   730→                          onClick={() => act(l, 'reject', rejectNote)}
   731→   731→                          className="rounded-xl bg-[#b3372f] px-4 py-2 text-xs font-black text-white transition hover:bg-[#96291f]"
   732→   732→                        >
   733→   733→                          ثبت رد
   734→   734→                        </button>
   735→   735→                        <button
   736→   736→                          onClick={() => { setRejectFor(null); setRejectNote('') }}
   737→   737→                          className="rounded-xl border border-border px-4 py-2 text-xs font-bold text-muted-foreground transition hover:bg-muted"
   738→   738→                        >
   739→   739→                          انصراف
   740→   740→                        </button>
   741→   741→                      </div>
   742→   742→                    ) : (
   743→   743→                      <div className="mt-2 flex gap-2">
   744→   744→                        <button
   745→   745→                          onClick={() => requestApprove(l)}
   746→   746→                          className="flex items-center gap-1 rounded-xl bg-[#0e7a4a] px-4 py-2 text-xs font-black text-white transition hover:bg-[#0b6a40]"

   748→   748→                          <Check size={14} /> تأیید

   750→   750→                        <button
   751→   751→                          onClick={() => { setRejectFor(l.id); setRejectNote('') }}
   752→   752→                          className="rounded-xl border border-[#b3372f] px-4 py-2 text-xs font-black text-[#b3372f] transition hover:bg-[#b3372f]/10"
   753→   753→                        >
   754→   754→                          رد
   755→   755→                        </button>
   756→   756→                      </div>
   757→   757→                    )}
   758→   758→                  </div>
   759→   759→                )
   760→   760→              })}
   761→   761→            </div>
   762→   762→          )}
   763→   763→        </SectionCard>
   764→   764→      )}
   765→   765→
   766→   766→      {/* مبانی علمی */}
   767→   767→      <SectionCard title="مبانی علمی" icon={<BookOpen size={18} />} subtitle="چرا نوبت‌بندی و تقویم مشترک؟">
   768→   768→        <ul className="space-y-2 text-sm leading-7 text-foreground/85">
   769→   769→          <li>
   770→   770→            قاعدهٔ «اولین ثبت، اولویت» (FCFS) کلاسیک‌ترین سازوکار انصاف در نظریهٔ صف‌ها و عدالت رویه‌ای است؛ پژوهش‌ها نشان می‌دهد
   771→   771→            ادراک بی‌طرفی سامانه زمانی بیشینه می‌شود که نتیجه تنها تابع زمان ثبت باشد، نه جایگاه یا رابطه
   772→   772→            <span className="text-[11px] text-muted-foreground"> (Rafaeli et al., 1997; Su, 2009)</span>.
   773→   773→          </li>
   774→   774→          <li>
   775→   775→            تقویم مشترک مرخصی نیز همان «برنامه‌ریزی روزهای عدم حضور» در ادبیات زمان‌بندی نیروی کار است: رؤیت‌پذیری زودهنگام
   776→   776→            غیبت‌ها امکان پوشش شیفت‌ها و روزهای پیک فروش را بدون تعارض فراهم می‌کند
   777→   777→            <span className="text-[11px] text-muted-foreground"> (Ernst et al., 2004; Van den Bergh et al., 2013)</span>.
   778→   778→          </li>
   779→   779→        </ul>
   780→   780→      </SectionCard>
   781→   781→
   782→   782→      {/* مودال درخواست جدید */}
   783→   783→      {addOpen && (
   784→   784→        <Modal title="درخواست مرخصی جدید" onClose={() => setAddOpen(false)}>
   785→   785→          <Labeled label="نوع مرخصی">
   786→   786→            <div className="grid grid-cols-2 gap-2">
   787→   787→              {Object.entries(TYPE_META).map(([key, meta]) => (
   788→   788→                <button
   789→   789→                  key={key}
   790→   790→                  type="button"
   791→   791→                  onClick={() => pickType(key)}
   792→   792→                  className={cn(
   793→   793→                    'rounded-xl border px-3 py-2.5 text-sm font-bold transition-all',
   794→   794→                    type === key ? 'border-transparent text-white shadow-md' : 'border-border bg-white/80 hover:bg-secondary'
   795→   795→                  )}
   796→   796→                  style={type === key ? { background: meta.color } : undefined}
   797→   797→                >
   798→   798→                  {meta.emoji} {meta.label}
   799→   799→                </button>
   800→   800→              ))}
   801→   801→            </div>
   802→   802→          </Labeled>
   803→   803→
   804→   804→          {type === 'HOURLY' ? (
   805→   805→            <>
   806→   806→              <Labeled label="روز مرخصی (ساعتی — فقط همان روز)" hint="مرخصی ساعتی معادل نیم‌روز محاسبه می‌شود">
   807→   807→                <JalaliDatePicker value={fromDate} onChange={pickFrom} holidays={holidays} minDate={addDaysIso(-1)} warnHoliday={false} />
   808→   808→              </Labeled>
   809→   809→              <div className="grid grid-cols-2 gap-2">
   810→   810→                <Labeled label="از ساعت">
   811→   811→                  <input type="time" dir="ltr" value={fromHour} onChange={(e) => setFromHour(e.target.value)} className={cn(INPUT_CLS, 'text-center font-bold')} />
   812→   812→                </Labeled>
   813→   813→                <Labeled label="تا ساعت">
   814→   814→                  <input type="time" dir="ltr" value={toHour} onChange={(e) => setToHour(e.target.value)} className={cn(INPUT_CLS, 'text-center font-bold')} />
   815→   815→                </Labeled>
   816→   816→              </div>
   817→   817→            </>
   818→   818→          ) : (
   819→   819→            <>
   820→   820→              <Labeled label="از تاریخ">
   821→   821→                <JalaliDatePicker value={fromDate} onChange={pickFrom} holidays={holidays} minDate={addDaysIso(-1)} warnHoliday={false} />
   822→   822→              </Labeled>
   823→   823→              <Labeled label="تا تاریخ">
   824→   824→                <JalaliDatePicker value={toDate} onChange={setToDate} holidays={holidays} minDate={fromDate} warnHoliday={false} />
   825→   825→              </Labeled>
   826→   826→            </>
   827→   827→          )}
   828→   828→
   829→   829→          <Labeled label="دلیل (اختیاری)">
   830→   830→            <textarea
   831→   831→              value={reason}
   832→   832→              onChange={(e) => setReason(e.target.value)}
   833→   833→              rows={2}
   834→   834→              maxLength={300}
   835→   835→              placeholder="مثلاً کار اداری بانک"
   836→   836→              className={cn(INPUT_CLS, 'resize-none')}
   837→   837→            />
   838→   838→          </Labeled>
   839→   839→
   840→   840→          <p className="rounded-xl bg-[#c9a227]/10 px-3 py-2 text-[11px] font-bold leading-5 text-[#8a6d10]">
   841→   841→            نوبت‌بندی به‌صورت «اولین ثبت، اولویت» است؛ ثبت درخواست به معنای تأیید نیست و تصمیم نهایی با مدیر عملیات است.
   842→   842→          </p>
   843→   843→
   844→   844→          <button
   845→   845→            type="button"
   846→   846→            disabled={saving}
   847→   847→            onClick={submit}
   848→   848→            className="w-full rounded-xl bg-[#0e7a4a] py-3 text-sm font-black text-white shadow-md transition hover:bg-[#0b6a40] disabled:opacity-60"
   849→   849→          >
   850→   850→            {saving ? 'در حال ثبت…' : 'ثبت درخواست'}
   851→   851→          </button>
   852→   852→        </Modal>
   853→   853→      )}
   854→   854→    </div>
   855→   855→  )
