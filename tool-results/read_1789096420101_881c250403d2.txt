    67→function lastCheckLabel(row: Row) {
    68→  if (row.checkedToday) return 'امروز کنترل شد ✓'
    69→  if (row.daysSince === null) return 'هرگز کنترل نشده'
    70→  if (row.daysSince === 1) return 'دیروز'
    71→  if (row.daysSince > 1) return `${faNum(row.daysSince)} روز پیش`
    72→  return '—'
    73→}

    75→/** tiny inline SVG trend of the product's last purchase costs — the inflation story at a glance */
    76→function CostSpark({ points }: { points: { price: number; j: string }[] }) {
    77→  if (points.length < 2) return null
    78→  const w = 56
    79→  const h = 20
    80→  const pad = 2
    81→  const min = Math.min(...points.map((p) => p.price))
    82→  const max = Math.max(...points.map((p) => p.price))
    83→  const span = max - min || 1
    84→  const step = (w - pad * 2) / (points.length - 1)
    85→  const coords = points.map((p, i) => [pad + i * step, h - pad - ((p.price - min) / span) * (h - pad * 2)] as const)
    86→  const d = coords.map((c, i) => `${i === 0 ? 'M' : 'L'}${c[0].toFixed(1)},${c[1].toFixed(1)}`).join(' ')
    87→  const up = points[points.length - 1].price >= points[0].price
    88→  const tip = points.map((p) => `${p.j}: ${faMoney(p.price)}`).join(' | ')
    89→  const gid = `spark-${points.length}-${min}-${max}`
    90→  return (
    91→    <svg width={w} height={h} viewBox={`0 0 ${w} ${h}`} className="spark-line shrink-0" role="img" aria-label="روند هزینه خرید">
    92→      <title>{tip}</title>
    93→      <defs>
    94→        <linearGradient id={gid} x1="0" y1="0" x2="1" y2="0">
    95→          <stop offset="0%" stopColor={up ? '#c9a227' : '#0e7a4a'} />
    96→          <stop offset="100%" stopColor={up ? '#b3372f' : '#12905a'} />
    97→        </linearGradient>
    98→      </defs>
    99→      <path d={d} fill="none" stroke={`url(#${gid})`} strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round" />
   100→      <circle cx={coords[coords.length - 1][0]} cy={coords[coords.length - 1][1]} r="2.2" fill={up ? '#b3372f' : '#0e7a4a'} />
   101→    </svg>
   102→  )
   103→}

   105→/** Code128 barcode rendered into a small SVG via JsBarcode — for the shelf label */
   106→function LabelBarcode({ value }: { value: string }) {
   107→  const ref = useRef<SVGSVGElement>(null)
   108→  useEffect(() => {
   109→    if (ref.current && value) {
   110→      try {
   111→        JsBarcode(ref.current, value, {
   112→          format: 'CODE128',
   113→          displayValue: false,
   114→          height: 30,
   115→          width: 1.05,
   116→          margin: 0,
   117→          lineColor: '#111111',
   118→        })
   119→      } catch {
   120→        /* invalid chars — leave the svg empty, the numeric text below is the fallback */
   121→      }
   122→    }
   123→  }, [value])
   124→  return <svg ref={ref} className="label-bc" />
   125→}
