    63→    63→export default function PeopleView({ ctx }: { ctx: AppCtx } & Record<string, unknown>) {
    64→    64→  const user = ctx.user
    65→    65→  const roleKeys = useMemo(
    66→    66→    () => (user ? Array.from(new Set([user.role, ...(user.secondaryRoles || [])])) : []),
    67→    67→    [user],
    68→    68→  )
    69→    69→
    70→    70→  const [tab, setTab] = useState<Tab>('reps')
    71→    71→  const [roleMap, setRoleMap] = useState<Record<string, RoleLite>>({})
    72→    72→
    73→    73→  const canManage = useMemo(() => {
    74→    74→    if (!user) return false
    75→    75→    if (MANAGE_ROLES.includes(user.role)) return true
    76→    76→    return roleKeys.some((k) => roleMap[k]?.caps?.includes('*') || roleMap[k]?.caps?.includes('people.manage'))
    77→    77→  }, [user, roleKeys, roleMap])
    78→    78→
    79→    79→  const canMfg = useMemo(() => {
    80→    80→    if (!user) return false
    81→    81→    if (MANAGE_ROLES.includes(user.role) || user.role === 'ACC') return true
    82→    82→    return roleKeys.some((k) => roleMap[k]?.caps?.includes('*') || roleMap[k]?.caps?.includes('manufacturers.manage'))
    83→    83→  }, [user, roleKeys, roleMap])
    84→    84→
    85→    85→  useEffect(() => {
    86→    86→    api<{ roles: Array<{ key: string; active: boolean; permissions?: { caps?: string[] } }> }>('/api/roles')
    87→    87→      .then((d) => {
    88→    88→        const m: Record<string, RoleLite> = {}
    89→    89→        for (const r of d.roles || []) m[r.key] = { key: r.key, active: r.active, caps: r.permissions?.caps || [] }
    90→    90→        setRoleMap(m)
    91→    91→      })
    92→    92→      .catch(() => { /* fallback to role list above */ })
    93→    93→  }, [])
    94→    94→
    95→    95→  return (
    96→    96→    <div className="space-y-4">
    97→    97→      <div className="flex flex-wrap gap-2 rounded-2xl border border-[#8a5a2b]/25 bg-[#f7efe2] p-1.5">

    99→    99→          ['reps', <Users key="i" size={15} />, 'نمایندگان و اشخاص'],
   100→   100→          ['manufacturers', <Factory key="i" size={15} />, 'تولیدکنندگان'],
   101→   101→          ['guide', <BookOpen key="i" size={15} />, 'راهنما'],
   102→   102→        ] as [Tab, React.ReactNode, string][]).map(([k, icon, label]) => (
   103→   103→          <button
   104→   104→            key={k}
   105→   105→            type="button"
   106→   106→            onClick={() => setTab(k)}
   107→   107→            className={cn(
   108→   108→              'flex min-h-[44px] items-center gap-1.5 rounded-xl px-4 py-2 text-xs font-extrabold transition',
   109→   109→              tab === k ? 'bg-white text-[#0e7a4a] shadow' : 'text-[#8a5a2b] hover:bg-white/60',

   598→   598→function ManufacturerFormModal({ mfg, onClose, onSaved }: { mfg: Manufacturer | null; onClose: () => void; onSaved: () => void }) {
   599→   599→  const [f, setF] = useState({ name: mfg?.name || '', country: mfg?.country || 'ایران', website: mfg?.website || '', phone: mfg?.phone || '', notes: mfg?.notes || '' })

   601→   601→  const [brandInput, setBrandInput] = useState('')
   602→   602→  const [busy, setBusy] = useState(false)

   604→   604→
   605→   605→  const addBrand = () => {
   606→   606→    const b = brandInput.trim()
   607→   607→    if (b && !brands.includes(b)) setBrands((l) => [...l, b])
   608→   608→    setBrandInput('')
   609→   609→  }
   610→   610→
   611→   611→  const save = async () => {
   612→   612→    if (!f.name.trim()) return toast.error('نام تولیدکننده الزامی است')
   613→   613→    setBusy(true)
   614→   614→    try {
   615→   615→      if (mfg) await api(`/api/manufacturers/${mfg.id}`, { method: 'PATCH', body: { ...f, brands } })

   617→   617→      onSaved()
   618→   618→    } catch (e: any) {
   619→   619→      toast.error(e.message)
   620→   620→    } finally {
   621→   621→      setBusy(false)

   623→   623→  }

   654→   654→function AssignProvidersModal({ mfg, providers, onClose, onSaved }: { mfg: Manufacturer; providers: ProviderLite[]; onClose: () => void; onSaved: () => void }) {
   655→   655→  const [selected, setSelected] = useState<string[]>(mfg.providers.map((p) => p.id))
   656→   656→  const [busy, setBusy] = useState(false)

   658→   658→

   660→   660→    setBusy(true)
   661→   661→    try {
   662→   662→      await api(`/api/manufacturers/${mfg.id}`, { method: 'PATCH', body: { action: 'assign-providers', providerIds: selected } })
   663→   663→      onSaved()
   664→   664→    } catch (e: any) {
   665→   665→      toast.error(e.message)
   666→   666→    } finally {
   667→   667→      setBusy(false)
   668→   668→    }
   669→   669→  }
   670→   670→
   671→   671→  return (
   672→   672→    <Modal title={`تأمین‌کنندگان ${mfg.name}`} onClose={onClose} wide>

   692→   692→              >
   693→   693→                <span className="text-xs font-black">{p.name}</span>
   694→   694→                <span className={cn('rounded-full px-2.5 py-0.5 text-[10px] font-black', on ? 'bg-[#0e7a4a] text-white' : 'bg-muted text-muted-foreground')}>
   695→   695→                  {on ? '✓ متصل' : 'غیرمتصل'}
