   674→/** register a document → auto location assignment + line items builder + instant physical-location card */
   675→function IntakeModal({ binders, onClose, onSaved }: {
   676→  binders: Binder[]
   677→  onClose: () => void
   678→  onSaved: (doc: Doc, binder: any, pickReason: string, itemsCreated: number) => void
   679→}) {
   680→  type ItemRow = { productName: string; productId: string; qty: string; unitPrice: string; expiry: string }
   681→  const emptyRow = (): ItemRow => ({ productName: '', productId: '', qty: '', unitPrice: '', expiry: '' })
   682→
   683→  const [form, setForm] = useState({ title: '', docType: 'INVOICE', party: '', amount: '', invoiceNo: '', docDate: todayIso(), binderId: '', refOrderCode: '', notes: '' })
   684→  const [rows, setRows] = useState<ItemRow[]>([emptyRow()])
   685→  const [productOpts, setProductOpts] = useState<{ id: string; name: string; buyPrice: number; unit: string }[]>([])
   686→  const [busy, setBusy] = useState(false)
   687→  const set = (k: string, v: string) => setForm((f) => ({ ...f, [k]: v }))
   688→
   689→  useEffect(() => {
   690→    api<{ products: any[] }>('/api/products')
   691→      .then((d) => setProductOpts((d.products || []).slice(0, 200).map((p) => ({ id: p.id, name: p.name, buyPrice: p.buyPrice, unit: p.unit }))))
   692→      .catch(() => { /* free-text still works without suggestions */ })
   693→  }, [])
   694→
   695→  const setRow = (idx: number, patch: Partial<ItemRow>) =>
   696→    setRows((rs) => rs.map((r, i) => (i === idx ? { ...r, ...patch } : r)))
   697→
   698→  const onRowName = (idx: number, name: string) => {
   699→    const p = productOpts.find((o) => o.name === name)
   700→    if (p) setRow(idx, { productName: p.name, productId: p.id, unitPrice: rows[idx].unitPrice === '' && p.buyPrice > 0 ? String(p.buyPrice) : rows[idx].unitPrice })
   701→    else setRow(idx, { productName: name, productId: '' })
   702→  }
   703→
   704→  const rowsTotal = rows.reduce((s, r) => s + (Number(r.qty) || 0) * (Number(r.unitPrice) || 0), 0)
   705→  const validRows = rows.filter((r) => r.productName.trim())
   706→  const effectiveAmount = Number(form.amount) || (validRows.length ? rowsTotal : 0)
   707→
   708→  const save = async () => {
   709→    if (!form.title || !form.party) return toast.error('عنوان سند و طرف حساب الزامی است')
   710→    const items = validRows.map((r) => ({
   711→      productId: r.productId,
   712→      productName: r.productName.trim(),
   713→      qty: Number(r.qty) || 0,
   714→      unitPrice: Number(r.unitPrice) || 0,
   715→      expiryDate: r.expiry || '',
   716→    }))
   717→    setBusy(true)
   718→    try {
   719→      const res = await api<{ doc: Doc; binder: any; pickReason: string; itemsCreated: number }>('/api/archive', {
   720→        method: 'POST',
   721→        body: { kind: 'doc', ...form, amount: effectiveAmount, items },
   722→      })
   723→      toast.success('سند در آرشیو ثبت شد')
   724→      onSaved(res.doc, res.binder, res.pickReason, res.itemsCreated || 0)
   725→    } catch (e: any) { toast.error(e.message) } finally { setBusy(false) }
   726→  }
   727→
   728→  return (
   729→    <Modal title="ثبت سند در آرشیو فیزیکی" onClose={onClose} wide>
   730→      <div className="grid gap-3 sm:grid-cols-2">
   731→        <Labeled label="عنوان سند *"><input value={form.title} onChange={(e) => set('title', e.target.value)} className="w-full rounded-xl border border-input p-3 text-sm" placeholder="مثلاً: فاکتور شماره ۲۴۱ کاله" /></Labeled>
   732→        <Labeled label="نوع سند">
   733→          <select value={form.docType} onChange={(e) => set('docType', e.target.value)} className="w-full rounded-xl border border-input bg-white p-3 text-sm">
   734→            {Object.entries(DOC_TYPES).map(([k, v]) => <option key={k} value={k}>{v.emoji} {v.label}</option>)}
   735→          </select>
   736→        </Labeled>
   737→        <Labeled label="طرف حساب *"><input value={form.party} onChange={(e) => set('party', e.target.value)} className="w-full rounded-xl border border-input p-3 text-sm" placeholder="نام تأمین‌کننده / شرکت" /></Labeled>
   738→        <Labeled label="شماره فاکتور (روی سند فیزیکی)"><input value={form.invoiceNo} onChange={(e) => set('invoiceNo', e.target.value)} className="w-full rounded-xl border border-input p-3 text-sm" dir="ltr" placeholder="241-1405" /></Labeled>
   739→        <Labeled label="مبلغ سند (تومان)" hint={validRows.length > 0 ? `جمع اقلام: ${faMoney(rowsTotal)} تومان — اگر خالی بماند، از جمع اقلام محاسبه می‌شود` : undefined}>
   740→          <FaPriceInput
   741→            value={form.amount === '' ? '' : Number(form.amount)}
   742→            onChange={(v) => set('amount', v === '' ? '' : String(v))}
   743→            ariaLabel="مبلغ سند"
   744→            className="w-full rounded-xl border border-input p-3 text-sm"
   745→          />
   746→        </Labeled>
   747→        <Labeled label="تاریخ سند"><JalaliDatePicker value={form.docDate} onChange={(v) => set('docDate', v)} holidays={new Map()} /></Labeled>
   748→        <Labeled label="زونکن مقصد">
   749→          <select value={form.binderId} onChange={(e) => set('binderId', e.target.value)} className="w-full rounded-xl border border-input bg-white p-3 text-sm">
   750→            <option value="">پیشنهاد خودکار سامانه 🎯</option>
   751→            {binders.map((b) => <option key={b.id} value={b.id}>{b.code} — {b.title}</option>)}
   752→          </select>
   753→        </Labeled>
   754→        <Labeled label="کد سفارش مرتبط (اختیاری)"><input value={form.refOrderCode} onChange={(e) => set('refOrderCode', e.target.value)} className="w-full rounded-xl border border-input p-3 text-sm" dir="ltr" /></Labeled>
   755→      </div>
   756→      <Labeled label="توضیح (اختیاری)"><input value={form.notes} onChange={(e) => set('notes', e.target.value)} className="w-full rounded-xl border border-input p-3 text-sm" /></Labeled>
   757→
   758→      {/* ── line items builder (سجل کالا) ── */}
   759→      <div className="mt-4 rounded-2xl border border-[#8a5a2b]/25 bg-[#f7efe2] p-3">
   760→        <div className="mb-2 flex items-center justify-between">
   761→          <p className="text-xs font-black text-[#8a5a2b]">📋 اقلام فاکتور — سجل کالا (اختیاری)</p>
   762→          <p className="text-[11px] font-black text-[#0e7a4a]">جمع: {faMoney(rowsTotal)} تومان</p>
   763→        </div>
   764→        <div className="space-y-2">
   765→          {rows.map((r, idx) => (
   766→            <div key={idx} className="grid grid-cols-[1fr_auto] items-center gap-2 rounded-xl bg-white/85 p-2">
   767→              <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-4">
   768→                <input
   769→                  value={r.productName}
   770→                  onChange={(e) => onRowName(idx, e.target.value)}
   771→                  list="intake-product-opts"
   772→                  className="w-full rounded-lg border border-input p-2 text-xs"
   773→                  placeholder="نام کالا (انتخاب یا تایپ آزاد)"
   774→                />
   775→                <input
   776→                  value={r.qty}
   777→                  onChange={(e) => setRow(idx, { qty: e.target.value.replace(/[^\d.]/g, '') })}
   778→                  className="w-full rounded-lg border border-input p-2 text-xs"
   779→                  dir="ltr"
   780→                  inputMode="decimal"
   781→                  placeholder="مقدار"
   782→                />
   783→                <FaPriceInput
   784→                  value={r.unitPrice === '' ? '' : Number(r.unitPrice)}
   785→                  onChange={(v) => setRow(idx, { unitPrice: v === '' ? '' : String(v) })}
   786→                  ariaLabel={`قیمت واحد ردیف ${idx + 1}`}
   787→                  className="w-full rounded-lg border border-input p-2 text-xs"
   788→                />
   789→                <JalaliDatePicker value={r.expiry} onChange={(v) => setRow(idx, { expiry: v })} holidays={new Map()} />
   790→              </div>
   791→              <button
   792→                type="button"
   793→                onClick={() => setRows((rs) => (rs.length === 1 ? [emptyRow()] : rs.filter((_, i) => i !== idx)))}
   794→                aria-label="حذف ردیف"
   795→                className="flex h-11 w-11 items-center justify-center rounded-lg bg-[#b3372f]/10 text-sm font-black text-[#b3372f]"
   796→              >
   797→                ✕
   798→              </button>
   799→            </div>
   800→          ))}
   801→        </div>
   802→        <datalist id="intake-product-opts">
   803→          {productOpts.map((o) => <option key={o.id} value={o.name} />)}
   804→        </datalist>
   805→        <button
   806→          type="button"
   807→          onClick={() => setRows((rs) => [...rs, emptyRow()])}
   808→          className="mt-2 w-full rounded-xl border border-dashed border-[#8a5a2b]/40 bg-white/60 py-2.5 text-[11px] font-black text-[#8a5a2b]"
   809→        >
   810→          ＋ افزودن ردیف کالا
   811→        </button>
   812→        <p className="mt-2 text-[10px] leading-relaxed text-muted-foreground">
   813→          با ثبت اقلام، تاریخچهٔ هر کالا (خرید، انقضا، قیمت) به سجل کالا اضافه می‌شود و از تب «کارت کالا» تا سند فیزیکی قابل ردیابی است.
   814→        </p>
   815→      </div>
   816→
   817→      <p className="mt-2 rounded-xl bg-[#fdf6dd]/70 p-2.5 text-[11px] leading-relaxed text-muted-foreground">
   818→        در حالت «پیشنهاد خودکار»، سامانه بر پایهٔ تطبیق موضوعی طرف حساب با دستهٔ زونکن‌ها (و در نبود تطبیق، کم‌ترین پرشدگی) جایگاه سند را تعیین و کد ماندگار می‌سازد؛ طرف حساب همچنین با فهرست تأمین‌کنندگان تطبیق داده می‌شود.
   819→      </p>
   820→      <button onClick={save} disabled={busy} className="mt-3 w-full rounded-xl bg-primary py-3 text-sm font-extrabold text-white disabled:opacity-50">
   821→        {busy ? '…' : 'ثبت و تعیین جایگاه فیزیکی'}
   822→      </button>
   823→    </Modal>
   824→  )
   825→}
