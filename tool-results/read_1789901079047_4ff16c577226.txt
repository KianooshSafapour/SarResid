   139→/** ISO datetime → مقدار input[type=datetime-local] به وقت محلی */
   140→function toLocalInput(iso?: string): string {
   141→  const d = iso ? new Date(iso) : new Date()
   142→  if (isNaN(d.getTime())) return ''
   143→  const off = d.getTimezoneOffset() * 60000
   144→  return new Date(d.getTime() - off).toISOString().slice(0, 16)
   145→}
