    65→const WEEKDAY_SHORT = ['ش', 'ی', 'د', 'س', 'چ', 'پ', 'ج']
    66→function wdShort(iso: string): string {
    67→  const d = new Date(iso + 'T12:00:00').getDay()
    68→  return WEEKDAY_SHORT[d === 6 ? 0 : d + 1]
    69→}
    70→/** compact money for calendar cells: ۸۵م / ۱٫۲م / ۹۵۰ه */
    71→function faMoneyShort(v: number): string {
    72→  if (v >= 1_000_000_000) return `${faNum((v / 1_000_000_000).toFixed(1))} میلیارد`
    73→  if (v >= 1_000_000) return `${faNum(Math.round(v / 1_000_000))}م`
    74→  if (v >= 1000) return `${faNum(Math.round(v / 1000))}ه`
    75→  return faNum(v)
    76→}

   583→/** month calendar with per-day amounts + click-to-detail (+ schedule preview dots) */
   584→function ChequeCalendar({ byDate, holidays, flexRules, schedOverlay, jy, jm, setJy, setJm, selDay, onPick }: any) {
   585→  const cells: (number | null)[] = []
   586→  const len = jMonthLength(jy, jm)
   587→  const start = jMonthStartWeekday(jy, jm)
   588→  for (let i = 0; i < start; i++) cells.push(null)
   589→  for (let d = 1; d <= len; d++) cells.push(d)
   590→  const move = (dir: number) => {
   591→    let m = jm + dir, y = jy
   592→    if (m > 12) { m = 1; y++ }
   593→    if (m < 1) { m = 12; y-- }
   594→    setJm(m); setJy(y)
   595→  }
   596→  const previewByDate = useMemo(() => {
   597→    const m = new Map<string, { amount: number; opts: Set<string> }>()
   598→    for (const o of schedOverlay || []) {
   599→      const cur = m.get(o.dateIso) || { amount: 0, opts: new Set<string>() }
   600→      cur.amount += o.amount
   601→      cur.opts.add(o.option)
   602→      m.set(o.dateIso, cur)
   603→    }
   604→    return m
   605→  }, [schedOverlay])
   606→  return (
   607→    <div className="w-full max-w-[340px]">
   608→      <div className="mb-2 flex items-center justify-between">
   609→        <button onClick={() => move(1)} className="rounded-lg px-2 py-1 hover:bg-muted">›</button>
   610→        <span className="text-sm font-extrabold">{J_MONTHS[jm - 1]} {faNum(jy)}</span>
   611→        <button onClick={() => move(-1)} className="rounded-lg px-2 py-1 hover:bg-muted">‹</button>
   612→      </div>
   613→      <div className="grid grid-cols-7 gap-1 text-center text-[9px] font-bold text-muted-foreground">
   614→        {WEEKDAY_SHORT.map((d, i) => <span key={i} className={cn(i === 6 && 'text-[#b3372f]')}>{d}</span>)}
   615→      </div>
   616→      <div className="mt-1 grid grid-cols-7 gap-1">
   617→        {cells.map((d, i) => {
   618→          if (d === null) return <span key={i} />
   619→          const iso = jalaliToIso(jy, jm, d)
   620→          const cheques = byDate.get(iso) || []
   621→          const hol = holidays.get(iso)
   622→          const fri = new Date(iso + 'T12:00:00').getDay() === 5
   623→          const dayTotal = cheques.reduce((s: number, c: Cheque) => s + c.amount, 0)
   624→          const preview = previewByDate.get(iso)
   625→          const isSel = selDay === iso
   626→          let hijriTiny = ''
   627→          try { hijriTiny = faNum(isoToHijri(iso).hd) } catch { /* ignore */ }
   628→          return (
   629→            <button
   630→              key={i}
   631→              type="button"
   632→              onClick={() => onPick(iso)}
   633→              title={[
   634→                formatJalaliFull(iso),
   635→                formatHijri(iso) && `هجری: ${formatHijri(iso)}`,
   636→                hol && `تعطیل رسمی: ${hol}`,
   637→                cheques.map((c: Cheque) => `${c.recipientName}: ${faMoney(c.amount)}`).join('\n'),
   638→                preview && `پیش‌نمایش مشاور (${Array.from(preview.opts).map((o) => faNum(o === 'A' ? 1 : o === 'B' ? 2 : 3)).join('+')}): ${faMoney(preview.amount)}`,
   639→              ].filter(Boolean).join(' • ')}
   640→              className={cn(
   641→                'relative flex h-11 w-full flex-col items-center justify-center rounded-lg text-[11px] font-bold transition',
   642→                isSel ? 'ring-2 ring-[#c9a227]' : '',
   643→                cheques.length ? 'bg-[#c9a227]/20 ring-1 ring-[#c9a227]/70' : preview ? 'bg-[#c9a227]/10 ring-1 ring-dashed ring-[#b8860b]/60' : hol ? 'bg-[#b3372f]/8 text-[#b3372f]' : 'hover:bg-muted',
   644→                (hol || fri) && !cheques.length && 'text-[#b3372f]/70',
   645→              )}
   646→            >
   647→              <span>{faNum(d)}</span>
   648→              {hijriTiny && (
   649→                <span className="absolute left-1 top-0.5 text-[6.5px] font-bold leading-none text-muted-foreground/70">{hijriTiny}</span>
   650→              )}
   651→              {preview && (
   652→                <span className="absolute right-0.5 top-0.5 h-1.5 w-1.5 rounded-full border border-dashed border-[#b8860b] bg-[#c9a227]/60" title={`پیش‌نمایش: ${faMoney(preview.amount)}`} />
   653→              )}
   654→              {cheques.length > 0 && <span className="text-[8px] font-black leading-none text-[#8a5a2b]">{faMoneyShort(dayTotal)}</span>}
   655→              {cheques.length === 0 && hol && <span className="absolute bottom-0.5 h-1 w-1 rounded-full bg-[#b3372f]" />}
   656→              {cheques.length === 0 && !hol && fri && <span className="absolute bottom-0.5 h-1 w-1 rounded-full bg-[#1c2a22]/30" />}
   657→            </button>
   658→          )
   659→        })}
   660→      </div>
   661→      <p className="mt-1 text-center text-[8.5px] text-muted-foreground">عدد کوچک گوشهٔ هر خانه = روز هجری قمری (تبدیل واقعی) — نقطهٔ خط‌چین طلایی = پیش‌نمایش مشاور</p>
   662→    </div>
   663→  )
   664→}
