   401→function SmartReorderModal({ onClose, onDone }: { onClose: () => void; onDone: (created: number) => void }) {
   402→  const [products, setProducts] = useState<Product[]>([])
   403→  const [loaded, setLoaded] = useState(false)
   404→  const [providers, setProviders] = useState<ProviderRow[]>([])
   405→  const [qtys, setQtys] = useState<Record<string, number>>({})
   406→  const [excluded, setExcluded] = useState<Record<string, boolean>>({}) // by providerId
   407→  const [skipped, setSkipped] = useState<Record<string, boolean>>({}) // per product
   408→  const [busy, setBusy] = useState(false)
   409→
   410→  // فهرست کم‌موجودی‌ها از سرور (پارامتر lowstock=1 — مقیاس‌پذیر برای ۱۰هزار کالا)
   411→  useEffect(() => {
   412→    api<{ products: Product[] }>('/api/products?lowstock=1&page=1&pageSize=500')
   413→      .then((d) => { setProducts(d.products); setLoaded(true) })
   414→      .catch(() => setLoaded(true))
   415→    api<{ providers: ProviderRow[] }>('/api/providers').then((d) => setProviders(d.providers)).catch(() => {})
   416→  }, [])
   417→
   418→  const low = useMemo(() => products.filter((p) => p.stock <= p.reorderLevel), [products])
   419→  const suggested = (p: Product) => Math.max(p.reorderLevel * 2 - p.stock, 1)
   420→
   421→  // initialize suggestion quantities once
   422→  useEffect(() => {
   423→    if (Object.keys(qtys).length === 0 && low.length > 0) {
   424→      const init: Record<string, number> = {}
   425→      for (const p of low) init[p.id] = suggested(p)
   426→      setQtys(init)
   427→    }
   428→  }, [low.length])
   429→
   430→  const groups = useMemo(() => {
   431→    const m = new Map<string, Product[]>()
   432→    for (const p of low) {
   433→      const key = p.providerId || '_none'
   434→      if (!m.has(key)) m.set(key, [])
   435→      m.get(key)!.push(p)
   436→    }
   437→    return [...m.entries()]
   438→  }, [low])
   439→
   440→  const activeGroups = groups.filter(([pid]) => !excluded[pid])
   441→  const activeItems = activeGroups.flatMap(([, ps]) => ps).filter((p) => !skipped[p.id])
   442→  const estTotal = activeItems.reduce((s, p) => s + (qtys[p.id] || 0) * p.buyPrice, 0)
   443→
   444→  const create = async () => {
   445→    if (activeItems.length === 0) return
   446→    setBusy(true)
   447→    const d = new Date()
   448→    d.setDate(d.getDate() + 2)
   449→    const deliveryDate = d.toISOString().slice(0, 10)
   450→    let created = 0
   451→    try {
   452→      for (const [pid, ps] of activeGroups) {
   453→        const items = ps.filter((p) => !skipped[p.id])
   454→        if (items.length === 0) continue
   455→        const prov = providers.find((x) => x.id === pid)
   456→        await api('/api/orders', {
   457→          method: 'POST',
   458→          body: {
   459→            providerId: pid,
   460→            providerName: prov?.name || 'بدون تأمین‌کننده',
   461→            status: 'DRAFT',
   462→            deliveryDate,
   463→            notes: 'پیش‌نویس هوشمند — بر اساس نقطه سفارش و موجودی فعلی',
   464→            items: items.map((p) => ({ productId: p.id, qty: qtys[p.id] || suggested(p), unitBuyPrice: p.buyPrice })),
   465→          },
   466→        })
   467→        created++
   468→      }
   469→      onDone(created)
   470→    } catch (e: any) {
   471→      toast.error(e.message)
   472→    } finally {
   473→      setBusy(false)
   474→    }
   475→  }
   476→
   477→  return (
   478→    <Modal title="پیش‌نویس هوشمند سفارش 🤖" onClose={onClose} wide>
   479→      <p className="mb-4 rounded-xl border border-[#77934a]/30 bg-[#e9f0e4]/60 p-3 text-[11px] leading-5">
   480→        سامانه {faNum(low.length)} کالای رسیده به نقطه سفارش را پیدا کرده و بر اساس «دو برابر نقطه سفارش منهای موجودی فعلی» پیشنهاد تعداد می‌دهد.
   481→        تعدادها را اصلاح کنید، تأمین‌کننده‌های غیرضروری را بردارید و سفارش‌های پیش‌نویس را یک‌جا بسازید — بعد از بخش سفارش‌ها ارسالشان کنید.
   482→      </p>
   483→
   484→      {!loaded ? (
   485→        <div className="space-y-2">
   486→          {Array.from({ length: 4 }).map((_, i) => <div key={i} className="h-12 animate-pulse rounded-xl bg-muted/60" />)}
   487→        </div>
   488→      ) : low.length === 0 ? (
   489→        <EmptyState emoji="🌿" title="همه موجودی‌ها سالم‌اند" hint="فعلاً کالایی به نقطه سفارش نرسیده" />
   490→      ) : (
   491→        <div className="scroll-gold max-h-[52vh] space-y-3 overflow-y-auto pl-1">
   492→          {groups.map(([pid, ps]) => {
   493→            const prov = providers.find((x) => x.id === pid)
   494→            const isExcluded = !!excluded[pid]
   495→            const groupTotal = ps.reduce((s, p) => s + (qtys[p.id] || 0) * p.buyPrice, 0)
   496→            return (
   497→              <div key={pid} className={cn('rounded-2xl border p-3 transition', isExcluded ? 'border-border bg-muted/40 opacity-60' : 'glow-card border-[#77934a]/30 bg-white/80')}>
   498→                <div className="mb-2 flex flex-wrap items-center justify-between gap-2">
   499→                  <label className="flex cursor-pointer items-center gap-2">
   500→                    <input
   501→                      type="checkbox"
   502→                      checked={!isExcluded}
   503→                      onChange={(e) => setExcluded((x) => ({ ...x, [pid]: !e.target.checked }))}
   504→                      className="h-4 w-4 accent-[#77934a]"
   505→                    />
   506→                    <span className="text-sm font-black">🚚 {prov?.name || 'بدون تأمین‌کننده مشخص'}</span>
   507→                    <span className="rounded-full bg-secondary px-2 py-0.5 text-[10px] font-bold text-[#5c7236]">{faNum(ps.length)} قلم</span>
   508→                  </label>
   509→                  <span className="text-[11px] font-black text-[#8a6d10]">{faMoney(groupTotal)} تومان</span>
   510→                </div>
   511→                <div className="grid gap-1.5 sm:grid-cols-2">
   512→                  {ps.map((p) => (
   513→                    <div key={p.id} className={cn('flex items-center justify-between gap-2 rounded-xl border border-border/70 px-2.5 py-2', skipped[p.id] && 'opacity-40')}>
   514→                      <label className="flex min-w-0 cursor-pointer items-center gap-2">
   515→                        <input
   516→                          type="checkbox"
   517→                          checked={!skipped[p.id]}
   518→                          disabled={isExcluded}
   519→                          onChange={(e) => setSkipped((x) => ({ ...x, [p.id]: !e.target.checked }))}
   520→                          className="h-3.5 w-3.5 accent-[#0e7a4a]"
   521→                        />
   522→                        <span className="min-w-0">
   523→                          <span className="block truncate text-[11px] font-bold">{p.name}</span>
   524→                          <span className="block text-[9px] text-muted-foreground">
   525→                            موجودی {faNum(p.stock)} • نقطه سفارش {faNum(p.reorderLevel)} {p.stock === 0 ? '• 🔴 ناموجود' : ''}
   526→                          </span>
   527→                        </span>
   528→                      </label>
   529→                      <span className="flex shrink-0 items-center gap-1">
   530→                        <button
   531→                          disabled={isExcluded}
   532→                          onClick={() => setQtys((q) => ({ ...q, [p.id]: Math.max((q[p.id] || 1) - 1, 1) }))}

   535→                          −
   536→                        </button>

   554→

   562→            onClick={create}

  1208→            const found = Object.keys(r).find((rk) => rk.trim() === k || rk.includes(k))

  1224→          category: String(get('دسته', 'گروه', 'category') || 'عمومی'),
  1225→          duplicate: false,
  1226→        }
  1227→        return row
  1228→      }).filter((r: ImportRow) => r.name)
  1229→
  1230→      // duplicate detection
  1231→      const byBarcode = new Map<string, Product>()

  1244→    } catch (err: any) {
  1245→      toast.error('خواندن فایل ناموفق بود: ' + err.message)
  1246→    }

  1258→  return (
  1259→    <Modal title="ورود کالاها از فایل اکسل هلو" onClose={onClose} wide>
  1260→      {!rows.length ? (
  1261→        <div className="rounded-2xl border-2 border-dashed border-[#c9a227]/50 bg-[#fdf6dd]/40 p-8 text-center">

  1265→            ستون‌های رایج هلو (نام کالا، بارکد، کد، قیمت خرید، قیمت فروش، موجودی) به‌طور خودکار شناسایی می‌شوند و تکراری‌ها مشخص می‌گردند.
  1266→          </p>

  1329→function MergeWizard({ onClose, onDone }: { onClose: () => void; onDone: () => void }) {
  1330→  const [products, setProducts] = useState<Product[]>([])
  1331→  const [loaded, setLoaded] = useState(false)
  1332→
  1333→  // فهرست کامل از سرور (تشخیص تکراری سراسری، نه فقط صفحهٔ جاری)
  1334→  useEffect(() => {
  1335→    api<{ products: Product[] }>('/api/products')
  1336→      .then((d) => { setProducts(d.products); setLoaded(true) })
  1337→      .catch(() => setLoaded(true))
  1338→  }, [])
  1339→
  1340→  const groups = useMemo(() => {
  1341→    const byName = new Map<string, Product[]>()
  1342→    for (const p of products) {
  1343→      const key = p.name.replace(/\s+/g, ' ').trim().toLowerCase()
  1344→      const norm = key.replace(/[يیكک]/g, (m) => (m === 'ي' || m === 'ی' ? 'ی' : 'ک'))
  1345→      if (!byName.has(norm)) byName.set(norm, [])
  1346→      byName.get(norm)!.push(p)
  1347→    }
  1348→    return [...byName.values()].filter((g) => g.length > 1)
  1349→  }, [products])
  1350→  const [primary, setPrimary] = useState<Record<string, string>>({})
  1351→
  1352→  const merge = async (groupName: string, ids: string[]) => {
  1353→    const primaryId = primary[groupName] || ids[0]
  1354→    try {
  1355→      await api('/api/products/merge', { method: 'POST', body: { primaryId, duplicateIds: ids.filter((i) => i !== primaryId) } })
  1356→      toast.success('ادغام انجام شد ✅ — موجودی و بارکد‌ها ترکیب شدند')
  1357→      onDone()
  1358→    } catch (e: any) { toast.error(e.message) }
  1359→  }
  1360→
  1361→  return (
  1362→    <Modal title="پاک‌سازی کالاهای تکراری" onClose={onClose} wide>
  1363→      {!loaded ? (
  1364→        <div className="space-y-2">
  1365→          {Array.from({ length: 3 }).map((_, i) => <div key={i} className="h-16 animate-pulse rounded-xl bg-muted/60" />)}
  1366→        </div>
  1367→      ) : groups.length === 0 ? (
  1368→        <div className="py-6 text-center text-sm text-muted-foreground">
  1369→          <p className="text-3xl">✨</p>
