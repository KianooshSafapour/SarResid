    68→function lastCheckLabel(row: Row) {
    69→  if (row.checkedToday) return 'امروز کنترل شد ✓'
    70→  if (row.daysSince === null) return 'هرگز کنترل نشده'
    71→  if (row.daysSince === 1) return 'دیروز'
    72→  if (row.daysSince > 1) return `${faNum(row.daysSince)} روز پیش`
    73→  return '—'
    74→}

    76→/** tiny inline SVG trend of the product's last purchase costs — the inflation story at a glance */
    77→function CostSpark({ points }: { points: { price: number; j: string }[] }) {
    78→  if (points.length < 2) return null
    79→  const w = 56
    80→  const h = 20
    81→  const pad = 2
    82→  const min = Math.min(...points.map((p) => p.price))
    83→  const max = Math.max(...points.map((p) => p.price))
    84→  const span = max - min || 1
    85→  const step = (w - pad * 2) / (points.length - 1)
    86→  const coords = points.map((p, i) => [pad + i * step, h - pad - ((p.price - min) / span) * (h - pad * 2)] as const)
    87→  const d = coords.map((c, i) => `${i === 0 ? 'M' : 'L'}${c[0].toFixed(1)},${c[1].toFixed(1)}`).join(' ')
    88→  const up = points[points.length - 1].price >= points[0].price
    89→  const tip = points.map((p) => `${p.j}: ${faMoney(p.price)}`).join(' | ')
    90→  const gid = `spark-${points.length}-${min}-${max}`
    91→  return (
    92→    <svg width={w} height={h} viewBox={`0 0 ${w} ${h}`} className="spark-line shrink-0" role="img" aria-label="روند هزینه خرید">
    93→      <title>{tip}</title>
    94→      <defs>
    95→        <linearGradient id={gid} x1="0" y1="0" x2="1" y2="0">
    96→          <stop offset="0%" stopColor={up ? '#c9a227' : '#0e7a4a'} />
    97→          <stop offset="100%" stopColor={up ? '#b3372f' : '#12905a'} />
    98→        </linearGradient>
    99→      </defs>
   100→      <path d={d} fill="none" stroke={`url(#${gid})`} strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round" />
   101→      <circle cx={coords[coords.length - 1][0]} cy={coords[coords.length - 1][1]} r="2.2" fill={up ? '#b3372f' : '#0e7a4a'} />
   102→    </svg>
   103→  )
   104→}

   106→/** Code128 barcode rendered into a small SVG via JsBarcode — for the shelf label */
   107→function LabelBarcode({ value }: { value: string }) {
   108→  const ref = useRef<SVGSVGElement>(null)
   109→  useEffect(() => {
   110→    if (ref.current && value) {
   111→      try {
   112→        JsBarcode(ref.current, value, {
   113→          format: 'CODE128',
   114→          displayValue: false,
   115→          height: 30,
   116→          width: 1.05,
   117→          margin: 0,
   118→          lineColor: '#111111',
   119→        })
   120→      } catch {
   121→        /* invalid chars — leave the svg empty, the numeric text below is the fallback */
   122→      }
   123→    }
   124→  }, [value])
   125→  return <svg ref={ref} className="label-bc" />
   126→}
