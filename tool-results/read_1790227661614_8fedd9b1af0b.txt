    52→function catColor(cat: string): string {
    53→  let h = 0
    54→  for (let i = 0; i < cat.length; i++) h = (h * 31 + cat.charCodeAt(i)) >>> 0
    55→  return CAT_PALETTE[h % CAT_PALETTE.length]
    56→}

    58→const clampZoom = (z: number) => Math.min(ZOOM_MAX, Math.max(ZOOM_MIN, Math.round(z * 10) / 10))

    60→function toFacing(cell: CellVal): Facing {
    61→  if (typeof cell === 'string' && cell) return { id: cell, qty: 1 }
    62→  if (cell && typeof cell === 'object' && typeof cell.id === 'string' && cell.id)
    63→    return { id: cell.id, qty: Math.min(99, Math.max(1, Math.round(Number(cell.qty) || 1))) }
    64→  return null
    65→}
    66→
    67→/** normalize layout matrix to exact rows×cols facings (API may return ragged/empty) */
    68→function normalizeFacing(layout: CellVal[][] | undefined, rows: number, cols: number): Facing[][] {
    69→  const out: Facing[][] = []
    70→  for (let r = 0; r < rows; r++) {
    71→    const row: Facing[] = []
    72→    for (let c = 0; c < cols; c++) row.push(toFacing(layout?.[r]?.[c] ?? null))
    73→    out.push(row)
    74→  }
    75→  return out
    76→}
    77→
    78→/** ذخیره: qty=1 → رشتهٔ ساده (سازگار با FloorPlan)، غیر از آن → شیء */
    79→function serializeFacing(f: Facing): CellVal {
    80→  if (!f) return null
    81→  return f.qty === 1 ? f.id : { id: f.id, qty: f.qty }
    82→}
    83→
    84→/** ماتریس رشته‌ای قدیمی برای پیش‌نمایش فهرست */
    85→function normalize(layout: CellVal[][] | undefined, rows: number, cols: number): (string | null)[][] {
    86→  const out: (string | null)[][] = []
    87→  for (let r = 0; r < rows; r++) {
    88→    const row: (string | null)[] = []
    89→    for (let c = 0; c < cols; c++) {
    90→      const cell = layout?.[r]?.[c]
    91→      row.push(typeof cell === 'string' ? cell || null : cell && typeof cell === 'object' ? cell.id || null : null)
    92→    }
    93→    out.push(row)
    94→  }
    95→  return out
    96→}

   821→            </button>
   822→          )
   823→        }
   824→      >
   825→        {plans.length === 0 ? (
   826→          <EmptyState
   827→            emoji="🧱"
   828→            title="پلانوگرامی ثبت نشده"
   829→            hint={isManager ? 'با دکمهٔ «پلانوگرام جدید» اولین چیدمان قفسه را بسازید' : 'هنوز مدیریتی پلانوگرامی منتشر نکرده است'}
   830→          />
   831→        ) : (
   832→          <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-3">
   833→            {plans.map((pl, idx) => {
   834→              const st = PL_STATUS[pl.status] || PL_STATUS.DRAFT
   835→              const norm = normalize(pl.layout, pl.rows, pl.cols)
   836→              const red = norm.flat().filter((id) => id && pmap[id] && stockStatus(pmap[id].stock, pmap[id].reorderLevel).key === 'red').length
   837→              const green = norm.flat().filter((id) => id && pmap[id] && stockStatus(pmap[id].stock, pmap[id].reorderLevel).key === 'green').length
   838→              return (
   839→                <button
   840→                  key={pl.id}
   841→                  type="button"
   842→                  onClick={() => setEditingId(pl.id)}
   843→                  className="glow-card fade-in-up rounded-2xl bg-card p-4 text-right transition-all hover:-translate-y-0.5"
   844→                  style={{ animationDelay: `${Math.min(idx * 45, 360)}ms` }}
   845→                >
   846→                  <div className="flex items-start justify-between gap-2">
   847→                    <div>
   848→                      <p className="text-sm font-black text-foreground">{pl.name}</p>
   849→                      <p className="mt-0.5 text-[11px] text-muted-foreground">
   850→                        {pl.section ? `قسمت: ${pl.section} • ` : ''}
   851→                        {faNum(pl.rows)} قفسه × {faNum(pl.cols)} فیسینگ • {faNum(pl.cells)} کالا
   852→                      </p>
   853→                    </div>
   854→                    <Pill label={st.label} color={st.color} />
   855→                  </div>
   856→                  {/* mini preview */}
   857→                  <div className="mt-3 grid gap-0.5" style={{ gridTemplateColumns: `repeat(${pl.cols}, 1fr)`, maxWidth: 150 }}>
   858→                    {norm.flat().map((id, i) => {
   859→                      const p = id ? pmap[id] : undefined
   860→                      return (
   861→                        <span
   862→                          key={i}
   863→                          className="h-1.5 rounded-sm"
   864→                          style={{ background: p ? stockStatus(p.stock, p.reorderLevel).color : '#e7e1cf' }}
   865→                        />
   866→                      )
   867→                    })}
   868→                  </div>
   869→                  <div className="mt-3 flex flex-wrap gap-2 text-[10px] font-bold">
   870→                    <span className="rounded-full bg-[#b3372f]/10 px-2 py-0.5 text-[#b3372f]">قرمز {faNum(red)}</span>
   871→                    <span className="rounded-full bg-[#0e7a4a]/10 px-2 py-0.5 text-[#0e7a4a]">سبز {faNum(green)}</span>
   872→                  </div>
   873→                </button>
   874→              )
   875→            })}
   876→          </div>
   877→        )}
   878→      </SectionCard>
   879→
   880→      {newOpen && (
   881→        <NewPlanogramModal
   882→          onClose={() => setNewOpen(false)}
   883→          onCreated={async (id) => { setNewOpen(false); await load(); setEditingId(id) }}
   884→        />
   885→      )}
   886→    </div>
   887→  )
   888→}

   890→/* ─────────────────────────── new planogram modal ─────────────────────────── */
   891→function NewPlanogramModal({ onClose, onCreated }: { onClose: () => void; onCreated: (id: string) => void }) {
   892→  const [form, setForm] = useState({ name: '', section: '', rows: '3', cols: '6' })
   893→  const [busy, setBusy] = useState(false)
   894→  const set = (k: string, v: string) => setForm((f) => ({ ...f, [k]: v }))
   895→
   896→  const save = async () => {
   897→    if (!form.name.trim()) return toast.error('نام پلانوگرام الزامی است')
   898→    const rows = Math.min(12, Math.max(1, Number(form.rows) || 3))
   899→    const cols = Math.min(40, Math.max(1, Number(form.cols) || 6))
   900→    setBusy(true)
   901→    try {
   902→      const res = await api<{ planogram: Planogram }>('/api/planograms', {
   903→        method: 'POST',
   904→        body: { name: form.name.trim(), section: form.section.trim(), rows, cols },
   905→      })
   906→      toast.success('پلانوگرام ساخته شد — حالا فیسینگ‌ها را پر کنید 🧱')
   907→      onCreated(res.planogram.id)
   908→    } catch (e: any) {
   909→      toast.error(e.message)
   910→      setBusy(false)
   911→    }
   912→  }
   913→
   914→  return (
   915→    <Modal title="پلانوگرام جدید" onClose={onClose}>
   916→      <Labeled label="نام پلانوگرام *">
   917→        <input
   918→          value={form.name}
   919→          onChange={(e) => set('name', e.target.value)}
   920→          placeholder="مثلاً: قفسه لبنیات — ردیف شرقی"
   921→          className="w-full rounded-xl border border-input bg-white/90 p-3 text-sm"
   922→        />
   923→      </Labeled>
   924→      <Labeled label="قسمت فروشگاه">
   925→        <input
   926→          value={form.section}
   927→          onChange={(e) => set('section', e.target.value)}
   928→          placeholder="مثلاً: لبنیات"
   929→          className="w-full rounded-xl border border-input bg-white/90 p-3 text-sm"
   930→        />
   931→      </Labeled>
   932→      <div className="grid grid-cols-2 gap-3">
   933→        <Labeled label="تعداد قفسه / ردیف (۱ تا ۱۲)">
   934→          <input
   935→            type="number" dir="ltr" min={1} max={12}
   936→            value={form.rows}
   937→            onChange={(e) => set('rows', e.target.value)}
   938→            className="w-full rounded-xl border border-input bg-white/90 p-3 text-center text-sm font-black"
   939→          />
   940→        </Labeled>
   941→        <Labeled label="فیسینگ در هر قفسه (۱ تا ۴۰)">
   942→          <input
   943→            type="number" dir="ltr" min={1} max={40}
   944→            value={form.cols}
   945→            onChange={(e) => set('cols', e.target.value)}
   946→            className="w-full rounded-xl border border-input bg-white/90 p-3 text-center text-sm font-black"
   947→          />
   948→        </Labeled>
   949→      </div>
   950→      <button onClick={save} disabled={busy} className="w-full rounded-xl bg-primary py-3 text-sm font-extrabold text-white disabled:opacity-50">
   951→        {busy ? '…' : 'ساخت پلانوگرام 🧱'}
   952→      </button>
   953→    </Modal>
   954→  )
   955→}
