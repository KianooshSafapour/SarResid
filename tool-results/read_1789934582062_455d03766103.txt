   284→function DangerTab() {

   293→    return () => clearTimeout(t)
   294→  }, [armed])
   295→
   296→  const runAction = async (action: string, confirm?: string) => {
   297→    setBusy(true)
   298→    try {
   299→      const res = await api<{ ok: boolean; summary?: { deleted: Record<string, number> } }>('/api/admin/db', {
   300→        method: 'POST',
   301→        body: { action, confirm },
   302→      })
   303→      const del = res.summary?.deleted || {}
   304→      const keys = Object.keys(del)
   305→      if (keys.length) {
   306→        toast.success(`انجام شد — ${keys.map((k) => `${faNum(del[k])} «${TABLE_FA[k] || k}»`).join('، ')} حذف شد`)
   307→      } else {
   308→        toast.info('انجام شد — موردی برای حذف وجود نداشت')
   309→      }
   310→      setPurge(null)
   311→      setTyped('')
   312→      setArmed(null)
   313→    } catch (e) {
   314→      toast.error((e as Error).message)
   315→    } finally {
   316→      setBusy(false)
   317→    }
   318→  }
   319→
   320→  return (
   321→    <div className="space-y-5">
   322→      {/* هشدار برگشت‌ناپذیری */}
   323→      <div className="flex items-start gap-3 rounded-2xl border border-[#b3372f]/40 bg-[#b3372f]/8 p-4 fade-in-up">

   325→        <div>
   326→          <p className="text-sm font-black text-[#b3372f]">منطقهٔ خطر — این عملیات‌ها برگشت‌ناپذیرند</p>

   443→const MAP_KEY = (kind: string) => `hz-import-map-${kind}`

   445→const normHeader = (s: string) => s.toLowerCase().replace(/\u200c/g, '').replace(/\s+/g, ' ').trim()

   447→function autoGuess(headers: string[], kind: string): Record<string, number> {
   448→  const aliases = GUESS[kind] || {}
   449→  const map: Record<string, number> = {}
   450→  const claimed = new Set<number>()
   451→  // گذر ۱ — تطبیق دقیق
   452→  for (const key of Object.keys(aliases)) {
   453→    for (const a of aliases[key]) {
   454→      const i = headers.findIndex((h, idx) => !claimed.has(idx) && normHeader(h) === normHeader(a))
   455→      if (i >= 0) { map[key] = i; claimed.add(i); break }
   456→    }
   457→  }
   458→  // گذر ۲ — تطبیق جزئی (شامل‌مشتق دوطرفه با حداقل طول)
   459→  for (const key of Object.keys(aliases)) {
   460→    if (map[key] !== undefined) continue
   461→    for (const a of aliases[key]) {
   462→      const needle = normHeader(a)
   463→      if (needle.length < 3) continue
   464→      const i = headers.findIndex((h, idx) => {
   465→        if (claimed.has(idx)) return false
   466→        const n = normHeader(h)
   467→        return n.length >= 3 && (n.includes(needle) || needle.includes(n))
   468→      })
   469→      if (i >= 0) { map[key] = i; claimed.add(i); break }
   470→    }
   471→  }
   472→  return map
   473→}

   890→/** دکمهٔ تسلیح‌شونده — کلیک اول قرمز می‌شود («مطمئنید؟ دوباره بزنید») و کلیک دوم اجرا می‌کند */
   891→function ArmedButton({
   892→  label,
   893→  armedLabel = 'مطمئنید؟ دوباره بزنید',
   894→  onClick,
   895→  disabled,
   896→  className,
   897→  tone = 'rose',
   898→}: {
   899→  label: string
   900→  armedLabel?: string
   901→  onClick: () => void
   902→  disabled?: boolean
   903→  className?: string
   904→  tone?: 'rose' | 'gold'
   905→}) {
   906→  const [armed, setArmed] = useState(false)
   907→  useEffect(() => {
   908→    if (!armed) return
   909→    const t = setTimeout(() => setArmed(false), 5000)
   910→    return () => clearTimeout(t)
   911→  }, [armed])
   912→  return (
   913→    <button
   914→      type="button"
   915→      disabled={disabled}
   916→      onClick={() => {
   917→        if (armed) {
   918→          setArmed(false)
   919→          onClick()
   920→        } else {
   921→          setArmed(true)
   922→        }
   923→      }}
   924→      className={cn(
   925→        'rounded-xl px-4 py-2.5 text-xs font-extrabold transition-all disabled:opacity-40',
   926→        armed
   927→          ? 'animate-pulse bg-[#b3372f] text-white shadow-lg ring-2 ring-[#b3372f]/40'
   928→          : tone === 'rose'
   929→            ? 'border border-[#b3372f]/40 bg-[#b3372f]/10 text-[#b3372f] hover:bg-[#b3372f]/20'
   930→            : 'border border-[#c9a227]/50 bg-[#c9a227]/10 text-[#8a6d10] hover:bg-[#c9a227]/20',
   931→        className
   932→      )}
   933→    >
   934→      {armed ? armedLabel : label}
   935→    </button>
   936→  )
   937→}
   938→
   939→/** آمار کوچک خلاصهٔ نتیجه */

   944→      <p className="text-[10px] font-bold text-muted-foreground">{label}</p>
   945→    </div>
   946→  )
   947→}
