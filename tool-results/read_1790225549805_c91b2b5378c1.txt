   101→/* ── 17-E — دقت دقیقه در مرخصی ساعتی: "HH:MM" → دقیقه ── */
   102→/** "HH:MM" → دقیقه از نیمه‌شب؛ نامعتبر → null */
   103→function hmToMin(s: string): number | null {
   104→  const m = /^(\d{1,2}):(\d{2})$/.exec(String(s || '').trim())
   105→  if (!m) return null
   106→  const h = Number(m[1])
   107→  const mm = Number(m[2])
   108→  if (h > 23 || mm > 59) return null
   109→  return h * 60 + mm
   110→}
   111→/** مدت مرخصی ساعتی به دقیقه — بازهٔ نامعتبر → ۰ */
   112→function hourlyMinutes(from: string, to: string): number {
   113→  const a = hmToMin(from)
   114→  const b = hmToMin(to)
   115→  if (a == null || b == null) return 0
   116→  return Math.max(0, b - a)
   117→}
   118→/** «۱ ساعت و ۴۵ دقیقه» با ارقام فارسی */
   119→function faMinLong(minutes: number): string {
   120→  const m = Math.max(0, Math.round(minutes || 0))
   121→  const h = Math.floor(m / 60)
   122→  const r = m % 60
   123→  if (h && r) return `${faNum(h)} ساعت و ${faNum(r)} دقیقه`
   124→  if (h) return `${faNum(h)} ساعت`
   125→  if (r) return `${faNum(r)} دقیقه`
   126→  return '۰ دقیقه'
   127→}
   128→/** «۱:۴۵» فشرده با ارقام فارسی */
   129→function faHm(minutes: number): string {
   130→  const m = Math.max(0, Math.round(minutes || 0))
   131→  return `${faNum(Math.floor(m / 60))}:${faNum(String(m % 60).padStart(2, '0'))}`
   132→}

   816→      {addOpen && (
   817→        <Modal title="درخواست مرخصی جدید" onClose={() => setAddOpen(false)}>
   818→          <Labeled label="نوع مرخصی">
   819→            <div className="grid grid-cols-2 gap-2">
   820→              {Object.entries(TYPE_META).map(([key, meta]) => (
   821→                <button
   822→                  key={key}
   823→                  type="button"
   824→                  onClick={() => pickType(key)}
   825→                  className={cn(
   826→                    'rounded-xl border px-3 py-2.5 text-sm font-bold transition-all',
   827→                    type === key ? 'border-transparent text-white shadow-md' : 'border-border bg-white/80 hover:bg-secondary'
   828→                  )}
   829→                  style={type === key ? { background: meta.color } : undefined}
   830→                >
   831→                  {meta.emoji} {meta.label}
   832→                </button>
   833→              ))}
   834→            </div>
   835→          </Labeled>
   836→
   837→          {type === 'HOURLY' ? (
   838→            <>
   839→              <Labeled label="روز مرخصی (ساعتی — فقط همان روز)" hint="مرخصی ساعتی معادل نیم‌روز محاسبه می‌شود؛ مدت دقیق با دقیقه سنجیده و در حقوق لحاظ می‌شود">
   840→                <JalaliDatePicker value={fromDate} onChange={pickFrom} holidays={holidays} minDate={addDaysIso(-1)} warnHoliday={false} />
   841→              </Labeled>
   842→              <div className="grid grid-cols-2 gap-2">
   843→                <Labeled label="از ساعت" hint="دقیقه‌دار — مثلاً ۱۳:۳۰">
   844→                  <input type="time" step={60} dir="ltr" value={fromHour} onChange={(e) => setFromHour(e.target.value)} className={cn(INPUT_CLS, 'text-center font-bold')} />
   845→                </Labeled>
   846→                <Labeled label="تا ساعت" hint="دقیقه‌دار — مثلاً ۱۵:۱۵">
   847→                  <input type="time" step={60} dir="ltr" value={toHour} onChange={(e) => setToHour(e.target.value)} className={cn(INPUT_CLS, 'text-center font-bold')} />
   848→                </Labeled>
   849→              </div>
   850→              {(() => {
   851→                const mins = hourlyMinutes(fromHour, toHour)
   852→                const valid = hmToMin(fromHour) != null && hmToMin(toHour) != null && fromHour < toHour
   853→                return (
   854→                  <p className={cn('rounded-xl px-3 py-2 text-[11px] font-bold', valid && mins > 0 ? 'bg-[#0e7a4a]/10 text-[#0e7a4a]' : 'bg-muted/60 text-muted-foreground')}>
   855→                    مدت محاسبه‌شده (زنده): {valid && mins > 0 ? `${faHm(mins)} — ${faMinLong(mins)}` : '—'}
   856→                  </p>
   857→                )
   858→              })()}
   859→            </>
   860→          ) : (
   861→            <>
   862→              <Labeled label="از تاریخ">
   863→                <JalaliDatePicker value={fromDate} onChange={pickFrom} holidays={holidays} minDate={addDaysIso(-1)} warnHoliday={false} />
   864→              </Labeled>
   865→              <Labeled label="تا تاریخ">
   866→                <JalaliDatePicker value={toDate} onChange={setToDate} holidays={holidays} minDate={fromDate} warnHoliday={false} />
   867→              </Labeled>
   868→            </>
   869→          )}
   870→
   871→          <Labeled label="دلیل (اختیاری)">
   872→            <textarea
   873→              value={reason}
   874→              onChange={(e) => setReason(e.target.value)}
   875→              rows={2}
   876→              maxLength={300}
   877→              placeholder="مثلاً کار اداری بانک"
   878→              className={cn(INPUT_CLS, 'resize-none')}
   879→            />
   880→          </Labeled>
   881→
   882→          <p className="rounded-xl bg-[#c9a227]/10 px-3 py-2 text-[11px] font-bold leading-5 text-[#8a6d10]">
   883→            نوبت‌بندی به‌صورت «اولین ثبت، اولویت» است؛ ثبت درخواست به معنای تأیید نیست و تصمیم نهایی با مدیر عملیات است.
   884→          </p>
   885→
   886→          <button
   887→            type="button"
   888→            disabled={saving}
   889→            onClick={submit}
   890→            className="w-full rounded-xl bg-[#0e7a4a] py-3 text-sm font-black text-white shadow-md transition hover:bg-[#0b6a40] disabled:opacity-60"
   891→          >
   892→            {saving ? 'در حال ثبت…' : 'ثبت درخواست'}
   893→          </button>
   894→        </Modal>
   895→      )}
   896→    </div>
   897→  )
   898→}
