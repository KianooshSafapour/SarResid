    65→export default function PeopleView({ ctx }: { ctx: AppCtx } & Record<string, unknown>) {
    66→  const user = ctx.user
    67→  const roleKeys = useMemo(
    68→    () => (user ? Array.from(new Set([user.role, ...(user.secondaryRoles || [])])) : []),
    69→    [user],
    70→  )
    71→
    72→  const [tab, setTab] = useState<Tab>('reps')
    73→  const [roleMap, setRoleMap] = useState<Record<string, RoleLite>>({})
    74→
    75→  const canManage = useMemo(() => {
    76→    if (!user) return false
    77→    if (MANAGE_ROLES.includes(user.role)) return true
    78→    return roleKeys.some((k) => roleMap[k]?.caps?.includes('*') || roleMap[k]?.caps?.includes('people.manage'))
    79→  }, [user, roleKeys, roleMap])
    80→
    81→  const canMfg = useMemo(() => {
    82→    if (!user) return false
    83→    if (MANAGE_ROLES.includes(user.role) || user.role === 'ACC') return true
    84→    return roleKeys.some((k) => roleMap[k]?.caps?.includes('*') || roleMap[k]?.caps?.includes('manufacturers.manage'))
    85→  }, [user, roleKeys, roleMap])
    86→
    87→  // ── deep-link شبکهٔ اشخاص (round-18-C) ──
    88→  // navigate('people', repId) → تب «نمایندگان و اشخاص» باز می‌شود و پروندهٔ همان شخص
    89→  // خودکار بالا می‌آید (لینک «📇 پرونده در دفتر» از کارت تأمین‌کننده و نتایج PeopleNetBar).
    90→  // الگوی رسمی «تنظیم state در رندر» — سازگار با قاعدهٔ set-state-in-effect.
    91→  const [focusId, setFocusId] = useState('')
    92→  const [lastParam, setLastParam] = useState('')
    93→  if (ctx.param !== lastParam) {
    94→    setLastParam(ctx.param)
    95→    if (ctx.view === 'people' && ctx.param) {
    96→      setFocusId(ctx.param)
    97→      if (tab !== 'reps') setTab('reps')
    98→    } else if (!ctx.param) {
    99→      setFocusId('')
   100→    }
   101→  }
   102→
   103→  useEffect(() => {
   104→    api<{ roles: Array<{ key: string; active: boolean; permissions?: { caps?: string[] } }> }>('/api/roles')
   105→      .then((d) => {
   106→        const m: Record<string, RoleLite> = {}
   107→        for (const r of d.roles || []) m[r.key] = { key: r.key, active: r.active, caps: r.permissions?.caps || [] }
   108→        setRoleMap(m)
   109→      })
   110→      .catch(() => { /* fallback to role list above */ })
   111→  }, [])
   112→
   113→  return (
   114→    <div className="space-y-4">
   115→      {/* نوار مشترک شبکهٔ اشخاص — دفتر اشخاص + تأمین‌کنندگان + CRM (round-18-C) */}
   116→      <PeopleNetBar active="people" ctx={ctx} />
   117→      <div className="flex flex-wrap gap-2 rounded-2xl border border-[#8a5a2b]/25 bg-[#f7efe2] p-1.5">
   118→        {([
   119→          ['reps', <Users key="i" size={15} />, 'نمایندگان و اشخاص'],
   120→          ['manufacturers', <Factory key="i" size={15} />, 'تولیدکنندگان'],
   121→          ['guide', <BookOpen key="i" size={15} />, 'راهنما'],
   122→        ] as [Tab, React.ReactNode, string][]).map(([k, icon, label]) => (
   123→          <button
   124→            key={k}
   125→            type="button"
   126→            onClick={() => setTab(k)}
   127→            className={cn(
   128→              'flex min-h-[44px] items-center gap-1.5 rounded-xl px-4 py-2 text-xs font-extrabold transition',
   129→              tab === k ? 'bg-white text-[#0e7a4a] shadow' : 'text-[#8a5a2b] hover:bg-white/60',
   130→            )}
   131→          >
   132→            {icon} {label}
   133→          </button>
   134→        ))}
   135→      </div>
   136→
   137→      {tab === 'reps' && <RepDirectory canManage={canManage} ctx={ctx} focusId={focusId} />}
   138→      {tab === 'manufacturers' && <ManufacturerBoard canManage={canMfg} />}
   139→      {tab === 'guide' && <GuideCard />}
   140→    </div>
   141→  )
   142→}

   149→  const [q, setQ] = useState('')
   150→  const [kindFilter, setKindFilter] = useState('')
   151→  const [providerFilter, setProviderFilter] = useState('')
   152→  const [loading, setLoading] = useState(true)
   153→
   154→  const [formOpen, setFormOpen] = useState(false)
   155→  const [editing, setEditing] = useState<Rep | null>(null)
   156→  const [profileId, setProfileId] = useState('')
   157→  const [profile, setProfile] = useState<ProfileRes | null>(null)
   158→
   159→  const load = useCallback(async () => {
   160→    setLoading(true)
   161→    try {
   162→      const params = new URLSearchParams()
   163→      params.set('all', '1')
   164→      if (q.trim()) params.set('q', q.trim())
   165→      if (kindFilter) params.set('kind', kindFilter)
   166→      if (providerFilter) params.set('providerId', providerFilter)
   167→      const d = await api<{ reps: Rep[] }>(`/api/sales-reps?${params}`)
   168→      setReps(d.reps || [])
   169→    } catch (e: any) {
   170→      toast.error(e.message)
   171→    } finally {
   172→      setLoading(false)
   173→    }
   174→  }, [q, kindFilter, providerFilter])
   175→
   176→  useEffect(() => { load() }, [load])
   177→  useEffect(() => {
   178→    api<{ providers: ProviderLite[] }>('/api/providers').then((d) => setProviders(d.providers || [])).catch(() => { /* free-text still works */ })
   179→  }, [])
   180→
   181→  const openProfile = async (id: string) => {
   182→    setProfile(null)

   194→  // پس از بستن پرونده (بدون تعویض نما) دوباره باز نمی‌شود چون param ناوبری تغییری نمی‌کند.
   195→  const [lastFocus, setLastFocus] = useState('')
   196→  if (focusId && focusId !== lastFocus) {

   199→  }

   203→      await api(`/api/sales-reps?id=${r.id}`, { method: 'PATCH', body: { active: !r.active } })

   211→    try {
   212→      const res = await api<{ mode: string; message?: string }>(`/api/sales-reps?id=${r.id}`, { method: 'DELETE' })
   213→      if (res.mode === 'soft') toast.warning(res.message || 'شخص دارای رخداد است — غیرفعال شد', { duration: 7000 })

   217→  }
   218→
   219→  return (

   224→        icon={<Users size={18} />}
   225→        actions={canManage && (
   226→          <button onClick={() => { setEditing(null); setFormOpen(true) }} className="flex items-center gap-1.5 rounded-xl bg-primary px-4 py-2 text-xs font-extrabold text-white">

   655→      if (mfg) await api(`/api/manufacturers/${mfg.id}`, { method: 'PATCH', body: { ...f, brands } })
   656→      else await api('/api/manufacturers', { method: 'POST', body: { ...f, brands } })
   657→      onSaved()
   658→    } catch (e: any) {
   659→      toast.error(e.message)
   660→    } finally {
   661→      setBusy(false)
   662→    }
   663→  }
   664→
   665→  return (
   666→    <Modal title={mfg ? `ویرایش تولیدکننده — ${mfg.name}` : 'ثبت تولیدکنندهٔ جدید'} onClose={onClose} wide>
   667→      <div className="grid gap-3 sm:grid-cols-2">
   668→        <Labeled label="نام تولیدکننده *"><input value={f.name} onChange={(e) => set('name', e.target.value)} className="w-full rounded-xl border border-input px-2.5 py-2 text-[13px]" placeholder="مثلاً: کارtière پگاه" /></Labeled>
   669→        <Labeled label="کشور"><input value={f.country} onChange={(e) => set('country', e.target.value)} className="w-full rounded-xl border border-input px-2.5 py-2 text-[13px]" /></Labeled>
   670→        <Labeled label="وب‌سایت"><input value={f.website} onChange={(e) => set('website', e.target.value)} className="w-full rounded-xl border border-input px-2.5 py-2 text-[13px]" dir="ltr" placeholder="example.ir" /></Labeled>
   671→        <Labeled label="تلفن"><input value={f.phone} onChange={(e) => set('phone', e.target.value)} className="w-full rounded-xl border border-input px-2.5 py-2 text-[13px]" dir="ltr" /></Labeled>
   672→        <Labeled label="برندها" hint="نام برند + Enter — با برندِ کالاها تطبیق داده می‌شود تا تعداد محصولات شمرده شود">
   673→          <input value={brandInput} onChange={(e) => setBrandInput(e.target.value)} onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); addBrand() } }} className="w-full rounded-xl border border-input px-2.5 py-2 text-[13px]" placeholder="مثلاً: پگاه" />
   674→        </Labeled>
   675→        <Labeled label="یادداشت"><input value={f.notes} onChange={(e) => set('notes', e.target.value)} className="w-full rounded-xl border border-input px-2.5 py-2 text-[13px]" /></Labeled>
   676→      </div>
   677→      {brands.length > 0 && (
   678→        <div className="flex flex-wrap gap-1.5">
   679→          {brands.map((b) => (
   680→            <button key={b} type="button" onClick={() => setBrands((l) => l.filter((x) => x !== b))} className="rounded-full bg-[#c9a227]/15 px-2.5 py-1 text-[10px] font-black text-[#8a5a2b]">
   681→              {b} ×
   682→            </button>
   683→          ))}
   684→        </div>
   685→      )}
   686→      <button onClick={save} disabled={busy} className="mt-3 w-full rounded-xl bg-primary py-3 text-sm font-extrabold text-white disabled:opacity-50">
   687→        {busy ? '…' : mfg ? 'ذخیرهٔ تغییرات' : 'ثبت تولیدکننده'}
   688→      </button>
   689→    </Modal>
   690→  )
   691→}

   693→/** اتصال/جداکردن تأمین‌کنندگان به تولیدکننده (multi-select) */
   694→function AssignProvidersModal({ mfg, providers, onClose, onSaved }: { mfg: Manufacturer; providers: ProviderLite[]; onClose: () => void; onSaved: () => void }) {
   695→  const [selected, setSelected] = useState<string[]>(mfg.providers.map((p) => p.id))
   696→  const [busy, setBusy] = useState(false)
   697→  const toggle = (id: string) => setSelected((s) => (s.includes(id) ? s.filter((x) => x !== id) : [...s, id]))
   698→
   699→  const save = async () => {
   700→    setBusy(true)
   701→    try {
   702→      await api(`/api/manufacturers/${mfg.id}`, { method: 'PATCH', body: { action: 'assign-providers', providerIds: selected } })
   703→      onSaved()
   704→    } catch (e: any) {
   705→      toast.error(e.message)
   706→    } finally {
   707→      setBusy(false)
   708→    }
   709→  }
   710→
   711→  return (
   712→    <Modal title={`تأمین‌کنندگان ${mfg.name}`} onClose={onClose} wide>
   713→      <p className="rounded-xl bg-[#fdf6dd] p-3 text-[11px] font-bold text-[#8a5a2b]">
   714→        <Building2 size={13} className="ml-1 inline" />
   715→        هر تأمین‌کننده/پخش را که محصولات این تولیدکننده را پشتیبانی می‌کند، علامت بزنید — تغییرات در یک فراخوان ذخیره می‌شود.
   716→      </p>
   717→      {providers.length === 0 ? (
   718→        <EmptyState emoji="🚛" title="تأمین‌کننده‌ای ثبت نشده" hint="ابتدا از نمای «تأمین‌کنندگان» پخش‌ها را ثبت کنید" />
   719→      ) : (
   720→        <div className="scroll-gold max-h-[42vh] space-y-1.5 overflow-y-auto pl-1">
   721→          {providers.map((p) => {
   722→            const on = selected.includes(p.id)
   723→            return (
   724→              <button
   725→                key={p.id}
   726→                type="button"
   727→                onClick={() => toggle(p.id)}
   728→                className={cn(
   729→                  'flex w-full items-center justify-between gap-2 rounded-xl bg-white/85 p-2.5 text-right transition hover:shadow-sm',
   730→                  on && 'ring-2 ring-[#0e7a4a]/60',
   731→                )}
   732→              >
   733→                <span className="text-xs font-black">{p.name}</span>
   734→                <span className={cn('rounded-full px-2.5 py-0.5 text-[10px] font-black', on ? 'bg-[#0e7a4a] text-white' : 'bg-muted text-muted-foreground')}>
   735→                  {on ? '✓ متصل' : 'غیرمتصل'}
   736→                </span>
   737→              </button>
   738→            )
   739→          })}
   740→        </div>
   741→      )}
   742→      <button onClick={save} disabled={busy} className="mt-3 w-full rounded-xl bg-primary py-3 text-sm font-extrabold text-white disabled:opacity-50">
   743→        {busy ? '…' : `ذخیره (${faNum(selected.length)} تأمین‌کننده متصل)`}
   744→      </button>
   745→    </Modal>
   746→  )
   747→}
