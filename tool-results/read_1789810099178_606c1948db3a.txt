   215→function AwardSection({ leaderboard, onDone }: { leaderboard: LBRow[]; onDone: () => void }) {
   216→  const [targetId, setTargetId] = useState('')

   219→  const [reason, setReason] = useState('')
   220→  const [busy, setBusy] = useState(false)
   221→
   222→  const effective = Number(custom) > 0 ? Number(custom) : selPoints
   223→
   224→  const submit = async () => {
   225→    if (!targetId) return toast.error('همکار را انتخاب کنید')

   229→    try {
   230→      await api('/api/gamification/award', { method: 'POST', body: { userId: targetId, points: effective, reason: reason.trim() } })
   231→      toast.success('امتیاز اهدا شد — همکارتان مطلع می‌شود 🎉')
   232→      setTargetId('')
   233→      setSelPoints(null)
   234→      setCustom('')
   235→      setReason('')

   238→      toast.error(e.message)
   239→    } finally {
   240→      setBusy(false)
   241→    }
   242→  }
   243→
   244→  return (
   245→    <SectionCard
   246→      title="اعطای امتیاز به همکار 🎁"

   263→              ))}
   264→            </select>
   265→          </Labeled>
   266→
   267→          <Labeled label="تعداد امتیاز *" hint="می‌توانید عدد دلخواه هم وارد کنید">

   269→              {[5, 10, 15, 20, 25, 50].map((p) => (
   270→                <button
   271→                  key={p}
   272→                  type="button"
   273→                  onClick={() => { setSelPoints(p); setCustom('') }}
   274→                  className={cn(
   275→                    'rounded-full px-4 py-2 text-xs font-black transition',
   276→                    effective === p && !custom
   277→                      ? 'bg-primary text-white shadow-md'

   281→                  {faNum(p)}

   350→function SplhSection({ ctx }: { ctx: AppCtx }) {
   351→  const [data, setData] = useState<SplhData | null>(null)
   352→  const [days, setDays] = useState(30)
   353→  const [users, setUsers] = useState<{ id: string; name: string; role: string }[]>([])
   354→  const [busy, setBusy] = useState(false)
   355→  const [edUserId, setEdUserId] = useState('')
   356→  const [edDate, setEdDate] = useState(todayIso())
   357→  const [edHours, setEdHours] = useState<number | ''>('')
   358→  const [edKind, setEdKind] = useState<'REGULAR' | 'OVERTIME'>('REGULAR')
   359→  const [edNote, setEdNote] = useState('')
   360→  const [targetDraft, setTargetDraft] = useState<number | ''>('')
   361→
   362→  const load = useCallback(async (d: number) => {
   363→    try {
   364→      const res = await api<SplhData>(`/api/splh?days=${d}`)
   365→      setData(res)
   366→      setTargetDraft(res.totals.target)
   367→    } catch (e: any) {
   368→      toast.error(e.message)
   369→    }
   370→  }, [])
   371→  useEffect(() => {
   372→    load(days)
   373→  }, [days, load])
   374→
   375→  // فهرست همکاران فعال — فقط برای ویرایشگرها
   376→  useEffect(() => {
   377→    if (!data?.canEdit) return
   378→    api<{ users: { id: string; name: string; role: string; active: boolean }[] }>('/api/users')
   379→      .then((r) => setUsers(r.users.filter((u) => u.active)))
   380→      .catch(() => {})
   381→  }, [data?.canEdit])
   382→
   383→  const submitHours = async () => {
   384→    if (!edUserId) return toast.error('همکار را انتخاب کنید')
   385→    if (!edDate) return toast.error('تاریخ را انتخاب کنید')
   386→    const h = Number(edHours)
   387→    if (!h || h <= 0 || h > 16) return toast.error('ساعت کار باید بیشتر از صفر و حداکثر ۱۶ باشد')
   388→    setBusy(true)
   389→    try {
   390→      await api('/api/splh', { method: 'POST', body: { userId: edUserId, forDate: edDate, hours: h, kind: edKind, note: edNote.trim() } })
   391→      toast.success('ساعت کار ثبت شد ✅')
   392→      setEdHours('')
   393→      setEdNote('')
   394→      load(days)
   395→    } catch (e: any) {
   396→      toast.error(e.message)
   397→    } finally {
   398→      setBusy(false)
   399→    }
   400→  }
   401→
   402→  const removeLog = async (id: string) => {
   403→    setBusy(true)
   404→    try {
   405→      await api(`/api/splh?id=${id}`, { method: 'DELETE' })
   406→      toast.success('رکورد ساعت کار حذف شد')
   407→      load(days)
   408→    } catch (e: any) {
   409→      toast.error(e.message)
   410→    } finally {
   411→      setBusy(false)
   412→    }
   413→  }
   414→
   415→  const saveTarget = async () => {
   416→    const t = Number(targetDraft)
   417→    if (!t || t <= 0) return toast.error('هدف را وارد کنید (تومان بر ساعت)')
   418→    setBusy(true)
   419→    try {
   420→      await api('/api/splh', { method: 'PUT', body: { target: t } })
   421→      toast.success('هدف بهره‌وری ذخیره شد 🎯')
   422→      load(days)
   423→    } catch (e: any) {
   424→      toast.error(e.message)
   425→    } finally {
   426→      setBusy(false)
   427→    }
   428→  }
